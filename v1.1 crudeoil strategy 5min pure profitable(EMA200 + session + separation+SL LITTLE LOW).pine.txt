//@version=5
strategy("EMA 9/22 Pullback Continuation (5m) v6.2", overlay=true,
     initial_capital=100000,
     default_qty_type=strategy.fixed, default_qty_value=10,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     max_lines_count=500, max_labels_count=500)

// ============ INPUTS ============
emaFastLen  = input.int(9,   "EMA Fast")
emaMedLen   = input.int(22,  "EMA Med")
emaSlowLen  = input.int(200, "EMA Slow")
use200      = input.bool(true, "R1: only trade with 5m EMA200")
useHA       = input.bool(false, "Require raw HA candle colour")
pbToEma22   = input.bool(false, "Pullback must reach EMA22 (else EMA9)")
sepMlt      = input.float(0.35, "R3: min EMA9-EMA22 separation (x ATR14, 0=off)", step=0.05)
allowCoil   = input.bool(true, "COIL: trade the EMA-convergence break")
clusterMlt  = input.float(1.0, "  coil: 3-EMA span <= x ATR14", step=0.1)
clusterBuf  = input.float(0.25, "  coil SL buffer beyond band (x ATR14)", step=0.05)
useSession  = input.bool(true, "R4: restrict to session hours")
sessTime    = input.session("1200-2330", "Session (exchange time)")
rsiLen      = input.int(3,   "RSI Length")
rsiMaxLong  = input.int(85,  "No long if RSI(3) above")
rsiMinShort = input.int(15,  "No short if RSI(3) below")
pbLookback  = input.int(10,  "Pullback must start within N bars of an above-both close")
reclaimWin  = input.int(8,   "Max bars: touch -> reclaim close -> entry")
coolBars    = input.int(0,   "Cooldown bars after an exit")
rMultiple   = input.float(2.0, "R:R (target)", step=0.1)
atrBuffMlt   = input.float(0.15, "SL buffer beyond pullback low (x ATR14)", step=0.05)
stopFloorATR = input.float(0.5,  "Min SL distance floor (x ATR14, 0=off)", step=0.05)
tgtFromTight = input.bool(true,  "Target distance from PRE-floor (tight) risk")
minStopATR  = input.float(0.0, "Reject if SL distance < (x ATR14, 0=off)", step=0.05)
maxStopATR  = input.float(2.0, "Max stop distance (x ATR14, 0=off)", step=0.25)
showBG      = input.bool(true, "Trend background")
showMarks   = input.bool(true, "Entry/exit marks")
showDebug   = input.bool(false, "DEBUG tags")

// ============ INDICATORS ============
ema9   = ta.ema(close, emaFastLen)
ema22  = ta.ema(close, emaMedLen)
ema200 = ta.ema(close, emaSlowLen)
rsi3   = ta.rsi(close, rsiLen)
atr14  = ta.atr(14)

haClose = (open + high + low + close) / 4
var float haOpen = na
haOpen := na(haOpen[1]) ? (open + close) / 2 : (haOpen[1] + haClose[1]) / 2
haBull = haClose > haOpen
haBear = haClose < haOpen

emaHi3 = math.max(ema9, math.max(ema22, ema200))
emaLo3 = math.min(ema9, math.min(ema22, ema200))
coilOn = allowCoil and (emaHi3 - emaLo3) <= clusterMlt * atr14

inSession = not useSession or not na(time(timeframe.period, sessTime))
emaSepOK  = sepMlt == 0 or math.abs(ema9 - ema22) >= sepMlt * atr14 or coilOn

// ============ 5m REGIME ============
regimeUp = (ema9 > ema22 or coilOn) and (not use200 or close > ema200)
regimeDn = (ema9 < ema22 or coilOn) and (not use200 or close < ema200)
// normal: close reclaims both EMAs.  coil: decisive close beyond the WHOLE band.
aboveBoth = coilOn ? close > emaHi3 : (close > ema9 and close > ema22)
belowBoth = coilOn ? close < emaLo3 : (close < ema9 and close < ema22)
pbLvlL = pbToEma22 ? ema22 : ema9
pbLvlS = pbToEma22 ? ema22 : ema9

// ============ PULLBACK TRACKER ============
var bool  pbActL   = false
var float pbLowL   = na
var int   pbBarL   = na
var int   lastAbvL = na
var bool  pbActS   = false
var float pbHiS    = na
var int   pbBarS   = na
var int   lastBlwS = na

if not regimeUp
    pbActL := false
    pbLowL := na
    pbBarL := na
else
    if aboveBoth
        lastAbvL := bar_index
    if low <= pbLvlL and not na(lastAbvL) and (bar_index - lastAbvL) <= pbLookback
        fresh = not pbActL or (not na(pbBarL) and lastAbvL >= pbBarL)
        pbLowL := fresh ? low : math.min(pbLowL, low)
        pbActL := true
        pbBarL := bar_index

if not regimeDn
    pbActS := false
    pbHiS  := na
    pbBarS := na
else
    if belowBoth
        lastBlwS := bar_index
    if high >= pbLvlS and not na(lastBlwS) and (bar_index - lastBlwS) <= pbLookback
        fresh = not pbActS or (not na(pbBarS) and lastBlwS >= pbBarS)
        pbHiS := fresh ? high : math.max(pbHiS, high)
        pbActS := true
        pbBarS := bar_index

// ============ COOLDOWN ============
var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars

// ============ TRIGGER SETUP ============
var bool  waitL   = false
var int   waitBarL = na
var float trigL   = na
var float slL     = na
var float tpL     = na
var bool  waitS   = false
var int   waitBarS = na
var float trigS   = na
var float slS     = na
var float tpS     = na

haOKL = not useHA or haBull
haOKS = not useHA or haBear
gateL = regimeUp and aboveBoth and haOKL and emaSepOK and inSession and rsi3 < rsiMaxLong
gateS = regimeDn and belowBoth and haOKS and emaSepOK and inSession and rsi3 > rsiMinShort

if strategy.opentrades == 0 and coolOK and not waitL and pbActL and gateL and not na(pbBarL) and (bar_index - pbBarL) <= reclaimWin
    _slTight = coilOn ? (emaLo3 - clusterBuf * atr14) : (pbLowL - atrBuffMlt * atr14)
    _sl      = stopFloorATR > 0 ? math.min(_slTight, high - stopFloorATR * atr14) : _slTight
    _riskSL  = high - _sl
    _riskTgt = tgtFromTight ? (high - _slTight) : _riskSL
    if _riskSL > 0 and (minStopATR == 0 or _riskSL >= minStopATR * atr14) and (maxStopATR == 0 or _riskSL <= maxStopATR * atr14)
        waitL := true
        waitBarL := bar_index
        trigL := high
        slL := _sl
        tpL := high + _riskTgt * rMultiple
        pbActL := false

if strategy.opentrades == 0 and coolOK and not waitS and pbActS and gateS and not na(pbBarS) and (bar_index - pbBarS) <= reclaimWin
    _slTight = coilOn ? (emaHi3 + clusterBuf * atr14) : (pbHiS + atrBuffMlt * atr14)
    _sl      = stopFloorATR > 0 ? math.max(_slTight, low + stopFloorATR * atr14) : _slTight
    _riskSL  = _sl - low
    _riskTgt = tgtFromTight ? (_slTight - low) : _riskSL
    if _riskSL > 0 and (minStopATR == 0 or _riskSL >= minStopATR * atr14) and (maxStopATR == 0 or _riskSL <= maxStopATR * atr14)
        waitS := true
        waitBarS := bar_index
        trigS := low
        slS := _sl
        tpS := low - _riskTgt * rMultiple
        pbActS := false

if waitL and (bar_index - waitBarL > reclaimWin or not regimeUp)
    waitL := false
if waitS and (bar_index - waitBarS > reclaimWin or not regimeDn)
    waitS := false

if waitL
    strategy.entry("Long", strategy.long, stop=trigL)
else
    strategy.cancel("Long")
if waitS
    strategy.entry("Short", strategy.short, stop=trigS)
else
    strategy.cancel("Short")

// ============ ENTRY / EXIT BAR DETECTION ============
longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
exitBar       = strategy.position_size == 0 and strategy.position_size[1] != 0
var bool lastWasLong = false
if longEntryBar
    lastWasLong := true
    waitL := false
if shortEntryBar
    lastWasLong := false
    waitS := false

// ============ EXITS ============
if strategy.position_size > 0
    waitL := false
    strategy.exit("Exit L", from_entry="Long", stop=slL, limit=tpL)
if strategy.position_size < 0
    waitS := false
    strategy.exit("Exit S", from_entry="Short", stop=slS, limit=tpS)

// ============ VISUALS ============
bgcolor(showBG and regimeUp and aboveBoth ? color.new(color.green, 92) : na, title="Bull")
bgcolor(showBG and regimeDn and belowBoth ? color.new(color.red,   92) : na, title="Bear")
bgcolor(showBG and useSession and not inSession ? color.new(color.gray, 92) : na, title="Off-session")

plot(ema9,   "EMA9",   color=color.aqua,   linewidth=1)
plot(ema22,  "EMA22",  color=color.orange, linewidth=1)
plot(ema200, "EMA200", color=color.new(color.purple, 0), linewidth=2)

plotshape(showMarks and waitL and not waitL[1], "Armed L", shape.circle, location.belowbar, color.new(color.yellow, 0), size=size.tiny)
plotshape(showMarks and waitS and not waitS[1], "Armed S", shape.circle, location.abovebar, color.new(color.yellow, 0), size=size.tiny)
plotshape(showMarks and longEntryBar,  "Long",  shape.triangleup,   location.belowbar, color.new(color.teal, 0), size=size.small, text="LONG",  textcolor=color.white)
plotshape(showMarks and shortEntryBar, "Short", shape.triangledown, location.abovebar, color.new(color.red, 0),  size=size.small, text="SHORT", textcolor=color.white)

var float entryPxV  = na
var float slAtEntry = na
if longEntryBar or shortEntryBar
    entryPxV  := strategy.position_avg_price
    slAtEntry := longEntryBar ? slL : slS

exWin = exitBar and strategy.closedtrades.profit(strategy.closedtrades - 1) >= 0
plotshape(showMarks and exitBar and exWin,     "Win",  shape.xcross, location.abovebar, color.new(color.green, 0), size=size.tiny)
plotshape(showMarks and exitBar and not exWin, "Loss", shape.xcross, location.abovebar, color.new(color.red,   0), size=size.tiny)
if showMarks and exitBar and not na(entryPxV) and not na(slAtEntry)
    exPx = strategy.closedtrades.exit_price(strategy.closedtrades - 1)
    risk = math.abs(entryPxV - slAtEntry)
    rr   = risk > 0 ? (lastWasLong ? (exPx - entryPxV) : (entryPxV - exPx)) / risk : 0.0
    label.new(bar_index, high, (rr >= 0 ? "+" : "") + str.tostring(rr, "0.0") + "R", yloc=yloc.abovebar, style=label.style_label_down, color=exWin ? color.new(color.green,0) : color.new(color.red,0), textcolor=color.white, size=size.tiny)

plot(strategy.position_size != 0 ? entryPxV : na, "Entry", color=color.new(color.gray, 0), style=plot.style_linebr, linewidth=1)
plot(strategy.position_size > 0 ? tpL : strategy.position_size < 0 ? tpS : na, "TP", color=color.new(color.green, 0), style=plot.style_linebr, linewidth=1)
plot(strategy.position_size > 0 ? slL : strategy.position_size < 0 ? slS : na, "SL", color=color.new(color.red, 0), style=plot.style_linebr, linewidth=1)

// ============ DEBUG ============
_atrN = na(atr14) ? 0.0 : atr14
if showDebug and regimeUp and pbActL and not waitL and strategy.position_size == 0
    _slT = coilOn ? (emaLo3 - clusterBuf * _atrN) : (pbLowL - atrBuffMlt * _atrN)
    _slF = stopFloorATR > 0 ? math.min(_slT, high - stopFloorATR * _atrN) : _slT
    _rk  = high - _slF
    r = not aboveBoth ? "no reclaim close" : not emaSepOK ? "EMAs tangled (sep)" : not inSession ? "off-session" : (bar_index - pbBarL > reclaimWin) ? "reclaim too late" : rsi3 >= rsiMaxLong ? "RSI(3) veto" : (maxStopATR > 0 and _rk > maxStopATR * _atrN) ? ("SL wide " + str.tostring(_rk/_atrN,"0.0") + "ATR") : "?"
    label.new(bar_index, low, "L: " + r + (coilOn ? " [coil]" : ""), yloc=yloc.belowbar, style=label.style_label_up, size=size.tiny, color=color.new(color.blue, 20), textcolor=color.white)
if showDebug and regimeDn and pbActS and not waitS and strategy.position_size == 0
    _slT = coilOn ? (emaHi3 + clusterBuf * _atrN) : (pbHiS + atrBuffMlt * _atrN)
    _slF = stopFloorATR > 0 ? math.max(_slT, low + stopFloorATR * _atrN) : _slT
    _rk  = _slF - low
    r = not belowBoth ? "no reclaim close" : not emaSepOK ? "EMAs tangled (sep)" : not inSession ? "off-session" : (bar_index - pbBarS > reclaimWin) ? "reclaim too late" : rsi3 <= rsiMinShort ? "RSI(3) veto" : (maxStopATR > 0 and _rk > maxStopATR * _atrN) ? ("SL wide " + str.tostring(_rk/_atrN,"0.0") + "ATR") : "?"
    label.new(bar_index, high, "S: " + r + (coilOn ? " [coil]" : ""), yloc=yloc.abovebar, style=label.style_label_down, size=size.tiny, color=color.new(color.maroon, 20), textcolor=color.white)
