//@version=5
// ============================================================================
//  BankNifty MTF Pullback  --  v1.0   (NSE:BANKNIFTY1! futures, intraday)
//  BRAND-NEW strategy, independent of the CrudeOil v1.0 script.
//  Concept (user spec): the 15-minute EMA CROSSOVER sets directional bias, then
//  trade the first clean lower-timeframe PULLBACK in that direction.
//    HTF (15m) EMA9 vs EMA21  -> bias        (request.security, lookahead OFF)
//    only act within N chart-bars of a FRESH 15m cross  (htfRecentBars)
//    LTF pullback to EMA9 while price rides above EMA9 & EMA21
//    enter stop @ reclaim-bar high ; SL below pullback low (buffered/floored/
//    capped) ; fixed-R target ; force-flat 15:15 ; 1 position, cooldown.
//  qty=1 = ONE BNF lot ; 0.02% commission ; 5-pt slippage.
//
//  ============================================================================
//  !!! CORRECTION (do not trust the numbers this header originally claimed) !!!
//  This strategy is NOT window-robust. The SAME config, SAME defaults:
//    window A (BNF falling, ~4mo)  : 45 tr | PF 1.31 | 51% win | +4.2%
//    window B (BNF rising, choppy) : 57 tr | PF 0.33 | 21% win | -16.2%
//  The "PF 1.31 / reproducible edge" claim was a single favourable 4-month
//  sample. On a different 4 months the identical strategy loses badly. Treat
//  every BankNifty backtest in this repo as UNVERIFIED pending a proper
//  walk-forward test across 8-12 separate quarters. Kept for reference only.
//  ============================================================================
//  (original in-sample numbers, window A only - NOT to be relied on):
//    3m : 45 tr | PF 1.31 | 51% win    5m : 85 tr | PF 1.12 | 38% win
//  The "fresh 15m cross" gate reduced trade count but did not create a stable
//  edge; on choppy directional windows the 15m EMA whipsaws and the strategy
//  shorts into up-moves / longs into down-moves and gets stopped.
// ============================================================================
strategy("BankNifty MTF Pullback v1.0", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     slippage=5,
     max_lines_count=500, max_labels_count=500)

// ============ INPUTS ============  (// N = sweep index)
htfTF        = input.timeframe("15", "HTF bias timeframe",  group="HTF bias")   // 0
htfFastLen   = input.int(9,  "HTF EMA fast",               group="HTF bias")   // 1
htfSlowLen   = input.int(21, "HTF EMA slow",               group="HTF bias")   // 2
htfUseSlope  = input.bool(false, "HTF: also require fast-EMA slope w/ bias", group="HTF bias") // 3
htfRecentBars= input.int(12, "HTF: only within N chart-bars of fresh cross (0=off)", group="HTF bias") // 4

emaFastLen   = input.int(9,   "LTF EMA fast",  group="LTF entry")              // 5
emaMedLen    = input.int(21,  "LTF EMA med",   group="LTF entry")              // 6
emaSlowLen   = input.int(200, "LTF EMA slow",  group="LTF entry")              // 7
use200       = input.bool(false, "LTF: require close vs EMA200", group="LTF entry") // 8
align3m      = input.bool(true, "LTF: require EMA9 vs EMA21 aligned w/ bias", group="LTF entry") // 9

useSession   = input.bool(true, "Session filter",          group="Session")    // 10
sessTime     = input.session("0925-1500", "Entry window (IST)", group="Session") // 11
skipOpenMin  = input.int(15, "Skip first N minutes",       group="Session")    // 12
flatTime     = input.session("1515-1530", "Force-flat window", group="Session") // 13
skipMidday   = input.bool(false, "Skip midday chop window", group="Session")   // 14
middayS      = input.session("1145-1315", "Midday window", group="Session")    // 15

atrMinPts    = input.float(35.0, "Min ATR14 points (0=off)", step=5, group="Volatility") // 16
atrMaxPts    = input.float(0.0,  "Max ATR14 points (0=off)", step=10, group="Volatility") // 17

atrBuffMlt   = input.float(0.15, "SL buffer beyond pullback low (x ATR14)", step=0.05, group="Risk") // 18
stopFloorATR = input.float(0.5,  "Min SL floor (x ATR14)",  step=0.05, group="Risk") // 19
maxStopATR   = input.float(2.0,  "Max stop (x ATR14, 0=off)", step=0.25, group="Risk") // 20
rMultiple    = input.float(2.5,  "R:R target",              step=0.1, group="Risk") // 21

pbLookback   = input.int(12, "Pullback within N bars of above-both", group="Engine") // 22
reclaimWin   = input.int(10, "Max bars: touch -> reclaim -> entry", group="Engine") // 23
coolBars     = input.int(2,  "Cooldown bars after exit",   group="Engine")    // 24

useWick      = input.bool(false, "Filter: pullback rejection wick", group="Filters") // 25
wickFrac     = input.float(0.5, "  wick >= frac of bar range", step=0.05, group="Filters") // 26
useVol       = input.bool(false, "Filter: entry bar volume >= x SMA20", group="Filters") // 27
volMlt       = input.float(1.2, "  volume mult",            step=0.1, group="Filters") // 28
rsiLen       = input.int(3,  "RSI length",                  group="Filters")   // 29
rsiMaxLong   = input.int(85, "No long if RSI(3) above",     group="Filters")   // 30
rsiMinShort  = input.int(15, "No short if RSI(3) below",    group="Filters")   // 31

allowCoil    = input.bool(false, "COIL: trade EMA-convergence break", group="Coil") // 32
clusterMlt   = input.float(1.0, "  coil: 3-EMA span <= x ATR14", step=0.1, group="Coil") // 33
clusterBuf   = input.float(0.25, "  coil SL buf beyond band (x ATR14)", step=0.05, group="Coil") // 34

beAfterR     = input.float(0.0, "EXP: SL to breakeven after +R (0=off)", step=0.1, group="Experimental") // 35
useTrail     = input.bool(false, "EXP: trail SL 0.5xATR after +1R", group="Experimental") // 36
skipExpiry   = input.bool(false, "EXP: skip Thursday (weekly expiry)", group="Experimental") // 37
showBG       = input.bool(true, "Trend background",         group="Visuals")   // 38
showMarks    = input.bool(true, "Entry/exit marks",         group="Visuals")   // 39

// ============ HTF BIAS (15m EMA crossover) ============
htfF = request.security(syminfo.tickerid, htfTF, ta.ema(close, htfFastLen), lookahead=barmerge.lookahead_off)
htfS = request.security(syminfo.tickerid, htfTF, ta.ema(close, htfSlowLen), lookahead=barmerge.lookahead_off)
htfFprev = request.security(syminfo.tickerid, htfTF, ta.ema(close, htfFastLen)[1], lookahead=barmerge.lookahead_off)

htfBull = htfF > htfS
htfBear = htfF < htfS
htfSlopeUp = htfF >= htfFprev
htfSlopeDn = htfF <= htfFprev
biasUp = htfBull and (not htfUseSlope or htfSlopeUp)
biasDn = htfBear and (not htfUseSlope or htfSlopeDn)

// bars since the 15m bias last flipped (counted in chart bars)
var int barsSinceFlip = 100000
biasSign = htfBull ? 1 : htfBear ? -1 : 0
barsSinceFlip := ta.change(biasSign) != 0 ? 0 : barsSinceFlip + 1
recentOK = htfRecentBars == 0 or barsSinceFlip <= htfRecentBars

// ============ LTF INDICATORS ============
ema9   = ta.ema(close, emaFastLen)
ema21  = ta.ema(close, emaMedLen)
ema200 = ta.ema(close, emaSlowLen)
rsi3   = ta.rsi(close, rsiLen)
atr14  = ta.atr(14)
volSma = ta.sma(volume, 20)

emaHi3 = math.max(ema9, math.max(ema21, ema200))
emaLo3 = math.min(ema9, math.min(ema21, ema200))
coilOn = allowCoil and (emaHi3 - emaLo3) <= clusterMlt * atr14

// ---- session / time ----
inSess  = not useSession or not na(time(timeframe.period, sessTime))
inMid   = skipMidday and not na(time(timeframe.period, middayS))
inFlat  = not na(time(timeframe.period, flatTime))
newDay  = ta.change(time("D")) != 0
var int dayOpenTime = na
if newDay
    dayOpenTime := time
minsSinceOpen = na(dayOpenTime) ? na : (time - dayOpenTime) / 60000.0
afterOpenSkip = na(minsSinceOpen) or minsSinceOpen >= skipOpenMin
inSession = inSess and afterOpenSkip and not inFlat and not inMid

atrOK  = (atrMinPts == 0 or atr14 >= atrMinPts) and (atrMaxPts == 0 or atr14 <= atrMaxPts)
isThu  = dayofweek == dayofweek.thursday
dowOK  = not (skipExpiry and isThu)

rng       = math.max(high - low, syminfo.mintick)
lowerWick = math.min(open, close) - low
upperWick = high - math.max(open, close)
bigLowerWick = lowerWick >= wickFrac * rng
bigUpperWick = upperWick >= wickFrac * rng

// ============ REGIME ============
regimeUp = (biasUp and recentOK) and (not align3m or ema9 > ema21 or coilOn) and (not use200 or close > ema200)
regimeDn = (biasDn and recentOK) and (not align3m or ema9 < ema21 or coilOn) and (not use200 or close < ema200)
aboveBoth = coilOn ? close > emaHi3 : (close > ema9 and close > ema21)
belowBoth = coilOn ? close < emaLo3 : (close < ema9 and close < ema21)

// ============ PULLBACK TRACKER ============
var bool  pbActL   = false
var float pbLowL   = na
var int   pbBarL   = na
var int   lastAbvL = na
var bool  pbWickL  = false
var bool  pbActS   = false
var float pbHiS    = na
var int   pbBarS   = na
var int   lastBlwS = na
var bool  pbWickS  = false

if not regimeUp
    pbActL := false
    pbLowL := na
    pbBarL := na
    pbWickL := false
else
    if aboveBoth
        lastAbvL := bar_index
    if low <= ema9 and not na(lastAbvL) and (bar_index - lastAbvL) <= pbLookback
        fresh = not pbActL or (not na(pbBarL) and lastAbvL >= pbBarL)
        pbLowL := fresh ? low : math.min(pbLowL, low)
        pbWickL := fresh ? bigLowerWick : (pbWickL or bigLowerWick)
        pbActL := true
        pbBarL := bar_index

if not regimeDn
    pbActS := false
    pbHiS  := na
    pbBarS := na
    pbWickS := false
else
    if belowBoth
        lastBlwS := bar_index
    if high >= ema9 and not na(lastBlwS) and (bar_index - lastBlwS) <= pbLookback
        fresh = not pbActS or (not na(pbBarS) and lastBlwS >= pbBarS)
        pbHiS := fresh ? high : math.max(pbHiS, high)
        pbWickS := fresh ? bigUpperWick : (pbWickS or bigUpperWick)
        pbActS := true
        pbBarS := bar_index

// ============ COOLDOWN ============
var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars

// ============ TRIGGER SETUP ============
var bool  waitL   = false
var int   waitBarL = na
var float trigL   = na
var float slL     = na
var float tpL     = na
var bool  waitS   = false
var int   waitBarS = na
var float trigS   = na
var float slS     = na
var float tpS     = na

wickOKL = not useWick or pbWickL
wickOKS = not useWick or pbWickS
volOKL  = not useVol or volume >= volMlt * volSma
volOKS  = volOKL

gateL = regimeUp and aboveBoth and inSession and atrOK and dowOK and wickOKL and volOKL and rsi3 < rsiMaxLong
gateS = regimeDn and belowBoth and inSession and atrOK and dowOK and wickOKS and volOKS and rsi3 > rsiMinShort

if strategy.opentrades == 0 and coolOK and not waitL and pbActL and gateL and not na(pbBarL) and (bar_index - pbBarL) <= reclaimWin
    _slTight = coilOn ? (emaLo3 - clusterBuf * atr14) : (pbLowL - atrBuffMlt * atr14)
    _sl      = stopFloorATR > 0 ? math.min(_slTight, high - stopFloorATR * atr14) : _slTight
    _riskSL  = high - _sl
    if _riskSL > 0 and (maxStopATR == 0 or _riskSL <= maxStopATR * atr14)
        waitL := true
        waitBarL := bar_index
        trigL := high
        slL := _sl
        tpL := high + _riskSL * rMultiple
        pbActL := false

if strategy.opentrades == 0 and coolOK and not waitS and pbActS and gateS and not na(pbBarS) and (bar_index - pbBarS) <= reclaimWin
    _slTight = coilOn ? (emaHi3 + clusterBuf * atr14) : (pbHiS + atrBuffMlt * atr14)
    _sl      = stopFloorATR > 0 ? math.max(_slTight, low + stopFloorATR * atr14) : _slTight
    _riskSL  = _sl - low
    if _riskSL > 0 and (maxStopATR == 0 or _riskSL <= maxStopATR * atr14)
        waitS := true
        waitBarS := bar_index
        trigS := low
        slS := _sl
        tpS := low - _riskSL * rMultiple
        pbActS := false

if waitL and (bar_index - waitBarL > reclaimWin or not regimeUp)
    waitL := false
if waitS and (bar_index - waitBarS > reclaimWin or not regimeDn)
    waitS := false

// ============ ENTRIES ============
if waitL and not inFlat
    strategy.entry("Long", strategy.long, stop=trigL)
else
    strategy.cancel("Long")
if waitS and not inFlat
    strategy.entry("Short", strategy.short, stop=trigS)
else
    strategy.cancel("Short")

longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
var float riskL0 = na
if longEntryBar
    waitL := false
    riskL0 := strategy.position_avg_price - slL
if shortEntryBar
    waitS := false
    riskL0 := slS - strategy.position_avg_price

// ============ EXITS ============
if strategy.position_size > 0
    waitL := false
    _sl = slL
    if beAfterR > 0 and not na(riskL0) and high >= strategy.position_avg_price + beAfterR * riskL0
        _sl := math.max(_sl, strategy.position_avg_price)
    if useTrail and not na(riskL0) and high >= strategy.position_avg_price + riskL0
        _sl := math.max(_sl, strategy.position_avg_price, high - 0.5 * atr14)
    slL := _sl
    strategy.exit("Exit L", from_entry="Long", stop=slL, limit=tpL)
    if inFlat
        strategy.close("Long", comment="EOD")
if strategy.position_size < 0
    waitS := false
    _sl = slS
    if beAfterR > 0 and not na(riskL0) and low <= strategy.position_avg_price - beAfterR * riskL0
        _sl := math.min(_sl, strategy.position_avg_price)
    if useTrail and not na(riskL0) and low <= strategy.position_avg_price - riskL0
        _sl := math.min(_sl, strategy.position_avg_price, low + 0.5 * atr14)
    slS := _sl
    strategy.exit("Exit S", from_entry="Short", stop=slS, limit=tpS)
    if inFlat
        strategy.close("Short", comment="EOD")

// ============ VISUALS ============
bgcolor(showBG and regimeUp ? color.new(color.green, 92) : na)
bgcolor(showBG and regimeDn ? color.new(color.red,   92) : na)
bgcolor(showBG and useSession and not inSession ? color.new(color.gray, 92) : na)
plot(ema9,   "LTF EMA9",  color=color.aqua,   linewidth=1)
plot(ema21,  "LTF EMA21", color=color.orange, linewidth=1)
plot(ema200, "LTF EMA200", color=color.new(color.purple, 0), linewidth=2)
plotshape(showMarks and longEntryBar,  "Long",  shape.triangleup,   location.belowbar, color.new(color.teal, 0), size=size.small, text="LONG",  textcolor=color.white)
plotshape(showMarks and shortEntryBar, "Short", shape.triangledown, location.abovebar, color.new(color.red, 0),  size=size.small, text="SHORT", textcolor=color.white)
plot(strategy.position_size > 0 ? tpL : strategy.position_size < 0 ? tpS : na, "TP", color=color.new(color.green, 0), style=plot.style_linebr, linewidth=1)
plot(strategy.position_size > 0 ? slL : strategy.position_size < 0 ? slS : na, "SL", color=color.new(color.red, 0), style=plot.style_linebr, linewidth=1)
