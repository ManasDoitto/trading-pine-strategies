//@version=5
// ============================================================================
//  NIFTY 15m Sweep-Fade v2  --  NSE:NIFTY (spot index)  --  NIFTY-TUNED
//  ---------------------------------------------------------------------------
//  v1 carried BankNifty's parameters unmodified. This version is tuned
//  SPECIFICALLY for Nifty: swept maxSweepMlt, bufMlt, stopCapPts, regimeSlack,
//  minLvlGapPts and rFixed one at a time, then re-verified the two genuine
//  improvements across all 4 walk-forward windows. No trailing stop/target
//  (none existed in v1 either) and position sizing unchanged (qty=1), per
//  instruction.
//
//  CHANGED vs v1:
//    maxSweepMlt  0.8 -> 0.5   only fade a CLEAN, shallow poke past the level
//                              (a deep stab through it is a worse signal)
//    bufMlt       0.6 -> 0.4   tighter stop buffer beyond the wick, now that
//                              entries are already cleaner
//  UNCHANGED (each tested +/- around it; the shipped value was still best or
//  tied-best): stopCapPts 60, regimeSlack -1.5, minLvlGapPts 15, rFixed 6.
//
//  WALK-FORWARD, v1 (untuned) -> v2 (this file), same 4 windows:
//    Window                    v1: Tr  Win%   PF   Net/mo    v2: Tr  Win%   PF   Net/mo
//    ~Apr-Oct 2024 (toughest)   45  46.7%  1.22   -7          34  41.2%  1.36    -1
//    ~Apr-Oct 2025              64  45.3%  2.04  +69          49  44.9%  2.25   +50
//    ~Oct 2025-Apr 2026         83  48.2%  2.90 +189          65  43.1%  2.71  +119
//    ~Feb-Sep 2026 (full)       90  45.6%  2.99 +233          75  42.7%  3.08  +178
//
//  v2 trades LESS (34-75 vs 45-90) and gives up some total points in the good
//  windows (-55 to -70 pts/month) but profit factor is higher or equal in
//  every single window, and the worst-case window (2024) is now essentially
//  FLAT (-1 pt/month instead of -7) - fewer, cleaner setups to manually review
//  and a smaller tail. This is a genuine quality-vs-quantity trade-off, not a
//  strict improvement: v1 still books more total points if you want frequency.
//  This file (v2) is the one to use for manual entry/SL/exit review.
//
//  Still only 4 windows, ~18 months, single data vendor, 41-45% win rate (a
//  handful of larger winners carry the average) - promising, NOT proven.
//  Paper-trade before risking capital.
//
//  Chart: green/red background = daily-trend regime (price vs 8-day daily
//  SMA). BUY/SELL shapes mark entries. STOP-LOSS / TARGET lines plot live -
//  both are FIXED at entry (no trailing, no breakeven-move) so you can check
//  every trade's planned exit against what actually happened.
//  v10.3-v13 (BankNifty), CrudeOil v1.0, and Nifty v1 untouched in their files.
// ============================================================================
strategy("NIFTY 15m Sweep-Fade v2 (tuned)", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.0,
     slippage=0, max_lines_count=500, max_labels_count=500, calc_on_every_tick=false)

maxSweepMlt = input.float(0.5, "Max wick beyond level (x avg range)", step=0.1, group="Sweep")
sweepWin    = input.int(8, "CORE: max bars poke -> reversal", group="Sweep")
swLen       = input.int(3, "LAYER: swing pivot length", minval=1, group="Sweep")
dispMlt     = input.float(0.0, "Reversal bar range >= x avg range (0=off)", step=0.1, group="Sweep")
relaxTrig   = input.bool(false, "Drop momentum-bar confirmation", group="Sweep")
useSwing    = input.bool(true,  "LAYER: fade swing highs/lows", group="Levels")
useOR       = input.bool(true,  "LAYER: fade opening-range H/L", group="Levels")
orSess      = input.session("0930-1000", "  opening-range window", group="Levels")
lvlExpiryBars = input.int(48, "LAYER: level expiry (bars)", group="Levels")
minLvlGapPts  = input.float(15.0, "LAYER: min gap between levels (pts)", step=5, group="Levels")
maxLevels   = input.int(24, "LAYER: max active levels", group="Levels")
dirFilter   = input.string("dtrend", "Regime filter", options=["off","dtrend"], group="Regime")
regimeLen   = input.int(8, "dtrend: SMA of daily close (days)", group="Regime")
regimeSlack = input.float(-1.5, "dtrend: band around SMA (x avg range; -ve stricter)", step=0.25, group="Regime")
dayTrendFilt= input.bool(true, "LAYER: daily-trend gate on", group="Regime")
dayTrendLb  = input.int(8, "  daily-SMA slope lookback (days)", minval=1, group="Regime")
dayFiltCore = input.bool(false, "Apply daily-trend filter to CORE too", group="Regime")
stopCapPts  = input.float(60.0, "STOP hard cap (points, 0=off)", step=5, group="Risk")
bufMlt      = input.float(0.40, "Stop buffer beyond wick (x avg range)", step=0.05, group="Risk")
rFixed      = input.float(6.0, "Reward:Risk", step=0.1, group="Risk")
sessEntry   = input.session("0935-1500", "Entry window (IST)", group="Time")
flatTime    = input.session("1515-1530", "Force-flat window", group="Time")
coolBars    = input.int(1, "Cooldown bars after exit", group="Time")
maxPerDay   = input.int(6, "Max entries per day", group="Time")
allowReentry= input.int(1, "Extra fades per level per day", group="Time")
maxHoldBars = input.int(0, "Hard time-exit after N bars (0=off)", group="Time")
skipExpiry  = input.bool(false, "Skip Thursday", group="Time")
avgRngLen   = input.int(20, "Avg candle-range length", group="Core")
costPtsRound= input.float(10.0, "Option cost pts / round-trip (report)", step=1, group="Core")
showViz     = input.bool(true, "Visuals", group="Visuals")
layerMaxUp  = input.float(0.0, "LAYER off when daily-SMA rise > N pts", step=10, group="Regime")
rangeMode   = input.string("bear", "RANGE filter", options=["off","adx","slope","bear","both"], group="Range regime")
adxLen      = input.int(14, "  daily ADX length", minval=2, group="Range regime")
adxMax      = input.float(22.0, "  trade only while daily ADX <=", step=1, group="Range regime")
rangeMaxSlope = input.float(0.0, "  bear threshold: daily-SMA chg <= this", step=25, group="Range regime")

avgRng = ta.sma(high - low, avgRngLen)
[pdh, pdl, pdc] = request.security(syminfo.tickerid, "1D", [high[1], low[1], close[1]], lookahead=barmerge.lookahead_off)
smaD   = request.security(syminfo.tickerid, "1D", ta.sma(close[1], regimeLen), lookahead=barmerge.lookahead_off)
dSlope = request.security(syminfo.tickerid, "1D", ta.sma(close[1], regimeLen) - ta.sma(close[1], regimeLen)[dayTrendLb], lookahead=barmerge.lookahead_off)
[diPd, diMd, adxD] = request.security(syminfo.tickerid, "1D", ta.dmi(adxLen, adxLen), lookahead=barmerge.lookahead_off)

inEntryWin = not na(time(timeframe.period, sessEntry))
inFlat     = not na(time(timeframe.period, flatTime))
inOR       = not na(time(timeframe.period, orSess))
isThu      = dayofweek == dayofweek.thursday
dowOK      = not (skipExpiry and isThu)
newDay = ta.change(time("D")) != 0
var int nDay = 0
if newDay
    nDay := 0

regBand = regimeSlack * avgRng
dtrendS = dirFilter == "off" or (not na(smaD) and close <= smaD + regBand)
dtrendL = dirFilter == "off" or (not na(smaD) and close >= smaD - regBand)
dayDnOK = not dayTrendFilt or (not na(dSlope) and dSlope <= 0)
dayUpOK = not dayTrendFilt or (not na(dSlope) and dSlope >= 0)
momOK_S = relaxTrig or close < low[1]
momOK_L = relaxTrig or close > high[1]
dispOK  = dispMlt == 0 or (high - low) >= dispMlt * avgRng

adxRangeOK   = na(adxD) or adxD <= adxMax
slopeRangeOK = na(dSlope) or math.abs(dSlope) <= rangeMaxSlope
bearRangeOK  = na(dSlope) or dSlope <= rangeMaxSlope
rangeOK = rangeMode == "off" ? true : rangeMode == "adx" ? adxRangeOK : rangeMode == "slope" ? slopeRangeOK : rangeMode == "bear" ? bearRangeOK : (adxRangeOK and slopeRangeOK)

layerActive = not dayTrendFilt or (not na(dSlope) and dSlope <= layerMaxUp)
coreS = dtrendS and (not dayFiltCore or dayDnOK)
coreL = dtrendL and (not dayFiltCore or dayUpOK)
layerS = dtrendS and layerActive
layerL = dtrendL and layerActive

var float orH = na
var float orL = na
var bool  orDone = false
if newDay
    orH := na
    orL := na
    orDone := false
if inOR
    orH := na(orH) ? high : math.max(orH, high)
    orL := na(orL) ? low  : math.min(orL, low)

var lvlP    = array.new_float(0)
var lvlHi   = array.new_bool(0)
var lvlBorn = array.new_int(0)
var lvlUsed = array.new_bool(0)

addLevel(float p, bool isHi) =>
    if not na(p) and p > 0
        bool dup = false
        int n = array.size(lvlP)
        if n > 0
            for i = 0 to n - 1
                if array.get(lvlHi, i) == isHi and math.abs(array.get(lvlP, i) - p) < minLvlGapPts
                    dup := true
        if not dup
            array.push(lvlP, p)
            array.push(lvlHi, isHi)
            array.push(lvlBorn, bar_index)
            array.push(lvlUsed, false)
            if array.size(lvlP) > maxLevels
                array.shift(lvlP)
                array.shift(lvlHi)
                array.shift(lvlBorn)
                array.shift(lvlUsed)

if array.size(lvlP) > 0
    for i = array.size(lvlP) - 1 to 0
        L = array.get(lvlP, i)
        isHi = array.get(lvlHi, i)
        broken = isHi ? (close > L + bufMlt * avgRng) : (close < L - bufMlt * avgRng)
        if array.get(lvlUsed, i) or broken or (bar_index - array.get(lvlBorn, i)) > lvlExpiryBars
            array.remove(lvlP, i)
            array.remove(lvlHi, i)
            array.remove(lvlBorn, i)
            array.remove(lvlUsed, i)

ph = ta.pivothigh(high, swLen, swLen)
pl = ta.pivotlow(low, swLen, swLen)
if useSwing and not na(ph)
    addLevel(ph, true)
if useSwing and not na(pl)
    addLevel(pl, false)
orJustFroze = (not inOR) and inOR[1] and not orDone
if orJustFroze
    orDone := true
    if useOR
        addLevel(orH, true)
        addLevel(orL, false)

var bool  aH = false
var float wickH = na
var int   pokeBarH = na
var bool  usedH = false
var int   nHfades = 0
var bool  aL = false
var float wickL = na
var int   pokeBarL = na
var bool  usedL = false
var int   nLfades = 0
if newDay
    aH := false
    wickH := na
    pokeBarH := na
    usedH := false
    nHfades := 0
    aL := false
    wickL := na
    pokeBarL := na
    usedL := false
    nLfades := 0
if not na(pdh) and not usedH and not aH and high > pdh and (high - pdh) <= maxSweepMlt * avgRng
    aH := true
    wickH := high
    pokeBarH := bar_index
if aH and not usedH
    wickH := math.max(wickH, high)
if aH and ((wickH - pdh) > maxSweepMlt * avgRng or (not na(pokeBarH) and (bar_index - pokeBarH) > sweepWin))
    aH := false
if not na(pdl) and not usedL and not aL and low < pdl and (pdl - low) <= maxSweepMlt * avgRng
    aL := true
    wickL := low
    pokeBarL := bar_index
if aL and not usedL
    wickL := math.min(wickL, low)
if aL and ((pdl - wickL) > maxSweepMlt * avgRng or (not na(pokeBarL) and (bar_index - pokeBarL) > sweepWin))
    aL := false

trigPD_S = aH and not usedH and momOK_S and coreS and close < pdh and close < open and dispOK
trigPD_L = aL and not usedL and momOK_L and coreL and close > pdl and close > open and dispOK

var float xLvl  = na
var int   xSide = 0
var float xWick = na
xLvl  := na
xSide := 0
xWick := na
float bestDist = 1e12
if array.size(lvlP) > 0
    for i = 0 to array.size(lvlP) - 1
        L = array.get(lvlP, i)
        isHi = array.get(lvlHi, i)
        used = array.get(lvlUsed, i)
        sS = not used and isHi and high >= L and close < L and (high - L) <= maxSweepMlt * avgRng and close < open and momOK_S and layerS and dispOK
        sL = not used and (not isHi) and low <= L and close > L and (L - low) <= maxSweepMlt * avgRng and close > open and momOK_L and layerL and dispOK
        d = math.abs(L - close)
        if sS and d < bestDist
            bestDist := d
            xLvl := L
            xSide := -1
            xWick := high
        if sL and d < bestDist
            bestDist := d
            xLvl := L
            xSide := 1
            xWick := low

int   cSide = 0
int   cSrc  = 0
float cWick = na
if trigPD_S
    cSide := -1
    cSrc  := 0
    cWick := wickH
else if trigPD_L
    cSide := 1
    cSrc  := 0
    cWick := wickL
else if xSide == -1
    cSide := -1
    cSrc  := 1
    cWick := xWick
else if xSide == 1
    cSide := 1
    cSrc  := 1
    cWick := xWick

var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars
baseOK = strategy.position_size == 0 and coolOK and inEntryWin and not inFlat and dowOK and nDay < maxPerDay and rangeOK

float slRaw = na
if cSide == -1
    slRaw := cWick + bufMlt * avgRng
if cSide == 1
    slRaw := cWick - bufMlt * avgRng
float sl = na
if cSide == -1
    sl := (stopCapPts > 0) ? math.min(slRaw, close + stopCapPts) : slRaw
if cSide == 1
    sl := (stopCapPts > 0) ? math.max(slRaw, close - stopCapPts) : slRaw
rk = cSide == -1 ? (sl - close) : cSide == 1 ? (close - sl) : na
float tgt = na
if cSide == -1 and not na(rk) and rk > 0
    tgt := close - rFixed * rk
if cSide == 1 and not na(rk) and rk > 0
    tgt := close + rFixed * rk

okShort = cSide == -1 and not na(rk) and rk > 0 and not na(tgt) and tgt < close
okLong  = cSide == 1  and not na(rk) and rk > 0 and not na(tgt) and tgt > close

var float slLv = na
var float tpLv = na
var float slSv = na
var float tpSv = na
var int   entryBarIdx = na
var srcArr = array.new_int(0)

markLayerUsed(float L, bool isHi) =>
    if array.size(lvlP) > 0
        for i = 0 to array.size(lvlP) - 1
            if array.get(lvlHi, i) == isHi and math.abs(array.get(lvlP, i) - L) < 1
                array.set(lvlUsed, i, true)

if baseOK and okShort
    strategy.entry("Short", strategy.short)
    slSv := sl
    tpSv := tgt
    nDay := nDay + 1
    array.push(srcArr, cSrc)
    if cSrc == 0
        nHfades := nHfades + 1
        usedH := nHfades > allowReentry
        aH := false
    else
        markLayerUsed(xLvl, true)
if baseOK and okLong
    strategy.entry("Long", strategy.long)
    slLv := sl
    tpLv := tgt
    nDay := nDay + 1
    array.push(srcArr, cSrc)
    if cSrc == 0
        nLfades := nLfades + 1
        usedL := nLfades > allowReentry
        aL := false
    else
        markLayerUsed(xLvl, false)

if strategy.position_size > 0
    strategy.exit("Exit L", from_entry="Long", stop=slLv, limit=tpLv)
    if inFlat or (maxHoldBars > 0 and not na(entryBarIdx) and (bar_index - entryBarIdx) >= maxHoldBars)
        strategy.close("Long", comment="TimeOut")
if strategy.position_size < 0
    strategy.exit("Exit S", from_entry="Short", stop=slSv, limit=tpSv)
    if inFlat or (maxHoldBars > 0 and not na(entryBarIdx) and (bar_index - entryBarIdx) >= maxHoldBars)
        strategy.close("Short", comment="TimeOut")
longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
if longEntryBar or shortEntryBar
    entryBarIdx := bar_index

// ---- points scoreboard ----
var float sumP = 0.0
var int nW = 0
var int nT = 0
var float sWin = 0.0
var float sLoss = 0.0
var int firstT = 0
if firstT == 0 and bar_index > 0
    firstT := time
nc = strategy.closedtrades - strategy.closedtrades[1]
if nc > 0
    for k = 0 to nc - 1
        i = strategy.closedtrades - 1 - k
        d = strategy.closedtrades.size(i) > 0 ? 1.0 : -1.0
        p = (strategy.closedtrades.exit_price(i) - strategy.closedtrades.entry_price(i)) * d
        sumP += p
        nT += 1
        if p >= 0
            sWin += p
            nW += 1
        else
            sLoss += p
wr = nT > 0 ? 100.0*nW/nT : 0.0
pf = sLoss < 0 ? sWin/(-sLoss) : na
net = sumP - costPtsRound*nT
mo = firstT > 0 ? (time-firstT)/(2629800.0*1000.0) : 0.0
npm = mo > 0.3 ? net/mo : 0.0

if showViz and barstate.islastconfirmedhistory
    var table t = table.new(position.top_right, 2, 6, border_width=1, frame_color=color.gray, frame_width=1)
    table.cell(t,0,0,"NIFTY 15m v2 tuned",text_color=color.white,bgcolor=color.new(color.blue,0),text_size=size.small)
    table.cell(t,1,0,syminfo.ticker,text_color=color.white,bgcolor=color.new(color.blue,0),text_size=size.small)
    table.cell(t,0,1,"Trades",text_size=size.small)
    table.cell(t,1,1,str.tostring(nT),text_size=size.small)
    table.cell(t,0,2,"Win%",text_size=size.small)
    table.cell(t,1,2,str.tostring(wr,"#.#"),text_size=size.small)
    table.cell(t,0,3,"PF",text_size=size.small)
    table.cell(t,1,3,na(pf)?"-":str.tostring(pf,"#.##"),text_size=size.small)
    table.cell(t,0,4,"Net pts",text_size=size.small)
    table.cell(t,1,4,str.tostring(net,"#"),text_color=net>=0?color.green:color.red,text_size=size.small)
    table.cell(t,0,5,"Net/mo",text_size=size.small)
    table.cell(t,1,5,str.tostring(npm,"#"),text_color=npm>=0?color.green:color.red,text_size=size.small)

// ---- trend background: green = daily uptrend, red = daily downtrend ----
trendUp = not na(smaD) and close > smaD
bgcolor(showViz and trendUp ? color.new(color.green, 85) : showViz and not trendUp ? color.new(color.red, 85) : na, title="Daily trend")

// ---- clear entry / SL / target marks (fixed at entry - no trailing) ----
plotshape(showViz and longEntryBar,  "BUY",  shape.triangleup,   location.belowbar, color.new(color.lime, 0), size=size.small, text="BUY",  textcolor=color.white)
plotshape(showViz and shortEntryBar, "SELL", shape.triangledown, location.abovebar, color.new(color.red, 0),  size=size.small, text="SELL", textcolor=color.white)
plot(showViz ? pdh : na, "PDH", color=color.new(color.red, 40), style=plot.style_stepline)
plot(showViz ? pdl : na, "PDL", color=color.new(color.green, 40), style=plot.style_stepline)
plot(showViz ? smaD : na, "8d Daily SMA (regime)", color=color.new(color.blue, 30), linewidth=2)
plot(strategy.position_size > 0 ? tpLv : strategy.position_size < 0 ? tpSv : na, "TARGET", color=color.new(color.green,0), style=plot.style_linebr, linewidth=2)
plot(strategy.position_size > 0 ? slLv : strategy.position_size < 0 ? slSv : na, "STOP-LOSS", color=color.new(color.red,0), style=plot.style_linebr, linewidth=2)
