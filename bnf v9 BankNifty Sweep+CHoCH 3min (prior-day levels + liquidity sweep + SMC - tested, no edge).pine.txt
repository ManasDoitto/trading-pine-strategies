//@version=5
// ============================================================================
//  BankNifty Sweep + CHoCH v9  --  NSE:BANKNIFTY1!  (3-minute)
//
//  !!! TESTED - NO EDGE.  Research record only. !!!
//  Built to the exact brief: "levels from last day's trading range + liquidity
//  sweep + SMC concept."
//    LEVELS  : PDH, PDL, PP=(H+L+C)/3, range mid (premium/discount pivot).
//    SWEEP   : wick beyond a level by <= maxSweepMlt x avg candle range, then
//              back inside (liquidity taken, break failed).
//    LOCATION: only fade a high sweep in discount / a low sweep in premium.
//    CHoCH   : the first 3m close that breaks the last micro swing in the fade
//              direction with a displacement bar (>= dispMlt x avg range).
//    ENTRY   : that CHoCH close.  STOP: beyond the sweep wick + buffer.
//    TARGET  : PP / opposite range level, else fixed R.
//
//  RESULT (3m tester window, B&H ~ +106k):
//    location filter OFF, disp 0-0.8    : 34-35 tr | 24-26% win | PF 0.37-0.40 | -12%
//    location filter ON                 : 0 trades (contradiction: you sweep the
//      HIGH = premium, but the filter then demands price be in discount to short)
//  Win rate 24-26% - identical to v6 (crude level fakeout) and to all other
//  families.  Adding the CHoCH confirmation + displacement + PP levels did not
//  change the outcome: a failed sweep of PDH/PDL on BankNifty does not reverse
//  reliably enough to fade for profit after cost.
//
//  Concept family #12.  This concludes the search.  Across the project:
//    #1 EMA pullback     #2 ORB break      #3 ORB fade      #4 momentum
//    #5 MTF confluence   #6 Session+SMC    #7 OB pullback    #8 VWAP revert
//    #9 morning fade     #10 level fakeout #11 FVG+OF        #12 pure SMC
//    #13 sweep+CHoCH  (this file)
//  ~65 configurations, walk-forward tested. EVERY approach: 22-37% win, PF
//  0.4-0.9. The 3 that tuned positive on one window (v1.0, v4.1, v7) failed
//  walk-forward. There is no regime-robust, commission-surviving mechanical
//  BankNifty 3m intraday edge in this rule-space.
// ============================================================================
strategy("BankNifty Sweep + CHoCH v9", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     slippage=5, max_lines_count=500, max_labels_count=500)

// ============ INPUTS ============  (// N)
useLoc     = input.bool(false, "Premium/discount location filter", group="Levels") // 0
locRef     = input.string("PP", "Location pivot", options=["PP","mid"], group="Levels") // 1
maxSweepMlt = input.float(1.0, "Max wick beyond level (x avg range)", step=0.1, group="Sweep") // 2
sweepWin   = input.int(15, "Max bars: sweep -> CHoCH", group="Sweep") // 3
chochLook  = input.int(5, "CHoCH: break micro swing of last N bars", group="CHoCH") // 4
dispMlt    = input.float(0.8, "CHoCH bar range >= x avg range", step=0.1, group="CHoCH") // 5
tgtMode    = input.string("level", "Target", options=["level","R"], group="Risk") // 6
tgtLevel   = input.string("PP", "Level target", options=["PP","opposite"], group="Risk") // 7
rFixed     = input.float(2.4, "Fixed R:R (or fallback)", step=0.1, group="Risk") // 8
minRR      = input.float(1.2, "Min R:R to accept a level target", step=0.1, group="Risk") // 9
bufMlt     = input.float(0.30, "Stop buffer beyond wick (x avg range)", step=0.05, group="Risk") // 10
maxStopMlt = input.float(3.0, "Reject if stop > x avg range (0=off)", step=0.25, group="Risk") // 11
sessEntry  = input.session("0930-1500", "Entry window (IST)", group="Time") // 12
flatTime   = input.session("1515-1530", "Force-flat window", group="Time") // 13
coolBars   = input.int(2, "Cooldown bars after exit", group="Time") // 14
maxPerDay  = input.int(4, "Max entries per day", group="Time") // 15
skipExpiry = input.bool(false, "Skip Thursday", group="Time") // 16
beAfterR   = input.float(0.0, "EXP: SL to BE after +R (0=off)", step=0.1, group="Exits") // 17
useTrail   = input.bool(false, "EXP: trail after +1R", group="Exits") // 18
trailMlt   = input.float(1.0, "  trail dist (x avg range)", step=0.1, group="Exits") // 19
avgRngLen  = input.int(20, "Avg candle-range length", group="Core") // 20
showViz    = input.bool(true, "Visuals", group="Visuals") // 21

// ============ CORE ============
avgRng = ta.sma(high - low, avgRngLen)
[pdh, pdl, pdc] = request.security(syminfo.tickerid, "1D", [high[1], low[1], close[1]], lookahead=barmerge.lookahead_off)
pp  = (pdh + pdl + pdc) / 3.0
mid = (pdh + pdl) / 2.0
locP = locRef == "PP" ? pp : mid

inEntryWin = not na(time(timeframe.period, sessEntry))
inFlat     = not na(time(timeframe.period, flatTime))
isThu      = dayofweek == dayofweek.thursday
dowOK      = not (skipExpiry and isThu)
newDay = ta.change(time("D")) != 0
var int nDay = 0
if newDay
    nDay := 0

microLo = ta.lowest(low, chochLook)
microHi = ta.highest(high, chochLook)

// ============ PDH SWEEP -> short ============
var bool  aH = false
var float wickH = na
var int   barH = na
var bool  usedH = false
if newDay
    aH := false
    wickH := na
    barH := na
    usedH := false
if not na(pdh) and not usedH and high > pdh and (high - pdh) <= maxSweepMlt * avgRng
    aH := true
    wickH := na(wickH) ? high : math.max(wickH, high)
    barH := na(barH) ? bar_index : barH
if aH and not na(barH) and (bar_index - barH) > sweepWin
    aH := false
    wickH := na
    barH := na
if aH and not na(wickH) and (high - pdh) > maxSweepMlt * avgRng
    aH := false
    wickH := na
    barH := na
locOKs = not useLoc or close < locP
chochS = aH and not usedH and locOKs and close < microLo[1] and close < open and (high - low) >= dispMlt * avgRng and close < pdh and not na(barH) and (bar_index - barH) <= sweepWin

// ============ PDL SWEEP -> long ============
var bool  aL = false
var float wickL = na
var int   barL = na
var bool  usedL = false
if newDay
    aL := false
    wickL := na
    barL := na
    usedL := false
if not na(pdl) and not usedL and low < pdl and (pdl - low) <= maxSweepMlt * avgRng
    aL := true
    wickL := na(wickL) ? low : math.min(wickL, low)
    barL := na(barL) ? bar_index : barL
if aL and not na(barL) and (bar_index - barL) > sweepWin
    aL := false
    wickL := na
    barL := na
if aL and not na(wickL) and (pdl - low) > maxSweepMlt * avgRng
    aL := false
    wickL := na
    barL := na
locOKl = not useLoc or close > locP
chochL = aL and not usedL and locOKl and close > microHi[1] and close > open and (high - low) >= dispMlt * avgRng and close > pdl and not na(barL) and (bar_index - barL) <= sweepWin

// ============ RISK ============
var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars
baseOK = strategy.position_size == 0 and coolOK and inEntryWin and not inFlat and dowOK and nDay < maxPerDay

float slS = na
float slL = na
if chochS and not na(wickH)
    slS := wickH + bufMlt * avgRng
if chochL and not na(wickL)
    slL := wickL - bufMlt * avgRng
rkS = not na(slS) ? slS - close : na
rkL = not na(slL) ? close - slL : na

float tgtS = na
float tgtL = na
if chochS and not na(rkS) and rkS > 0
    _lv = tgtLevel == "PP" ? pp : pdl
    tgtS := (tgtMode == "level" and not na(_lv) and _lv < close - minRR * rkS) ? _lv : close - rFixed * rkS
if chochL and not na(rkL) and rkL > 0
    _lv2 = tgtLevel == "PP" ? pp : pdh
    tgtL := (tgtMode == "level" and not na(_lv2) and _lv2 > close + minRR * rkL) ? _lv2 : close + rFixed * rkL

okS = chochS and not na(rkS) and rkS > 0 and (maxStopMlt == 0 or rkS <= maxStopMlt * avgRng) and not na(tgtS) and tgtS < close
okL = chochL and not na(rkL) and rkL > 0 and (maxStopMlt == 0 or rkL <= maxStopMlt * avgRng) and not na(tgtL) and tgtL > close

// ============ ENTRIES ============
var float slLv = na
var float tpLv = na
var float slSv = na
var float tpSv = na
var float riskInit = na

if baseOK and okS
    strategy.entry("Short", strategy.short)
    slSv := slS
    tpSv := tgtS
    nDay := nDay + 1
    usedH := true
if baseOK and okL
    strategy.entry("Long", strategy.long)
    slLv := slL
    tpLv := tgtL
    nDay := nDay + 1
    usedL := true

longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
if longEntryBar
    riskInit := strategy.position_avg_price - slLv
if shortEntryBar
    riskInit := slSv - strategy.position_avg_price

// ============ EXITS ============
if strategy.position_size > 0
    _s = slLv
    if beAfterR > 0 and not na(riskInit) and high >= strategy.position_avg_price + beAfterR * riskInit
        _s := math.max(_s, strategy.position_avg_price)
    if useTrail and not na(riskInit) and high >= strategy.position_avg_price + riskInit
        _s := math.max(_s, strategy.position_avg_price, high - trailMlt * avgRng)
    slLv := _s
    strategy.exit("Exit L", from_entry="Long", stop=slLv, limit=tpLv)
    if inFlat
        strategy.close("Long", comment="EOD")
if strategy.position_size < 0
    _s = slSv
    if beAfterR > 0 and not na(riskInit) and low <= strategy.position_avg_price - beAfterR * riskInit
        _s := math.min(_s, strategy.position_avg_price)
    if useTrail and not na(riskInit) and low <= strategy.position_avg_price - riskInit
        _s := math.min(_s, strategy.position_avg_price, low + trailMlt * avgRng)
    slSv := _s
    strategy.exit("Exit S", from_entry="Short", stop=slSv, limit=tpSv)
    if inFlat
        strategy.close("Short", comment="EOD")

// ============ VISUALS ============
plot(showViz ? pdh : na, "PDH", color=color.new(color.red, 30), style=plot.style_stepline)
plot(showViz ? pdl : na, "PDL", color=color.new(color.green, 30), style=plot.style_stepline)
plot(showViz ? pp  : na, "PP",  color=color.new(color.gray, 20), style=plot.style_stepline)
bgcolor(showViz and aH ? color.new(color.red, 92) : aL ? color.new(color.green, 92) : na)
plotshape(showViz and longEntryBar,  "L", shape.triangleup,   location.belowbar, color.new(color.teal,0), size=size.small, text="L")
plotshape(showViz and shortEntryBar, "S", shape.triangledown, location.abovebar, color.new(color.red,0),  size=size.small, text="S")
plot(strategy.position_size > 0 ? tpLv : strategy.position_size < 0 ? tpSv : na, "TP", color=color.new(color.green,0), style=plot.style_linebr, linewidth=1)
plot(strategy.position_size > 0 ? slLv : strategy.position_size < 0 ? slSv : na, "SL", color=color.new(color.red,0),   style=plot.style_linebr, linewidth=1)
