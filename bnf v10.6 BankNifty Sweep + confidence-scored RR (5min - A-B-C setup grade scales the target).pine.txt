//@version=5
// ============================================================================
//  BankNifty Sweep v10.6  --  NSE:BANKNIFTY1!  (5-minute)  --  CONFIDENCE-SCORED
//  ---------------------------------------------------------------------------
//  v10.5 engine (PDH/PDL sweep fade + daily-trend gate + 120-pt SL cap +
//  one re-entry/level) plus a discretionary-style CONFIDENCE SCORE (0-100)
//  graded A/B/C on every setup, shown live in the table with per-grade
//  n / win% / points-per-trade.
//
//  SCORE FACTORS (an experienced fade trader's checklist, max 100)
//    30  regime depth   : how far price is the "right" side of the 8d daily SMA
//    20  sweep clean     : shallow poke beyond the level (deep stab = worse)
//    20  reclaim strength: how far the bar closed back inside vs the poke size
//    15  not a trend day : intraday range so far still contained
//    10  time of day     : before 13:00 best, 13-14:00 ok, after 14:00 poor
//     5  prior-day close : weak close under PP for a short / strong for a long
//
//  >>> HONEST FINDING: on mechanical BankNifty PD-level fades the score does
//  NOT order outcomes. Across every window tested the mid band (B) matched or
//  beat the top band (A) on points/trade, and with a strict cut the highest-
//  scoring setups were the WORST (a very clean shallow poke = the level is
//  respected = small bounce, not a big reversal). Every "RR-by-grade" scheme
//  tried (A=6R/B=4R/C=2.5R, A=6/B=3, strict skip, etc.) came in at or below a
//  FLAT 6R on the walk-forward. So:
//    useConf = FALSE by default  -> identical to v10.5 (flat 6R, the proven
//              config: all 5 regime windows net-positive, worst trade -134).
//    The A/B/C table stays on as a DISCRETIONARY dashboard - a live read on
//    setup quality for a human, not an input to the mechanical RR.
//  Turn useConf on only to experiment; expect it to match or slightly lag
//  v10.5, not beat it.
//
//  v10.3/10.4/10.5 untouched in their own files.
// ============================================================================
strategy("BankNifty Sweep v10.6 CONF", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.0,
     slippage=0, max_lines_count=500, max_labels_count=500, calc_on_every_tick=false)

// ============ INPUTS ============  (// N)
maxSweepMlt = input.float(1.0, "Max wick beyond level (x avg range)", step=0.1, group="Sweep") // 0
reclaimBars = input.int(0, "grab B: reclaim within N bars (0=off)", group="Grab filters") // 1
rejWickFrac = input.float(0.0, "grab A: sweep-bar rejection wick >= frac (0=off)", step=0.05, group="Grab filters") // 2
stallBars   = input.int(0, "grab C: no new extreme for N bars (0=off)", group="Grab filters") // 3
poolLook    = input.int(40, "grab D: pool lookback (bars)", group="Grab filters") // 4
poolTol     = input.float(0.0, "grab D: pool swings within x avg range (0=off)", step=0.1, group="Grab filters") // 5
poolMin     = input.int(1, "grab D: min prior swings at the level", group="Grab filters") // 6
volMlt      = input.float(0.0, "grab E: sweep-bar volume >= x SMA20 (0=off)", step=0.1, group="Grab filters") // 7
engulfFrac  = input.float(0.0, "grab F: reversal close past x of sweep-bar (0=off)", step=0.05, group="Grab filters") // 8
swLen       = input.int(4, "Pool swing pivot length", group="Grab filters") // 9
sweepWin    = input.int(8, "Max bars: poke -> trigger", group="Sweep") // 10
dispMlt     = input.float(0.3, "Reversal bar range >= x avg range", step=0.1, group="Sweep") // 11
dirFilter   = input.string("dtrend", "Regime filter", options=["off","vs-open","pdc-vs-pp","dtrend","both"], group="Direction") // 12
tgtLevel    = input.string("opposite", "Level target (level mode)", options=["PP","opposite"], group="Risk") // 13
rFixed      = input.float(6.0, "Reward:Risk (R / fallback when useConf off)", step=0.1, group="Risk") // 14
minRR       = input.float(2.0, "Min R:R to accept a level target", step=0.1, group="Risk") // 15
bufMlt      = input.float(0.60, "Stop buffer beyond wick (x avg range)", step=0.05, group="Risk") // 16
maxStopMlt  = input.float(0.0, "Reject if stop > x avg range (0=off)", step=0.25, group="Risk") // 17
sessEntry   = input.session("0930-1500", "Entry window (IST)", group="Time") // 18
flatTime    = input.session("1515-1530", "Force-flat window", group="Time") // 19
coolBars    = input.int(1, "Cooldown bars after exit", group="Time") // 20
maxPerDay   = input.int(6, "Max entries per day", group="Time") // 21
skipExpiry  = input.bool(false, "Skip Thursday", group="Time") // 22
beAfterR    = input.float(0.0, "EXP: SL to BE after +R (0=off)", step=0.1, group="Exits") // 23
useTrail    = input.bool(false, "EXP: trail after +1R", group="Exits") // 24
trailMlt    = input.float(1.0, "  trail dist (x avg range)", step=0.1, group="Exits") // 25
avgRngLen   = input.int(20, "Avg candle-range length", group="Core") // 26
showViz     = input.bool(true, "Visuals", group="Visuals") // 27
regimeLen   = input.int(8, "dtrend: SMA of daily close (days)", group="Direction") // 28
regimeSlack = input.float(-1.5, "dtrend: band around SMA (x avg range; -ve = stricter)", step=0.25, group="Direction") // 29
minTgtPts   = input.float(0.0, "OPT: skip if target < N points away (0=off)", step=5, group="Option buyer") // 30
maxHoldBars = input.int(0, "OPT: hard time-exit after N bars (0=off)", group="Option buyer") // 31
costPtsRound= input.float(20.0, "OPT: option cost points / round-trip (report only)", step=1, group="Option buyer") // 32
stopMode    = input.string("capPts", "STOP mode", options=["wick","capPts","fixedPts"], group="SL / RR lab") // 33
stopCapPts  = input.float(120.0, "STOP hard cap (points, 0=off)", step=5, group="SL / RR lab") // 34
stopFixedPts= input.float(70.0, "STOP fixed points (fixedPts mode)", step=5, group="SL / RR lab") // 35
tgtMode2    = input.string("R", "TARGET mode (used when useConf off)", options=["level","R","fixedPts"], group="SL / RR lab") // 36
tgtFixedPts = input.float(140.0, "TARGET fixed points (fixedPts mode)", step=5, group="SL / RR lab") // 37
relaxTrig   = input.bool(false, "More trades: drop momentum-bar confirmation", group="SL / RR lab") // 38
allowReentry= input.int(1, "More trades: extra fades per level per day", group="SL / RR lab") // 39
// ---- confidence (new in v10.6) ----
useConf     = input.bool(false, "Confidence score scales the target (advisory only - see header)", group="Confidence") // 40
scoreMin    = input.int(30, "Min score to trade (below = grade D = skip)", group="Confidence") // 41
thrB        = input.int(50, "Score >= this  -> grade B", group="Confidence") // 42
thrA        = input.int(72, "Score >= this  -> grade A", group="Confidence") // 43
rrC         = input.float(2.5, "Grade C  reward:risk", step=0.1, group="Confidence") // 44
rrB         = input.float(4.0, "Grade B  reward:risk", step=0.1, group="Confidence") // 45
rrA         = input.float(6.0, "Grade A  reward:risk", step=0.1, group="Confidence") // 46

// ============ CORE ============
avgRng = ta.sma(high - low, avgRngLen)
volSma = ta.sma(volume, 20)
[pdh, pdl, pdc] = request.security(syminfo.tickerid, "1D", [high[1], low[1], close[1]], lookahead=barmerge.lookahead_off)
pp  = (pdh + pdl + pdc) / 3.0
smaD = request.security(syminfo.tickerid, "1D", ta.sma(close[1], regimeLen), lookahead=barmerge.lookahead_off)

inEntryWin = not na(time(timeframe.period, sessEntry))
inFlat     = not na(time(timeframe.period, flatTime))
isThu      = dayofweek == dayofweek.thursday
dowOK      = not (skipExpiry and isThu)
newDay = ta.change(time("D")) != 0
var int nDay = 0
var int nHfades = 0
var int nLfades = 0
var float dayOpenPx = na
var float dayHi = na
var float dayLo = na
if newDay
    nDay := 0
    nHfades := 0
    nLfades := 0
    dayOpenPx := open
    dayHi := high
    dayLo := low
else
    dayHi := math.max(dayHi, high)
    dayLo := math.min(dayLo, low)

pdcWeak   = not na(pdc) and not na(pp) and pdc <= pp
pdcStrong = not na(pdc) and not na(pp) and pdc >= pp
belowOpen = not na(dayOpenPx) and close < dayOpenPx
aboveOpen = not na(dayOpenPx) and close > dayOpenPx
regBand   = regimeSlack * avgRng
dtrendS   = not na(smaD) and close <= smaD + regBand
dtrendL   = not na(smaD) and close >= smaD - regBand
dirOKs = dirFilter == "off" ? true : dirFilter == "vs-open" ? belowOpen : dirFilter == "pdc-vs-pp" ? pdcWeak : dirFilter == "dtrend" ? dtrendS : (belowOpen and pdcWeak)
dirOKl = dirFilter == "off" ? true : dirFilter == "vs-open" ? aboveOpen : dirFilter == "pdc-vs-pp" ? pdcStrong : dirFilter == "dtrend" ? dtrendL : (aboveOpen and pdcStrong)

ph = ta.pivothigh(high, swLen, swLen)
pl = ta.pivotlow(low, swLen, swLen)
var phArr = array.new_float(0)
var plArr = array.new_float(0)
if not na(ph)
    array.push(phArr, ph)
    if array.size(phArr) > 60
        array.shift(phArr)
if not na(pl)
    array.push(plArr, pl)
    if array.size(plArr) > 60
        array.shift(plArr)

poolCntNear(arr, lvl) =>
    int c = 0
    n = array.size(arr)
    if poolTol > 0 and n > 0
        lo = math.max(0, n - 14)
        for i = lo to n - 1
            if math.abs(array.get(arr, i) - lvl) <= poolTol * avgRng
                c += 1
    c
poolHiOK = poolTol == 0 or poolCntNear(phArr, pdh) >= poolMin
poolLoOK = poolTol == 0 or poolCntNear(plArr, pdl) >= poolMin

wickUpFrac = (high - math.max(open, close)) / math.max(high - low, syminfo.mintick)
wickDnFrac = (math.min(open, close) - low) / math.max(high - low, syminfo.mintick)

// ============ CONFIDENCE SCORE ============
tmin = hour(time, "Asia/Kolkata") * 60 + minute(time, "Asia/Kolkata")
confScore(bool isShort, float wick, float lvl) =>
    float s = 0.0
    rd = isShort ? (smaD - close) / avgRng : (close - smaD) / avgRng
    s += rd >= 2.5 ? 30.0 : rd >= 1.5 ? 22.0 : rd >= 0.7 ? 12.0 : rd >= 0.0 ? 4.0 : 0.0
    poke = (isShort ? (wick - lvl) : (lvl - wick)) / avgRng
    s += poke <= 0.3 ? 20.0 : poke <= 0.6 ? 13.0 : poke <= 1.0 ? 6.0 : 0.0
    pk = math.max(isShort ? (wick - lvl) : (lvl - wick), avgRng * 0.1)
    rc = (isShort ? (lvl - close) : (close - lvl)) / pk
    s += rc >= 1.5 ? 20.0 : rc >= 0.8 ? 13.0 : rc >= 0.3 ? 6.0 : 0.0
    dr = (dayHi - dayLo) / avgRng
    s += dr <= 3.5 ? 15.0 : dr <= 5.0 ? 8.0 : dr <= 7.0 ? 3.0 : 0.0
    s += tmin <= 780 ? 10.0 : tmin <= 840 ? 5.0 : 0.0
    s += isShort ? (pdcWeak ? 5.0 : 0.0) : (pdcStrong ? 5.0 : 0.0)
    s
gradeOf(float sc) => sc >= thrA ? 3 : sc >= thrB ? 2 : sc >= scoreMin ? 1 : 0
rrOf(int g) => g == 3 ? rrA : g == 2 ? rrB : rrC

// ============ PDH SWEEP -> short ============
var bool  aH = false
var float wickH = na
var int   pokeBarH = na
var float sweepLowH = na
var bool  usedH = false
var bool  wickOKH = false
var bool  volOKH = false
if newDay
    aH := false
    wickH := na
    pokeBarH := na
    usedH := false
if not na(pdh) and not usedH and not aH and high > pdh and (high - pdh) <= maxSweepMlt * avgRng and poolHiOK
    aH := true
    wickH := high
    pokeBarH := bar_index
    sweepLowH := low
    wickOKH := rejWickFrac == 0 or wickUpFrac >= rejWickFrac
    volOKH := volMlt == 0 or na(volSma) or volume >= volMlt * volSma
if aH and not usedH
    wickH := math.max(wickH, high)
if aH and ((wickH - pdh) > maxSweepMlt * avgRng or (not na(pokeBarH) and (bar_index - pokeBarH) > sweepWin))
    aH := false
if aH and stallBars > 0 and not na(pokeBarH) and (bar_index - pokeBarH) > stallBars and high > wickH[1]
    aH := false
reclaimOKH = reclaimBars == 0 or (not na(pokeBarH) and (bar_index - pokeBarH) <= reclaimBars)
engulfOKH  = engulfFrac == 0 or (not na(sweepLowH) and close <= wickH - engulfFrac * (wickH - sweepLowH))
momOKH     = relaxTrig or close < low[1]
trigS = aH and not usedH and wickOKH and volOKH and reclaimOKH and engulfOKH and momOKH and dirOKs and close < pdh and close < open and (high - low) >= dispMlt * avgRng

// ============ PDL SWEEP -> long ============
var bool  aL = false
var float wickL = na
var int   pokeBarL = na
var float sweepHighL = na
var bool  usedL = false
var bool  wickOKL = false
var bool  volOKL = false
if newDay
    aL := false
    wickL := na
    pokeBarL := na
    usedL := false
if not na(pdl) and not usedL and not aL and low < pdl and (pdl - low) <= maxSweepMlt * avgRng and poolLoOK
    aL := true
    wickL := low
    pokeBarL := bar_index
    sweepHighL := high
    wickOKL := rejWickFrac == 0 or wickDnFrac >= rejWickFrac
    volOKL := volMlt == 0 or na(volSma) or volume >= volMlt * volSma
if aL and not usedL
    wickL := math.min(wickL, low)
if aL and ((pdl - wickL) > maxSweepMlt * avgRng or (not na(pokeBarL) and (bar_index - pokeBarL) > sweepWin))
    aL := false
if aL and stallBars > 0 and not na(pokeBarL) and (bar_index - pokeBarL) > stallBars and low < wickL[1]
    aL := false
reclaimOKL = reclaimBars == 0 or (not na(pokeBarL) and (bar_index - pokeBarL) <= reclaimBars)
engulfOKL  = engulfFrac == 0 or (not na(sweepHighL) and close >= wickL + engulfFrac * (sweepHighL - wickL))
momOKL     = relaxTrig or close > high[1]
trigL = aL and not usedL and wickOKL and volOKL and reclaimOKL and engulfOKL and momOKL and dirOKl and close > pdl and close > open and (high - low) >= dispMlt * avgRng

// ============ RISK ============
var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars
baseOK = strategy.position_size == 0 and coolOK and inEntryWin and not inFlat and dowOK and nDay < maxPerDay

wickStopS = not na(wickH) ? wickH + bufMlt * avgRng : na
wickStopL = not na(wickL) ? wickL - bufMlt * avgRng : na

float slS = na
float slL = na
if trigS
    slS := stopMode == "fixedPts" ? close + stopFixedPts : (stopMode == "capPts" and stopCapPts > 0) ? math.min(wickStopS, close + stopCapPts) : wickStopS
if trigL
    slL := stopMode == "fixedPts" ? close - stopFixedPts : (stopMode == "capPts" and stopCapPts > 0) ? math.max(wickStopL, close - stopCapPts) : wickStopL

rkS = not na(slS) ? slS - close : na
rkL = not na(slL) ? close - slL : na

capOKS = stopCapPts <= 0 or na(rkS) or rkS <= stopCapPts + syminfo.mintick
capOKL = stopCapPts <= 0 or na(rkL) or rkL <= stopCapPts + syminfo.mintick

// score / grade / rr for this bar's candidate
scoreS = (trigS and not na(rkS)) ? confScore(true, wickH, pdh) : na
scoreL = (trigL and not na(rkL)) ? confScore(false, wickL, pdl) : na
grS = not na(scoreS) ? gradeOf(scoreS) : 0
grL = not na(scoreL) ? gradeOf(scoreL) : 0
rrUsedS = useConf ? rrOf(grS) : rFixed
rrUsedL = useConf ? rrOf(grL) : rFixed

float tgtS = na
float tgtL = na
if trigS and not na(rkS) and rkS > 0
    _lv = tgtLevel == "PP" ? pp : pdl
    tgtS := useConf ? close - rrUsedS * rkS : tgtMode2 == "fixedPts" ? close - tgtFixedPts : tgtMode2 == "R" ? close - rFixed * rkS : ((not na(_lv) and _lv < close - minRR * rkS) ? _lv : close - rFixed * rkS)
if trigL and not na(rkL) and rkL > 0
    _lv2 = tgtLevel == "PP" ? pp : pdh
    tgtL := useConf ? close + rrUsedL * rkL : tgtMode2 == "fixedPts" ? close + tgtFixedPts : tgtMode2 == "R" ? close + rFixed * rkL : ((not na(_lv2) and _lv2 > close + minRR * rkL) ? _lv2 : close + rFixed * rkL)

gradeOKs = not useConf or grS > 0
gradeOKl = not useConf or grL > 0

okS = trigS and not na(rkS) and rkS > 0 and capOKS and gradeOKs and (maxStopMlt == 0 or rkS <= maxStopMlt * avgRng) and not na(tgtS) and tgtS < close and (minTgtPts == 0 or (close - tgtS) >= minTgtPts)
okL = trigL and not na(rkL) and rkL > 0 and capOKL and gradeOKl and (maxStopMlt == 0 or rkL <= maxStopMlt * avgRng) and not na(tgtL) and tgtL > close and (minTgtPts == 0 or (tgtL - close) >= minTgtPts)

// ============ ENTRIES ============
var float slLv = na
var float tpLv = na
var float slSv = na
var float tpSv = na
var float riskInit = na
var int   entryBarIdx = na
var float maxSLused = 0.0
var gradeArr = array.new_int(0)
var scoreArr = array.new_float(0)

if baseOK and okS
    strategy.entry("Short", strategy.short)
    slSv := slS
    tpSv := tgtS
    nDay := nDay + 1
    nHfades := nHfades + 1
    usedH := nHfades > allowReentry
    aH := false
    array.push(gradeArr, grS)
    array.push(scoreArr, scoreS)
if baseOK and okL
    strategy.entry("Long", strategy.long)
    slLv := slL
    tpLv := tgtL
    nDay := nDay + 1
    nLfades := nLfades + 1
    usedL := nLfades > allowReentry
    aL := false
    array.push(gradeArr, grL)
    array.push(scoreArr, scoreL)

longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
if longEntryBar
    riskInit := strategy.position_avg_price - slLv
    entryBarIdx := bar_index
    maxSLused := math.max(maxSLused, strategy.position_avg_price - slLv)
if shortEntryBar
    riskInit := slSv - strategy.position_avg_price
    entryBarIdx := bar_index
    maxSLused := math.max(maxSLused, slSv - strategy.position_avg_price)

if showViz and (longEntryBar or shortEntryBar)
    _g = array.size(gradeArr) > 0 ? array.get(gradeArr, array.size(gradeArr) - 1) : 0
    _sc = array.size(scoreArr) > 0 ? array.get(scoreArr, array.size(scoreArr) - 1) : na
    _txt = (_g == 3 ? "A" : _g == 2 ? "B" : _g == 1 ? "C" : "?") + " " + str.tostring(_sc, "#")
    label.new(bar_index, shortEntryBar ? high : low, _txt, style=shortEntryBar ? label.style_label_down : label.style_label_up, color=_g == 3 ? color.new(color.green, 0) : _g == 2 ? color.new(color.orange, 0) : color.new(color.gray, 0), textcolor=color.white, size=size.small)

// ============ EXITS ============
barsHeld = not na(entryBarIdx) ? bar_index - entryBarIdx : 0
if strategy.position_size > 0
    _s = slLv
    if beAfterR > 0 and not na(riskInit) and high >= strategy.position_avg_price + beAfterR * riskInit
        _s := math.max(_s, strategy.position_avg_price)
    if useTrail and not na(riskInit) and high >= strategy.position_avg_price + riskInit
        _s := math.max(_s, strategy.position_avg_price, high - trailMlt * avgRng)
    slLv := _s
    strategy.exit("Exit L", from_entry="Long", stop=slLv, limit=tpLv)
    if inFlat or (maxHoldBars > 0 and barsHeld >= maxHoldBars)
        strategy.close("Long", comment="TimeOut")
if strategy.position_size < 0
    _s = slSv
    if beAfterR > 0 and not na(riskInit) and low <= strategy.position_avg_price - beAfterR * riskInit
        _s := math.min(_s, strategy.position_avg_price)
    if useTrail and not na(riskInit) and low <= strategy.position_avg_price - riskInit
        _s := math.min(_s, strategy.position_avg_price, low + trailMlt * avgRng)
    slSv := _s
    strategy.exit("Exit S", from_entry="Short", stop=slSv, limit=tpSv)
    if inFlat or (maxHoldBars > 0 and barsHeld >= maxHoldBars)
        strategy.close("Short", comment="TimeOut")

// ============ SCOREBOARD ============
var float sumPts   = 0.0
var int   nWin     = 0
var int   nLoss    = 0
var int   nTr      = 0
var int   sumBars  = 0
var float sumWinP  = 0.0
var float sumLossP = 0.0
var float eqPts    = 0.0
var float eqPeak   = 0.0
var float maxDDpts = 0.0
var float bestT    = 0.0
var float worstT   = 0.0
var int   firstT   = 0
var int   nAt = 0
var int   nBt = 0
var int   nCt = 0
var int   wAt = 0
var int   wBt = 0
var int   wCt = 0
var float pAt = 0.0
var float pBt = 0.0
var float pCt = 0.0

if firstT == 0 and bar_index > 0
    firstT := time

newlyClosed = strategy.closedtrades - strategy.closedtrades[1]
if newlyClosed > 0
    for k = 0 to newlyClosed - 1
        idx = strategy.closedtrades - 1 - k
        dir = strategy.closedtrades.size(idx) > 0 ? 1.0 : -1.0
        pts = (strategy.closedtrades.exit_price(idx) - strategy.closedtrades.entry_price(idx)) * dir
        bars = strategy.closedtrades.exit_bar_index(idx) - strategy.closedtrades.entry_bar_index(idx)
        g = idx < array.size(gradeArr) ? array.get(gradeArr, idx) : 0
        sumPts += pts
        nTr += 1
        sumBars += bars
        if pts >= 0
            sumWinP += pts
            nWin += 1
        else
            sumLossP += pts
            nLoss += 1
        bestT  := math.max(bestT, pts)
        worstT := math.min(worstT, pts)
        eqPts  += pts
        eqPeak := math.max(eqPeak, eqPts)
        maxDDpts := math.max(maxDDpts, eqPeak - eqPts)
        if g == 3
            nAt += 1
            pAt += pts
            wAt += pts >= 0 ? 1 : 0
        else if g == 2
            nBt += 1
            pBt += pts
            wBt += pts >= 0 ? 1 : 0
        else
            nCt += 1
            pCt += pts
            wCt += pts >= 0 ? 1 : 0

avgPts   = nTr   > 0 ? sumPts   / nTr   : 0.0
winPct   = nTr   > 0 ? 100.0 * nWin / nTr : 0.0
