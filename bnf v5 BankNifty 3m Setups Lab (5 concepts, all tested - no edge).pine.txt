//@version=5
// ============================================================================
//  BankNifty 3m Setups Lab v5  --  NSE:BANKNIFTY1!  (single timeframe, no MTF)
//
//  !!! ALL 5 CONCEPTS TESTED - NONE PROFITABLE. Research record only. !!!
//
//  Five day-trading concepts, switchable via `setupMode`, tested on the
//  TradingView 3m tester window (an UP-trending ~2-month window, B&H ~ +100k):
//    vwap-revert   (fade N*ATR VWAP stretch)  : 200-300 tr | 22-37% win | PF 0.35-0.59
//    orb-fade      (fade false OR break)      : 210 tr     | 35% win    | PF 0.50
//    orb-break     (go with OR break)         : ~0 tr on this window
//    morning-fade  (fade the 9:15-11:00 move) :  29-32 tr  | 31-34% win | PF 0.46-0.55
//    vwap-pullback (trend pullback to VWAP/EMA): 205 tr    | 40% win    | PF 0.54
//
//  Findings:
//   - Mean-reversion FAILS on BankNifty: the further price stretches from VWAP,
//     the MORE likely it continues (win rate DROPS as the stretch threshold
//     rises - 1.5xATR -> 37% win, 2.5xATR -> 22% win).
//   - Fade-the-false-break FAILS: BankNifty OR breaks follow through often enough
//     that fading them bleeds.
//   - The fade modes also structurally over-trade (signal fires every bar the
//     condition holds, not just on the trigger event) - a design flaw, but even
//     the selective morning-fade (0.7 trades/day) is a net loser.
//   - Commission on 1 BNF lot (~Rs.700/round-trip) needs gross PF > ~1.3 just to
//     break even net; nothing here gets close.
//
//  This closes the search. Across the whole project (~30+ configs, 8 concept
//  families: EMA pullback, ORB, momentum, MTF-confluence, SMC, OB-pullback,
//  VWAP-revert, morning-fade) NO mechanical BankNifty 3m strategy shows a
//  regime-robust edge. Recommend: discretionary only, or a different market.
// ============================================================================
strategy("BankNifty 3m Setups Lab v5", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     slippage=5, max_lines_count=500, max_labels_count=500)

// ============ INPUTS ============  (// N)
setupMode = input.string("vwap-revert", "Setup", options=["vwap-revert","orb-fade","orb-break","morning-fade","vwap-pullback"], group="Setup") // 0
emaLen    = input.int(20, "EMA length", group="Setup")   // 1
atrLen    = input.int(14, "ATR length", group="Setup")   // 2
sessEntry = input.session("0930-1500", "Entry window (IST)", group="Time") // 3
flatTime  = input.session("1515-1530", "Force-flat window", group="Time") // 4
orbMin    = input.int(15, "Opening-range minutes", options=[5,10,15,20,30], group="ORB") // 5
vrExtATR  = input.float(1.5, "VWAP-revert: min dist from VWAP (x ATR)", step=0.1, group="vwap-revert") // 6
vrTgtFrac = input.float(1.0, "VWAP-revert: target = this frac of the way to VWAP", step=0.1, group="vwap-revert") // 7
mrfMoveATR = input.float(3.0, "Morning-fade: 09:15->11:00 move >= x ATR", step=0.5, group="morning-fade") // 8
mrfWin    = input.session("1245-1445", "Morning-fade: fade only in this window", group="morning-fade") // 9
rMultiple = input.float(1.5, "R:R target (go-with modes)", step=0.1, group="Risk") // 10
slBufATR  = input.float(0.25, "Stop buffer beyond structure (x ATR)", step=0.05, group="Risk") // 11
slFloorATR = input.float(0.5, "Min stop distance (x ATR)", step=0.05, group="Risk") // 12
maxStopATR = input.float(2.0, "Reject if stop > x ATR (0=off)", step=0.25, group="Risk") // 13
coolBars  = input.int(3, "Cooldown bars after exit", group="Risk") // 14
atrMinPts = input.float(0.0, "Min ATR pts to trade (0=off)", step=5, group="Risk") // 15
useTrail  = input.bool(false, "EXP: trail 0.5xATR after +1R", group="Exits") // 16
beAfterR  = input.float(0.0, "EXP: SL to BE after +R (0=off)", step=0.1, group="Exits") // 17
skipExpiry = input.bool(false, "Skip Thursday", group="Filters") // 18
maxPerDay = input.int(4, "Max entries per day", group="Filters") // 19
revBarNeed = input.bool(true, "Fade modes: require reversal-colour entry bar", group="Filters") // 20
showViz   = input.bool(true, "Visuals", group="Visuals") // 21

// ============ CORE ============
vwap = ta.vwap
ema  = ta.ema(close, emaLen)
atr  = ta.atr(atrLen)
hh20 = ta.highest(high, 20)
ll20 = ta.lowest(low, 20)

inEntryWin = not na(time(timeframe.period, sessEntry))
inFlat     = not na(time(timeframe.period, flatTime))
inMrfWin   = not na(time(timeframe.period, mrfWin))
atrOK      = atrMinPts == 0 or atr >= atrMinPts
isThu      = dayofweek == dayofweek.thursday
dowOK      = not (skipExpiry and isThu)

newDay = ta.change(time("D")) != 0
var int   dayOpenTime = na
var float dayOpenPx = na
var int   nDay = 0
if newDay
    dayOpenTime := time
    dayOpenPx := open
    nDay := 0
minsSinceOpen = na(dayOpenTime) ? na : (time - dayOpenTime) / 60000.0

var float orbHi = na
var float orbLo = na
var bool  hiBroken = false
var bool  loBroken = false
if newDay
    orbHi := high
    orbLo := low
    hiBroken := false
    loBroken := false
else if not na(minsSinceOpen) and minsSinceOpen < orbMin
    orbHi := math.max(orbHi, high)
    orbLo := math.min(orbLo, low)
orbDone = not na(minsSinceOpen) and minsSinceOpen >= orbMin
if orbDone and high > orbHi
    hiBroken := true
if orbDone and low < orbLo
    loBroken := true

var float px1100 = na
if newDay
    px1100 := na
if not na(minsSinceOpen) and minsSinceOpen >= 105 and na(px1100)
    px1100 := close
mornUp = not na(px1100) and (px1100 - dayOpenPx) >= mrfMoveATR * atr
mornDn = not na(px1100) and (dayOpenPx - px1100) >= mrfMoveATR * atr

rbOKL = not revBarNeed or close > open
rbOKS = not revBarNeed or close < open

// ============ PER-MODE SIGNALS ============
sigL = false
sigS = false
float stpL = na
float tgtL = na
float stpS = na
float tgtS = na

if setupMode == "vwap-revert"
    dAbove = close - vwap
    dBelow = vwap - close
    if dBelow >= vrExtATR * atr and rbOKL
        sigL := true
        stpL := math.min(low, low[1]) - slBufATR * atr
        tgtL := close + (vwap - close) * vrTgtFrac
    if dAbove >= vrExtATR * atr and rbOKS
        sigS := true
        stpS := math.max(high, high[1]) + slBufATR * atr
        tgtS := close - (close - vwap) * vrTgtFrac

if setupMode == "orb-fade"
    if orbDone and hiBroken and close < orbHi and rbOKS
        sigS := true
        stpS := math.max(high, high[1]) + slBufATR * atr
        tgtS := orbLo + (orbHi - orbLo) * 0.5
    if orbDone and loBroken and close > orbLo and rbOKL
        sigL := true
        stpL := math.min(low, low[1]) - slBufATR * atr
        tgtL := orbHi - (orbHi - orbLo) * 0.5

if setupMode == "orb-break"
    if orbDone and close > orbHi and close[1] <= orbHi
        sigL := true
        stpL := orbLo - slBufATR * atr
        tgtL := close + (close - (orbLo - slBufATR * atr)) * rMultiple
    if orbDone and close < orbLo and close[1] >= orbLo
        sigS := true
        stpS := orbHi + slBufATR * atr
        tgtS := close - ((orbHi + slBufATR * atr) - close) * rMultiple

if setupMode == "morning-fade"
    if inMrfWin and mornUp and close < ema and close[1] >= ema[1]
        sigS := true
        stpS := hh20 + slBufATR * atr
        tgtS := dayOpenPx + (px1100 - dayOpenPx) * 0.5
    if inMrfWin and mornDn and close > ema and close[1] <= ema[1]
        sigL := true
        stpL := ll20 - slBufATR * atr
        tgtL := dayOpenPx - (dayOpenPx - px1100) * 0.5

if setupMode == "vwap-pullback"
    aboveBoth = close > vwap and close > ema
    belowBoth = close < vwap and close < ema
    if aboveBoth and low <= math.max(vwap, ema) and close > open
        sigL := true
        stpL := math.min(low, vwap, ema) - slBufATR * atr
        tgtL := close + (close - (math.min(low, vwap, ema) - slBufATR * atr)) * rMultiple
    if belowBoth and high >= math.min(vwap, ema) and close < open
        sigS := true
        stpS := math.max(high, vwap, ema) + slBufATR * atr
        tgtS := close - ((math.max(high, vwap, ema) + slBufATR * atr) - close) * rMultiple

if sigL and not na(stpL)
    stpL := math.min(stpL, close - slFloorATR * atr)
if sigS and not na(stpS)
    stpS := math.max(stpS, close + slFloorATR * atr)

riskL = sigL and not na(stpL) ? close - stpL : na
riskS = sigS and not na(stpS) ? stpS - close : na
okL = sigL and not na(riskL) and riskL > 0 and (maxStopATR == 0 or riskL <= maxStopATR * atr) and not na(tgtL) and tgtL > close
okS = sigS and not na(riskS) and riskS > 0 and (maxStopATR == 0 or riskS <= maxStopATR * atr) and not na(tgtS) and tgtS < close

var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars
baseOK = strategy.position_size == 0 and coolOK and inEntryWin and not inFlat and atrOK and dowOK and nDay < maxPerDay

var float slL = na
var float tpL = na
var float slS = na
var float tpS = na
var float riskInit = na

if baseOK and okL
    strategy.entry("Long", strategy.long)
    slL := stpL
    tpL := tgtL
    nDay := nDay + 1
if baseOK and okS
    strategy.entry("Short", strategy.short)
    slS := stpS
    tpS := tgtS
    nDay := nDay + 1

longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
if longEntryBar
    riskInit := strategy.position_avg_price - slL
if shortEntryBar
    riskInit := slS - strategy.position_avg_price

if strategy.position_size > 0
    _s = slL
    if beAfterR > 0 and not na(riskInit) and high >= strategy.position_avg_price + beAfterR * riskInit
        _s := math.max(_s, strategy.position_avg_price)
    if useTrail and not na(riskInit) and high >= strategy.position_avg_price + riskInit
        _s := math.max(_s, strategy.position_avg_price, high - 0.5 * atr)
    slL := _s
    strategy.exit("Exit L", from_entry="Long", stop=slL, limit=tpL)
    if inFlat
        strategy.close("Long", comment="EOD")
if strategy.position_size < 0
    _s = slS
    if beAfterR > 0 and not na(riskInit) and low <= strategy.position_avg_price - beAfterR * riskInit
        _s := math.min(_s, strategy.position_avg_price)
    if useTrail and not na(riskInit) and low <= strategy.position_avg_price - riskInit
        _s := math.min(_s, strategy.position_avg_price, low + 0.5 * atr)
    slS := _s
    strategy.exit("Exit S", from_entry="Short", stop=slS, limit=tpS)
    if inFlat
        strategy.close("Short", comment="EOD")

plot(showViz ? vwap : na, "VWAP", color=color.new(color.blue, 0), linewidth=1)
plot(showViz ? ema : na, "EMA", color=color.orange, linewidth=1)
plot(showViz and orbDone ? orbHi : na, "ORB Hi", color=color.new(color.gray, 30), style=plot.style_linebr)
plot(showViz and orbDone ? orbLo : na, "ORB Lo", color=color.new(color.gray, 30), style=plot.style_linebr)
plotshape(showViz and longEntryBar,  "L", shape.triangleup,   location.belowbar, color.new(color.teal,0), size=size.tiny, text="L")
plotshape(showViz and shortEntryBar, "S", shape.triangledown, location.abovebar, color.new(color.red,0),  size=size.tiny, text="S")
plot(strategy.position_size > 0 ? tpL : strategy.position_size < 0 ? tpS : na, "TP", color=color.new(color.green,0), style=plot.style_linebr)
plot(strategy.position_size > 0 ? slL : strategy.position_size < 0 ? slS : na, "SL", color=color.new(color.red,0), style=plot.style_linebr)
