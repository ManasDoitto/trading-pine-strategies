//@version=5
strategy("EMA Pullback | Deep Sweep Breakout MTF (1:3)", overlay=true,
     initial_capital=100000,
     default_qty_type=strategy.fixed, default_qty_value=10,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     max_lines_count=500, max_labels_count=500)

// ============ INPUTS ============
biasTF      = input.timeframe("15", "Bias timeframe (regime gate)")
emaFastLen  = input.int(9,   "EMA Fast")
emaMedLen   = input.int(22,  "EMA Med")
emaSlowLen  = input.int(200, "EMA Slow")
rsiLen      = input.int(3,   "RSI Length")
rsiMaxLong  = input.int(80,  "No long if RSI(3) above")
rsiMinShort = input.int(20,  "No short if RSI(3) below")
reclaimWin  = input.int(6,   "Max bars from pullback candle to entry")
rMultiple   = input.float(3.0, "Target R-Multiple", step=0.1)
atrBuffMlt  = input.float(0.1, "SL buffer beyond candle (x ATR14)", step=0.05)
minStopATR  = input.float(0.0, "Min stop distance (x ATR14, 0=off)", step=0.05)
maxStopATR  = input.float(4.0, "Max stop distance (x ATR14, 0=off)", step=0.25)
useScaleOut = input.bool(false, "Scale-out mode (TP1 @ 2R + EMA9 trail)")
tp1R        = input.float(2.0, "TP1 R-multiple (scale-out)", step=0.1)
trailAfterR = input.float(1.0, "Arm EMA9 trail after +R (scale-out)", step=0.1)
showBG      = input.bool(true, "Trend background")
showMarks   = input.bool(true, "Entry/exit marks")
showDebug   = input.bool(false, "DEBUG: tag skipped setups")

// ============ EXECUTION-TF INDICATORS (chart TF = 5m intended) ============
ema9   = ta.ema(close, emaFastLen)
ema22  = ta.ema(close, emaMedLen)
ema200 = ta.ema(close, emaSlowLen)
rsi3   = ta.rsi(close, rsiLen)
atr14  = ta.atr(14)

// Pullback candle qualifies if its wick reaches EITHER EMA9 or EMA22
pbLevelL = math.max(ema9, ema22)   // long: low <= this  (touches the higher of the two)
pbLevelS = math.min(ema9, ema22)   // short: high >= this

// Raw Heikin Ashi candle colour (no smoothing)
haClose = (open + high + low + close) / 4
var float haOpen = na
haOpen := na(haOpen[1]) ? (open + close) / 2 : (haOpen[1] + haClose[1]) / 2
haBull = haClose > haOpen
haBear = haClose < haOpen

// ============ 15m BIAS (regime gate + background) ============
[e9_htf, e22_htf, e200_htf, c_htf] = request.security(syminfo.tickerid, biasTF,
     [ta.ema(close,emaFastLen), ta.ema(close,emaMedLen), ta.ema(close,emaSlowLen), close],
     lookahead=barmerge.lookahead_off)
biasLong  = c_htf > e200_htf and e9_htf > e22_htf
biasShort = c_htf < e200_htf and e9_htf < e22_htf

// ============ PULLBACK CANDLE TRACKER ============
// pullback candle = latest bar whose wick touches/pierces EMA9 or EMA22 while in bias.
// trigger = that candle's high (long) / low (short); SL = its opposite extreme.
var float pbTrigL = na
var float pbSLL   = na
var int   pbBarL  = na
var float pbTrigS = na
var float pbSLS   = na
var int   pbBarS  = na

if not biasLong
    pbTrigL := na
    pbSLL   := na
    pbBarL  := na
else if low <= pbLevelL
    pbTrigL := high
    pbSLL   := low
    pbBarL  := bar_index

if not biasShort
    pbTrigS := na
    pbSLS   := na
    pbBarS  := na
else if high >= pbLevelS
    pbTrigS := low
    pbSLS   := high
    pbBarS  := bar_index

barsSincePbL = na(pbBarL) ? 99999 : bar_index - pbBarL
barsSincePbS = na(pbBarS) ? 99999 : bar_index - pbBarS

// ============ ARM / TRIGGER ============
var bool  armedL = false
var float trigL  = na
var float slL    = na
var float tpL    = na
var float tp1L   = na
var bool  armedS = false
var float trigS  = na
var float slS    = na
var float tpS    = na
var float tp1S   = na

armOKL = not na(pbTrigL) and barsSincePbL >= 1 and barsSincePbL <= reclaimWin and haBull and rsi3 < rsiMaxLong
armOKS = not na(pbTrigS) and barsSincePbS >= 1 and barsSincePbS <= reclaimWin and haBear and rsi3 > rsiMinShort

if strategy.opentrades == 0 and armOKL
    _sl   = pbSLL - atrBuffMlt * atr14
    _risk = pbTrigL - _sl
    if _risk > 0 and (minStopATR == 0 or _risk >= minStopATR * atr14) and (maxStopATR == 0 or _risk <= maxStopATR * atr14)
        armedL := true
        armedS := false
        trigL  := pbTrigL
        slL    := _sl
        tpL    := pbTrigL + _risk * rMultiple
        tp1L   := pbTrigL + _risk * tp1R
    else
        armedL := false
else if strategy.opentrades == 0 and not armOKL
    armedL := false

if strategy.opentrades == 0 and armOKS
    _sl   = pbSLS + atrBuffMlt * atr14
    _risk = _sl - pbTrigS
    if _risk > 0 and (minStopATR == 0 or _risk >= minStopATR * atr14) and (maxStopATR == 0 or _risk <= maxStopATR * atr14)
        armedS := true
        armedL := false
        trigS  := pbTrigS
        slS    := _sl
        tpS    := pbTrigS - _risk * rMultiple
        tp1S   := pbTrigS - _risk * tp1R
    else
        armedS := false
else if strategy.opentrades == 0 and not armOKS
    armedS := false

if armedL
    strategy.entry("Long", strategy.long, stop=trigL)
else
    strategy.cancel("Long")

if armedS
    strategy.entry("Short", strategy.short, stop=trigS)
else
    strategy.cancel("Short")

// ============ ENTRY / EXIT BAR DETECTION ============
longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
exitBar       = strategy.position_size == 0 and strategy.position_size[1] != 0
var bool lastWasLong = false
if longEntryBar
    lastWasLong := true
    pbTrigL := na
    pbBarL  := na
if shortEntryBar
    lastWasLong := false
    pbTrigS := na
    pbBarS  := na

var bool trailArmL = false
var bool trailArmS = false
if longEntryBar
    trailArmL := false
if shortEntryBar
    trailArmS := false

// ============ EXITS ============
if strategy.position_size > 0
    armedL := false
    if useScaleOut
        r1 = strategy.position_avg_price + (strategy.position_avg_price - slL) * trailAfterR
        if high >= r1
            trailArmL := true
        strategy.exit("TP1 L", from_entry="Long", qty_percent=50, stop=slL, limit=tp1L)
        strategy.exit("Rnr L", from_entry="Long", stop=slL, limit=tpL)
        if trailArmL and close < ema9
            strategy.close("Long", comment="EMA9 trail")
    else
        strategy.exit("Exit L", from_entry="Long", stop=slL, limit=tpL)

if strategy.position_size < 0
    armedS := false
    if useScaleOut
        r1s = strategy.position_avg_price - (slS - strategy.position_avg_price) * trailAfterR
        if low <= r1s
            trailArmS := true
        strategy.exit("TP1 S", from_entry="Short", qty_percent=50, stop=slS, limit=tp1S)
        strategy.exit("Rnr S", from_entry="Short", stop=slS, limit=tpS)
        if trailArmS and close > ema9
            strategy.close("Short", comment="EMA9 trail")
    else
        strategy.exit("Exit S", from_entry="Short", stop=slS, limit=tpS)

// ============ TREND BACKGROUND (15m bias) ============
bgcolor(showBG and biasLong  ? color.new(color.green, 90) : na, title="Bull regime")
bgcolor(showBG and biasShort ? color.new(color.red,   90) : na, title="Bear regime")

// ============ EMA PLOTS ============
plot(ema9,   "EMA9",   color=color.aqua,   linewidth=1)
plot(ema22,  "EMA22",  color=color.orange, linewidth=1)
plot(ema200, "EMA200", color=color.purple, linewidth=2)

// ============ ENTRY / EXIT MARKS ============
plotshape(showMarks and armedL and not armedL[1], "Armed Long",  shape.circle, location.belowbar, color.new(color.yellow, 0), size=size.tiny)
plotshape(showMarks and armedS and not armedS[1], "Armed Short", shape.circle, location.abovebar, color.new(color.yellow, 0), size=size.tiny)

plotshape(showMarks and longEntryBar,  "Long Entry",  shape.triangleup,   location.belowbar, color.new(color.teal, 0), size=size.small, text="LONG",  textcolor=color.white)
plotshape(showMarks and shortEntryBar, "Short Entry", shape.triangledown, location.abovebar, color.new(color.red, 0),  size=size.small, text="SHORT", textcolor=color.white)

var float entryPxV  = na
var float slAtEntry = na
if longEntryBar or shortEntryBar
    entryPxV  := strategy.position_avg_price
    slAtEntry := longEntryBar ? slL : slS

exWin = exitBar and strategy.closedtrades.profit(strategy.closedtrades - 1) >= 0
plotshape(showMarks and exitBar and exWin,     "Exit Win",  shape.xcross, location.abovebar, color.new(color.green, 0), size=size.tiny)
plotshape(showMarks and exitBar and not exWin, "Exit Loss", shape.xcross, location.abovebar, color.new(color.red,   0), size=size.tiny)

if showMarks and exitBar and not na(entryPxV) and not na(slAtEntry)
    exPx = strategy.closedtrades.exit_price(strategy.closedtrades - 1)
    risk = math.abs(entryPxV - slAtEntry)
    rr   = risk > 0 ? (lastWasLong ? (exPx - entryPxV) : (entryPxV - exPx)) / risk : 0.0
    label.new(bar_index, high, (rr >= 0 ? "+" : "") + str.tostring(rr, "0.0") + "R",
         yloc=yloc.abovebar, style=label.style_label_down,
         color=exWin ? color.new(color.green, 0) : color.new(color.red, 0),
         textcolor=color.white, size=size.small)

plot(strategy.position_size != 0 ? entryPxV : na, "Entry Px", color=color.new(color.gray, 0),  style=plot.style_linebr, linewidth=1)
plot(strategy.position_size > 0 ? (useScaleOut ? tp1L : tpL) : strategy.position_size < 0 ? (useScaleOut ? tp1S : tpS) : na, "TP", color=color.new(color.green, 0), style=plot.style_linebr, linewidth=1)
plot(strategy.position_size > 0 ? slL : strategy.position_size < 0 ? slS : na, "SL", color=color.new(color.red,   0), style=plot.style_linebr, linewidth=1)

// ============ DEBUG ============
if showDebug and biasLong and not na(pbTrigL) and not armedL and strategy.position_size == 0
    r = barsSincePbL > reclaimWin ? "pb candle stale" : not haBull ? "HA not green" : rsi3 >= rsiMaxLong ? "RSI(3)>" + str.tostring(rsiMaxLong) : "risk out of band"
    label.new(bar_index, low, "L skip: " + r, yloc=yloc.belowbar, style=label.style_label_up, size=size.tiny, color=color.new(color.blue, 25), textcolor=color.white)
if showDebug and biasShort and not na(pbTrigS) and not armedS and strategy.position_size == 0
    r = barsSincePbS > reclaimWin ? "pb candle stale" : not haBear ? "HA not red" : rsi3 <= rsiMinShort ? "RSI(3)<" + str.tostring(rsiMinShort) : "risk out of band"
    label.new(bar_index, high, "S skip: " + r, yloc=yloc.abovebar, style=label.style_label_down, size=size.tiny, color=color.new(color.maroon, 25), textcolor=color.white)
