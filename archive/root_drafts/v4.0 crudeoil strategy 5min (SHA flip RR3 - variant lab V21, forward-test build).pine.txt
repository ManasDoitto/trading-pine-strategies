//@version=5
// ============================================================================
//  CrudeOil SHA-Flip RR3 v4.0  --  MCX:CRUDEOIL1! (5-minute), forward-test build
//  ---------------------------------------------------------------------------
//  Rule-for-rule copy of VARIANT LAB v1 variant V21 -- the only one of 192
//  variant/script/TF combinations that was net-profitable over all history:
//    Crude 5m, 23-Mar-24..11-Sep-26 (11 windows): 838 trades, 28.2% win,
//    PF 1.10, +2,240 pts net of 0.02%/side, daily Sharpe +0.53, 6/11 windows
//    positive, worst single-window drawdown 2,106 pts.
//  It was SELECTED from 192 tries on that same history, so it may be luck.
//  This build exists to forward-test it on data it has never seen.
//  Rules:
//    entry   : Smoothed Heikin Ashi (EMA10/EMA10) colour flips (green->long,
//              red->short) with EMA9 vs EMA22 agreeing; 09:15-23:30 IST
//    stop    : beyond the 10-bar swing + 0.1 ATR, at least 1.5 ATR away;
//              the trade is skipped if the stop would be wider than 3 ATR
//    target  : 3 x risk (R:R 1:3). No trailing, no SHA-flip exit, one
//              position at a time, may hold overnight (as tested)
//  The table splits BACKTEST trades (before `Forward test starts`) from
//  FORWARD-TEST trades (on/after it), net of commission, in points.
// ============================================================================
strategy("CRUDE 5m v4.0 SHA flip RR3 (forward test)", overlay=true,
     initial_capital=100000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     max_labels_count=500)

shaLen1  = input.int(10, "SHA pre-smooth EMA", group="Trend")
shaLen2  = input.int(10, "SHA post-smooth EMA", group="Trend")
useEmaAl = input.bool(true, "Require EMA9 vs EMA22 alignment", group="Trend")
use200   = input.bool(false, "Require close vs EMA200", group="Trend")
swLen    = input.int(10, "SL swing lookback (bars)", group="Risk")
swBuf    = input.float(0.1, "SL buffer beyond swing (x ATR)", step=0.05, group="Risk")
minSL    = input.float(1.5, "SL min distance (x ATR)", step=0.1, group="Risk")
maxSL    = input.float(3.0, "SL max distance (x ATR, skip wider)", step=0.1, group="Risk")
rr       = input.float(3.0, "Target R:R", step=0.1, group="Risk")
useSess  = input.bool(true, "Session filter", group="Time")
sess     = input.session("0915-2330", "Entry session", group="Time")
fwdStart = input.time(timestamp("2026-09-14T00:00:00+05:30"), "Forward test starts", group="Forward test")
showMarks = input.bool(true, "Entry / SL / TP marks", group="Visuals")

// ---- Smoothed Heikin Ashi (same maths as the public SHA indicator) ----
o1 = ta.ema(open, shaLen1)
c1 = ta.ema(close, shaLen1)
h1 = ta.ema(high, shaLen1)
l1 = ta.ema(low, shaLen1)
haC = (o1 + h1 + l1 + c1) / 4
var float haO = na
haO := na(haO[1]) ? (o1 + c1) / 2 : (haO[1] + haC[1]) / 2
shaUp = ta.ema(haC, shaLen2) > ta.ema(haO, shaLen2)
flipUp = shaUp and not shaUp[1]
flipDn = not shaUp and shaUp[1]

e9   = ta.ema(close, 9)
e22  = ta.ema(close, 22)
e200 = ta.ema(close, 200)
atr  = ta.atr(14)
swLo = ta.lowest(low, swLen)
swHi = ta.highest(high, swLen)

trendL = (not useEmaAl or e9 > e22) and (not use200 or close > e200)
trendS = (not useEmaAl or e9 < e22) and (not use200 or close < e200)
inSess = not useSess or not na(time(timeframe.period, sess))
flat = strategy.position_size == 0

riskL = math.max(close - (swLo - swBuf * atr), minSL * atr)
riskS = math.max((swHi + swBuf * atr) - close, minSL * atr)
okL = flat and inSess and flipUp and trendL and riskL <= maxSL * atr
okS = flat and inSess and flipDn and trendS and riskS <= maxSL * atr and not okL

var float slPx = na
var float tpPx = na
if okL
    slPx := close - riskL
    tpPx := close + rr * riskL
    strategy.entry("L", strategy.long)
    strategy.exit("xL", "L", stop=slPx, limit=tpPx)
    alert("CRUDE v4.0 LONG ~" + str.tostring(close, "#") + " SL " + str.tostring(slPx, "#") + " TP " + str.tostring(tpPx, "#"), alert.freq_once_per_bar_close)
    if showMarks
        label.new(bar_index, low - atr, "▲ LONG " + str.tostring(close, "#") + "\nSL " + str.tostring(slPx, "#") + "  (" + str.tostring(riskL, "#") + " pts)\nTP " + str.tostring(tpPx, "#") + "  (" + str.tostring(rr * riskL, "#") + " pts)\nR:R 1:" + str.tostring(rr, "#.#"), style=label.style_label_up, color=color.new(time >= fwdStart ? color.blue : color.teal, 10), textcolor=color.white, size=size.small, textalign=text.align_left)
if okS
    slPx := close + riskS
    tpPx := close - rr * riskS
    strategy.entry("S", strategy.short)
    strategy.exit("xS", "S", stop=slPx, limit=tpPx)
    alert("CRUDE v4.0 SHORT ~" + str.tostring(close, "#") + " SL " + str.tostring(slPx, "#") + " TP " + str.tostring(tpPx, "#"), alert.freq_once_per_bar_close)
    if showMarks
        label.new(bar_index, high + atr, "▼ SHORT " + str.tostring(close, "#") + "\nSL " + str.tostring(slPx, "#") + "  (" + str.tostring(riskS, "#") + " pts)\nTP " + str.tostring(tpPx, "#") + "  (" + str.tostring(rr * riskS, "#") + " pts)\nR:R 1:" + str.tostring(rr, "#.#"), style=label.style_label_down, color=color.new(time >= fwdStart ? color.purple : color.red, 10), textcolor=color.white, size=size.small, textalign=text.align_left)

plot(showMarks and strategy.position_size != 0 ? strategy.position_avg_price : na, "ENTRY", color=color.new(color.white, 0), style=plot.style_linebr, linewidth=1)
plot(showMarks and strategy.position_size != 0 ? slPx : na, "STOP-LOSS", color=color.new(color.red, 0), style=plot.style_linebr, linewidth=2)
plot(showMarks and strategy.position_size != 0 ? tpPx : na, "TARGET", color=color.new(color.green, 0), style=plot.style_linebr, linewidth=2)
bgcolor(time >= fwdStart and time[1] < fwdStart ? color.new(color.yellow, 60) : na, title="Forward-test start")

// ---- scoreboard: backtest vs forward test, net of commission, in points ----
var float[] bt = array.from(0.0, 0.0, 0.0, 0.0)   // trades, wins, sumWin, sumLoss
var float[] fw = array.from(0.0, 0.0, 0.0, 0.0)
nc = strategy.closedtrades - strategy.closedtrades[1]
if nc > 0
    for k = 0 to nc - 1
        i = strategy.closedtrades - 1 - k
        p = strategy.closedtrades.profit(i) / syminfo.pointvalue
        a = strategy.closedtrades.entry_time(i) >= fwdStart ? fw : bt
        array.set(a, 0, array.get(a, 0) + 1)
        if p > 0
            array.set(a, 1, array.get(a, 1) + 1)
            array.set(a, 2, array.get(a, 2) + p)
        else
            array.set(a, 3, array.get(a, 3) + p)

fmt(a) =>
    n = array.get(a, 0)
    w = array.get(a, 1)
    sw = array.get(a, 2)
    sl = array.get(a, 3)
    "trades " + str.tostring(n, "#") + " (" + str.tostring(w, "#") + "W/" + str.tostring(n - w, "#") + "L) win " + str.tostring(n > 0 ? 100 * w / n : 0, "#.#") + "% PF " + (sl < 0 ? str.tostring(sw / -sl, "#.##") : "-") + " | net " + str.tostring(sw + sl, "#") + " pts"

if barstate.islastconfirmedhistory or barstate.islast
    var table t = table.new(position.bottom_right, 1, 4, bgcolor=color.new(color.black, 10))
    var int firstT = time
    openTxt = strategy.position_size == 0 ? "flat" : (strategy.position_size > 0 ? "LONG" : "SHORT") + " @" + str.tostring(strategy.position_avg_price, "#") + " SL " + str.tostring(slPx, "#") + " TP " + str.tostring(tpPx, "#") + " open P/L " + str.tostring(strategy.openprofit / syminfo.pointvalue, "#") + " pts"
    table.cell(t, 0, 0, "v4.0 SHA flip RR3 | BACKTEST (bars loaded before " + str.format_time(fwdStart, "dd-MMM-yy", "Asia/Kolkata") + "): " + fmt(bt), text_color=color.white, text_size=size.tiny, text_halign=text.align_left)
    table.cell(t, 0, 1, "FORWARD TEST since " + str.format_time(fwdStart, "dd-MMM-yy HH:mm", "Asia/Kolkata") + ": " + fmt(fw) + " | now: " + openTxt, text_color=color.yellow, text_size=size.tiny, text_halign=text.align_left)
    table.cell(t, 0, 2, "Lab expectation (Mar-24..Sep-26): ~28 trades/month, 28% win, PF 1.10, ~+75 pts/month net, losing streaks and drawdowns up to ~2,100 pts are normal", text_color=color.silver, text_size=size.tiny, text_halign=text.align_left)
