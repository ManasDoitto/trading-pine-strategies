//@version=5
// ============================================================================
//  BankNifty Session + SMC v3  --  NSE:BANKNIFTY1!  (single timeframe: 3/5/15m)
//
//  !!! TESTED - DOES NOT WORK. Kept only as a research record. !!!
//  Built to spec: single timeframe (no MTF / no request.security), trade only
//  in the 2 best BankNifty move windows (09:45-11:00 + 13:00-14:30), gate with
//  a self-contained SMC pivot->BOS->bias engine, pullback OR breakout entry.
//
//  ---- 11 configs tested on NSE:BANKNIFTY1! (TradingView tester window;
//       buy&hold ranged +91k..+195k i.e. mostly a choppy uptrend, ~4-12mo) ----
//    3m pullback  | SMC with-bias        : 230 tr | 27% win | PF 0.43 | -53%
//    3m pullback  | SMC off              : 273 tr | 27% win | PF 0.41 | -63%
//    3m pullback  | with-bias + bos<=10  : 146 tr | 27% win | PF 0.39 | -38%
//    3m breakout  | SMC with-bias        :  82 tr | 23% win | PF 0.39 | -23%
//    5m pullback  | SMC with-bias        : 267 tr | 32% win | PF 0.67 | -42%
//    15m pullback | SMC with-bias        : 340 tr | 39% win | PF 0.71 | -62%
//    15m pullback | not-against+wide stop: 330 tr | 37% win | PF 0.73 | -65%
//    15m breakout | SMC with-bias        :  48 tr | 29% win | PF 0.48 | -23%
//    ...and 3 more param variants, all PF 0.39-0.73.
//
//  CONCLUSION: no setting is profitable. Win rate is pinned at 23-39% regardless
//  of TF, entry mode, SMC gate, session, R:R or stop width. The SMC structure
//  gate trims trade count but never lifts win rate. Restricting to the two
//  "best" windows does not help - the entries themselves have no edge on
//  BankNifty in these test windows.
//
//  This is consistent with the rest of the repo: NO mechanical BankNifty
//  EMA/SMC pullback-or-breakout strategy tested this project shows a
//  window-stable edge. Do not trade. Kept for reference / future walk-forward.
// ============================================================================
strategy("BankNifty Session + SMC v3", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     slippage=5, max_lines_count=500, max_labels_count=500)

// ============ INPUTS ============  (// N = sweep index)
entryMode = input.string("pullback", "Entry mode", options=["pullback","breakout"], group="Entry") // 0
emaFast   = input.int(9,   "EMA fast",  group="EMA")   // 1
emaMed    = input.int(21,  "EMA med",   group="EMA")   // 2
emaSlow   = input.int(200, "EMA slow",  group="EMA")   // 3
use200    = input.bool(false, "Require close vs EMA200", group="EMA")   // 4
useEmaTrend = input.bool(true, "Require EMA9 vs EMA21 dir", group="EMA") // 5

w1on = input.bool(true,  "Window 1 on",  group="Sessions")   // 6
w1   = input.session("0945-1100", "  window 1", group="Sessions") // 7
w2on = input.bool(true,  "Window 2 on",  group="Sessions")   // 8
w2   = input.session("1300-1430", "  window 2", group="Sessions") // 9
w3on = input.bool(false, "Window 3 on",  group="Sessions")   // 10
w3   = input.session("0930-1500", "  window 3", group="Sessions") // 11
flatTime = input.session("1515-1530", "Force-flat window", group="Sessions") // 12

swLen   = input.int(5, "SMC pivot length", group="SMC")   // 13
smcMode = input.string("with-bias", "SMC structure gate", options=["off","with-bias","not-against"], group="SMC") // 14
bosMaxBars = input.int(0, "SMC: only within N bars of a BOS (0=off)", group="SMC") // 15

atrMinPts   = input.float(25.0, "Min ATR14 pts (0=off)", step=5, group="Filters") // 16
rsiLen      = input.int(3,  "RSI length",           group="Filters") // 17
rsiMaxLong  = input.int(85, "No long if RSI(3) >",  group="Filters") // 18
rsiMinShort = input.int(15, "No short if RSI(3) <", group="Filters") // 19
pbLookback  = input.int(12, "Pullback within N bars of above-both", group="Pullback") // 20
reclaimWin  = input.int(10, "Max bars: touch->reclaim->entry", group="Pullback") // 21
brkLen      = input.int(10, "Breakout lookback", group="Breakout") // 22
brkBodyMlt  = input.float(0.5, "Breakout bar body >= x*ATR14", step=0.05, group="Breakout") // 23

rMultiple    = input.float(2.0, "R:R target", step=0.1, group="Risk") // 24
atrBuffMlt   = input.float(0.25, "SL buffer beyond struct (x ATR14)", step=0.05, group="Risk") // 25
stopFloorATR = input.float(0.6, "Min SL floor (x ATR14)", step=0.05, group="Risk") // 26
maxStopATR   = input.float(2.5, "Max stop (x ATR14, 0=off)", step=0.25, group="Risk") // 27
coolBars     = input.int(2, "Cooldown bars after exit", group="Engine") // 28
beAfterR     = input.float(0.0, "EXP: SL to BE after +R (0=off)", step=0.1, group="Exits") // 29
useTrail     = input.bool(false, "EXP: trail 0.5xATR after +1R", group="Exits") // 30
skipExpiry   = input.bool(false, "EXP: skip Thursday", group="Exits") // 31
showBG       = input.bool(true, "Trend background", group="Visuals") // 32
showMarks    = input.bool(true, "Entry/exit marks", group="Visuals") // 33

// ============ INDICATORS ============
ema9   = ta.ema(close, emaFast)
ema21  = ta.ema(close, emaMed)
ema200 = ta.ema(close, emaSlow)
rsi3   = ta.rsi(close, rsiLen)
atr14  = ta.atr(14)

// ---- sessions ----
inW1 = w1on and not na(time(timeframe.period, w1))
inW2 = w2on and not na(time(timeframe.period, w2))
inW3 = w3on and not na(time(timeframe.period, w3))
anyW = w1on or w2on or w3on
inWin  = (not anyW) or inW1 or inW2 or inW3
inFlat = not na(time(timeframe.period, flatTime))
atrOK  = atrMinPts == 0 or atr14 >= atrMinPts
isThu  = dayofweek == dayofweek.thursday
dowOK  = not (skipExpiry and isThu)

// ============ SMC STRUCTURE ============
ph = ta.pivothigh(high, swLen, swLen)
pl = ta.pivotlow(low, swLen, swLen)
var float lastPH = na
var float lastPL = na
var int   biasS  = 0
var int   bosBar = na
if not na(ph)
    lastPH := ph
if not na(pl)
    lastPL := pl
bosUp = not na(lastPH) and close > lastPH and close[1] <= lastPH
bosDn = not na(lastPL) and close < lastPL and close[1] >= lastPL
if bosUp
    biasS := 1
    bosBar := bar_index
if bosDn
    biasS := -1
    bosBar := bar_index
bosFresh = bosMaxBars == 0 or (not na(bosBar) and (bar_index - bosBar) <= bosMaxBars)
smcOKL = (smcMode == "off" ? true : smcMode == "with-bias" ? (biasS == 1) : (biasS != -1)) and bosFresh
smcOKS = (smcMode == "off" ? true : smcMode == "with-bias" ? (biasS == -1) : (biasS != 1)) and bosFresh

// ============ REGIME ============
trendUp = (not useEmaTrend or ema9 > ema21) and (not use200 or close > ema200)
trendDn = (not useEmaTrend or ema9 < ema21) and (not use200 or close < ema200)
aboveBoth = close > ema9 and close > ema21
belowBoth = close < ema9 and close < ema21

// ============ PULLBACK TRACKER ============
var bool  pbActL = false
var float pbLowL = na
var int   pbBarL = na
var int   lastAbvL = na
var bool  pbActS = false
var float pbHiS = na
var int   pbBarS = na
var int   lastBlwS = na

if not trendUp
    pbActL := false
    pbLowL := na
    pbBarL := na
else
    if aboveBoth
        lastAbvL := bar_index
    if low <= ema9 and not na(lastAbvL) and (bar_index - lastAbvL) <= pbLookback
        fr = not pbActL or (not na(pbBarL) and lastAbvL >= pbBarL)
        pbLowL := fr ? low : math.min(pbLowL, low)
        pbActL := true
        pbBarL := bar_index

if not trendDn
    pbActS := false
    pbHiS := na
    pbBarS := na
else
    if belowBoth
        lastBlwS := bar_index
    if high >= ema9 and not na(lastBlwS) and (bar_index - lastBlwS) <= pbLookback
        fr = not pbActS or (not na(pbBarS) and lastBlwS >= pbBarS)
        pbHiS := fr ? high : math.max(pbHiS, high)
        pbActS := true
        pbBarS := bar_index

// ============ BREAKOUT ============
brkHi = ta.highest(high, brkLen)[1]
brkLo = ta.lowest(low, brkLen)[1]
bodyOK = math.abs(close - open) >= brkBodyMlt * atr14
brkUpSig = close > brkHi and bodyOK and close > open
brkDnSig = close < brkLo and bodyOK and close < open

// ============ COOLDOWN ============
var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars

// ============ STATE ============
var bool  waitL = false
var int   waitBarL = na
var float trigL = na
var float slL = na
var float tpL = na
var bool  waitS = false
var int   waitBarS = na
var float trigS = na
var float slS = na
var float tpS = na
var float riskL0 = na

gateL = trendUp and inWin and not inFlat and atrOK and dowOK and smcOKL and rsi3 < rsiMaxLong
gateS = trendDn and inWin and not inFlat and atrOK and dowOK and smcOKS and rsi3 > rsiMinShort
isPB  = entryMode == "pullback"
isBRK = entryMode == "breakout"

// ---- pullback arm (stop at reclaim-bar high) ----
if isPB and strategy.opentrades == 0 and coolOK and not waitL and pbActL and gateL and aboveBoth and not na(pbBarL) and (bar_index - pbBarL) <= reclaimWin
    _slT = pbLowL - atrBuffMlt * atr14
    _sl  = stopFloorATR > 0 ? math.min(_slT, high - stopFloorATR * atr14) : _slT
    _rk  = high - _sl
    if _rk > 0 and (maxStopATR == 0 or _rk <= maxStopATR * atr14)
        waitL := true
        waitBarL := bar_index
        trigL := high
        slL := _sl
        tpL := high + _rk * rMultiple
        pbActL := false

if isPB and strategy.opentrades == 0 and coolOK and not waitS and pbActS and gateS and belowBoth and not na(pbBarS) and (bar_index - pbBarS) <= reclaimWin
    _slT = pbHiS + atrBuffMlt * atr14
    _sl  = stopFloorATR > 0 ? math.max(_slT, low + stopFloorATR * atr14) : _slT
    _rk  = _sl - low
    if _rk > 0 and (maxStopATR == 0 or _rk <= maxStopATR * atr14)
        waitS := true
        waitBarS := bar_index
        trigS := low
        slS := _sl
        tpS := low - _rk * rMultiple
        pbActS := false

if waitL and (bar_index - waitBarL > reclaimWin or not trendUp)
    waitL := false
if waitS and (bar_index - waitBarS > reclaimWin or not trendDn)
    waitS := false

if waitL and not inFlat
    strategy.entry("Long", strategy.long, stop=trigL)
else
    strategy.cancel("Long")
if waitS and not inFlat
    strategy.entry("Short", strategy.short, stop=trigS)
else
    strategy.cancel("Short")

// ---- breakout (market on signal close) ----
if isBRK and strategy.opentrades == 0 and coolOK and not inFlat
    if brkUpSig and gateL
        _slT = math.min(low, low[1]) - atrBuffMlt * atr14
        _sl  = stopFloorATR > 0 ? math.min(_slT, close - stopFloorATR * atr14) : _slT
        _rk  = close - _sl
        if _rk > 0 and (maxStopATR == 0 or _rk <= maxStopATR * atr14)
            strategy.entry("Long", strategy.long)
            slL := _sl
            tpL := close + _rk * rMultiple
    else if brkDnSig and gateS
        _slT = math.max(high, high[1]) + atrBuffMlt * atr14
        _sl  = stopFloorATR > 0 ? math.max(_slT, close + stopFloorATR * atr14) : _slT
        _rk  = _sl - close
        if _rk > 0 and (maxStopATR == 0 or _rk <= maxStopATR * atr14)
            strategy.entry("Short", strategy.short)
            slS := _sl
            tpS := close - _rk * rMultiple

longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
if longEntryBar
    waitL := false
    riskL0 := strategy.position_avg_price - slL
if shortEntryBar
    waitS := false
    riskL0 := slS - strategy.position_avg_price

// ============ EXITS ============
if strategy.position_size > 0
    _sl = slL
    if beAfterR > 0 and not na(riskL0) and high >= strategy.position_avg_price + beAfterR * riskL0
        _sl := math.max(_sl, strategy.position_avg_price)
    if useTrail and not na(riskL0) and high >= strategy.position_avg_price + riskL0
        _sl := math.max(_sl, strategy.position_avg_price, high - 0.5 * atr14)
    slL := _sl
    strategy.exit("Exit L", from_entry="Long", stop=slL, limit=tpL)
    if inFlat
        strategy.close("Long", comment="EOD")
if strategy.position_size < 0
    _sl = slS
    if beAfterR > 0 and not na(riskL0) and low <= strategy.position_avg_price - beAfterR * riskL0
        _sl := math.min(_sl, strategy.position_avg_price)
    if useTrail and not na(riskL0) and low <= strategy.position_avg_price - riskL0
        _sl := math.min(_sl, strategy.position_avg_price, low + 0.5 * atr14)
    slS := _sl
    strategy.exit("Exit S", from_entry="Short", stop=slS, limit=tpS)
    if inFlat
        strategy.close("Short", comment="EOD")

// ============ VISUALS ============
bgcolor(showBG and trendUp and inWin ? color.new(color.green, 92) : na)
bgcolor(showBG and trendDn and inWin ? color.new(color.red,   92) : na)
bgcolor(showBG and not inWin ? color.new(color.gray, 95) : na)
plot(ema9,  "EMA9",  color=color.aqua,   linewidth=1)
plot(ema21, "EMA21", color=color.orange, linewidth=1)
plot(ema200,"EMA200", color=color.new(color.purple,0), linewidth=2)
plot(lastPH, "last PH", color=color.new(color.red, 60), style=plot.style_circles)
plot(lastPL, "last PL", color=color.new(color.green, 60), style=plot.style_circles)
plotshape(showMarks and longEntryBar,  "L", shape.triangleup,   location.belowbar, color.new(color.teal,0), size=size.tiny, text="L")
plotshape(showMarks and shortEntryBar, "S", shape.triangledown, location.abovebar, color.new(color.red,0),  size=size.tiny, text="S")
plot(strategy.position_size > 0 ? tpL : strategy.position_size < 0 ? tpS : na, "TP", color=color.new(color.green,0), style=plot.style_linebr)
plot(strategy.position_size > 0 ? slL : strategy.position_size < 0 ? slS : na, "SL", color=color.new(color.red,0), style=plot.style_linebr)
