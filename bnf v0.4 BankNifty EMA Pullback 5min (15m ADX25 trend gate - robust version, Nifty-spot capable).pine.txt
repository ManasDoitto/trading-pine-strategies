//@version=5
// ============================================================================
//  BankNifty EMA Pullback v0.4  --  NSE:BANKNIFTY1!  5-minute
//  = v0.3 (4-condition reclaim gate: rejection wick + above-avg volume + VWAP
//  side + pullback held above EMA21) + ONE trend-strength gate:
//      htfAdxMin = 25 : previous COMPLETED 15m ADX(14) >= 25 (no lookahead)
//  (a daily-ADX gate is also available but OFF - it hurt this script).
//
//  Walk-forward, all available 5m data (5 windows tiled at UTC midnights,
//  Mar 2024 - Sep 2026), qty 1 lot (pointvalue 30), 0.02%/side + 5-pt slip:
//                       v0.3 (no gate)            v0.4 (this file)
//    trades             131  (4.4/month)          74   (2.5/month)
//    win rate           33.6%                     41.9%
//    profit factor      0.86                      1.29
//    net points         -1,229                    +1,277
//    daily Sharpe       -0.43                     +0.52
//    max drawdown       2,968 pts (Rs89k, 18%)    1,668 pts (Rs50k, 10%)
//    by window (old->new): v0.3 -88 -216 -1584 +282 +378
//                          v0.4 +816 +206 -533 -102 +889
//  Caveats: 74 trades in 30 months is a small sample; the gate value (25)
//  was carried over from the MTF script, not tuned here. Tested and
//  rejected: ATR floor 40 (4x trades, loses), daily-ADX gate (5 trades,
//  all losers in Sep25-Mar26). Forward-test before risking capital.
//
//  NIFTY 50 SPOT: the index has NO volume, so the core volume/VWAP condition
//  can never pass on NSE:NIFTY. Set `volSym` = NSE:NIFTY1! to take volume
//  from Nifty futures (VWAP is built from spot prices weighted by that
//  volume). Caveat found while testing: TradingView bar replay does NOT
//  return another symbol's volume, so this can only be verified on the
//  live (non-replay) chart. Live window Mar-Sep 2026 on spot: best was
//  atrMinPts 25 = 15 trades, PF 0.95, -21 pts. No robust Nifty version found.
//  qty=1 = one lot ; 0.02% commission ; 5-pt slippage ; flat by 15:15.
// ============================================================================
strategy("BankNifty EMA Pullback v0.4 (15m ADX gate)", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     slippage=5,
     max_lines_count=500, max_labels_count=500)

// ============ INPUTS ============  (index in comment for sweeps)
emaFastLen  = input.int(9,   "EMA Fast",  group="EMA")                 // 0
emaMedLen   = input.int(21,  "EMA Med",   group="EMA")                 // 1
emaSlowLen  = input.int(200, "EMA Slow",  group="EMA")                 // 2
use200      = input.bool(false, "Hard trend filter: EMA200", group="EMA") // 3

useSession  = input.bool(true, "Session filter",             group="Session") // 4
sessTime    = input.session("0930-1500", "Entry window (IST)", group="Session") // 5
skipOpenMin = input.int(15, "Skip first N minutes",          group="Session") // 6
flatTime    = input.session("1515-1530", "Force-flat window", group="Session") // 7
skipMidday  = input.bool(false, "Skip midday chop window",   group="Session") // 8
middayS     = input.session("1145-1315", "Midday window",    group="Session") // 9

atrMinPts   = input.float(50.0,  "Min ATR14 points (0=off)",  step=5, group="Volatility") // 10
atrMaxPts   = input.float(0.0,   "Max ATR14 points (0=off)",  step=10, group="Volatility") // 11

atrBuffMlt   = input.float(0.15, "SL buffer beyond pullback (x ATR14)", step=0.05, group="Risk") // 12
stopFloorATR = input.float(0.5,  "Min SL floor (x ATR14)",    step=0.05, group="Risk") // 13
maxStopATR  = input.float(2.0,   "Max stop (x ATR14, 0=off)", step=0.25, group="Risk") // 14
rMultiple   = input.float(2.5,   "R:R target",                step=0.1, group="Risk") // 15

minQual     = input.int(0,   "Quality: min extra score (0=off)", minval=0, maxval=6, group="Quality") // 16
wickFrac    = input.float(0.5, "  core: rejection wick >= frac of bar", step=0.05, group="Quality") // 17
qSepMlt     = input.float(0.4, "  q: EMA9-EMA21 sep >= x ATR14", step=0.05, group="Quality") // 18
volMlt      = input.float(1.0, "  core: entry volume >= x SMA20",  step=0.1, group="Quality") // 19
rsiResetL   = input.int(40,  "  q: long RSI(3) dipped <=",     group="Quality") // 20
rsiResetS   = input.int(60,  "  q: short RSI(3) spiked >=",    group="Quality") // 21
useVwapQ    = input.bool(true, "  core: require VWAP-side",     group="Quality") // 22

rsiLen      = input.int(3,   "RSI Length",                    group="RSI veto") // 23
rsiMaxLong  = input.int(80,  "No long if RSI(3) above",       group="RSI veto") // 24
rsiMinShort = input.int(20,  "No short if RSI(3) below",      group="RSI veto") // 25

useGap      = input.bool(false, "Gap-direction lock (first 60m)", group="Filters") // 26
gapMlt      = input.float(1.5, "  gap > x prior-day ATR",     step=0.1, group="Filters") // 27
skipExpiry  = input.bool(false, "Skip Thursday (weekly expiry)", group="Filters") // 28

pbLookback  = input.int(10,  "Pullback within N bars of above-both", group="Engine") // 29
reclaimWin  = input.int(8,   "Max bars: touch -> reclaim -> entry", group="Engine") // 30
coolBars    = input.int(3,   "Cooldown bars after exit",      group="Engine") // 31

allowCoil   = input.bool(true, "COIL: trade EMA-convergence break", group="Coil") // 32
clusterMlt  = input.float(1.0, "  coil: 3-EMA span <= x ATR14", step=0.1, group="Coil") // 33
clusterBuf  = input.float(0.25, "  coil SL buf beyond band (x ATR14)", step=0.05, group="Coil") // 34

useTrail    = input.bool(false, "EXP: trail SL after +1R", group="Experimental") // 35
beAfterR    = input.float(0.0,  "EXP: SL to BE after +R (0=off)", step=0.1, group="Experimental") // 36
tp1R        = input.float(0.0,  "EXP: partial TP at +R (0=off)", step=0.1, group="Experimental") // 37
tp1Pct      = input.int(50,     "EXP: partial TP % of position", group="Experimental") // 38
skipMon     = input.bool(false, "EXP: skip Monday",           group="Experimental") // 39
skipFri     = input.bool(false, "EXP: skip Friday",           group="Experimental") // 40
showBG      = input.bool(true,  "Trend background",           group="Visuals") // 41
showMarks   = input.bool(true,  "Entry/exit marks",           group="Visuals") // 42
coreWV      = input.bool(true, "Require wick+vol+vwap+hold (core edge)", group="Quality") // 43
dAdxMin     = input.float(0.0, "Trend gate: prev DAILY ADX14 >= (0=off)", step=1, group="Trend strength") // 44
htfAdxMin   = input.float(25.0, "Trend gate: prev 15m ADX14 >= (0=off)", step=1, group="Trend strength") // 45
volSym      = input.string("", "Volume source symbol (blank = chart; NIFTY spot -> NSE:NIFTY1!)", group="Data") // 46

// ============ TREND-STRENGTH GATES (completed bars only, no lookahead) ============
f_adxPrev() =>
    [_p, _m, _a] = ta.dmi(14, 14)
    _a[1]
dAdx = request.security(syminfo.tickerid, "D", f_adxPrev(), lookahead=barmerge.lookahead_on)
hAdx = request.security(syminfo.tickerid, "15", f_adxPrev(), lookahead=barmerge.lookahead_on)
adxOK = (dAdxMin == 0 or (not na(dAdx) and dAdx >= dAdxMin)) and (htfAdxMin == 0 or (not na(hAdx) and hAdx >= htfAdxMin))

// ============ VOLUME SOURCE + VWAP (chart prices weighted by source volume) ============
vSrc = volSym == "" ? syminfo.tickerid : volSym
volX = nz(request.security(vSrc, timeframe.period, volume, lookahead=barmerge.lookahead_off))
vwNewDay = ta.change(time("D")) != 0
var float cPV = 0.0
var float cV  = 0.0
if vwNewDay
    cPV := 0.0
    cV  := 0.0
cPV += hlc3 * volX
cV  += volX
vwap = cV > 0 ? cPV / cV : na

// ============ INDICATORS ============
ema9   = ta.ema(close, emaFastLen)
ema21  = ta.ema(close, emaMedLen)
ema200 = ta.ema(close, emaSlowLen)
rsi3   = ta.rsi(close, rsiLen)
atr14  = ta.atr(14)
volSma = ta.sma(volX, 20)
rsiLoN = ta.lowest(rsi3, pbLookback)
rsiHiN = ta.highest(rsi3, pbLookback)

emaHi3 = math.max(ema9, math.max(ema21, ema200))
emaLo3 = math.min(ema9, math.min(ema21, ema200))
coilOn = allowCoil and (emaHi3 - emaLo3) <= clusterMlt * atr14

inSess   = not useSession or not na(time(timeframe.period, sessTime))
inMid    = skipMidday and not na(time(timeframe.period, middayS))
inFlat   = not na(time(timeframe.period, flatTime))
newDay   = ta.change(time("D")) != 0
var int  dayOpenTime = na
var float dayOpenPx  = na
if newDay
    dayOpenTime := time
    dayOpenPx   := open
minsSinceOpen = na(dayOpenTime) ? na : (time - dayOpenTime) / 60000.0
afterOpenSkip = na(minsSinceOpen) or minsSinceOpen >= skipOpenMin
inSession = inSess and afterOpenSkip and not inFlat and not inMid

var float dHi = na
var float dLo = na
var float prevDayClose = na
var float prevDayATR   = na
if newDay
    if not na(dHi) and not na(dLo)
        _tr = dHi - dLo
        prevDayATR := na(prevDayATR) ? _tr : (prevDayATR * 13 + _tr) / 14.0
    prevDayClose := close[1]
    dHi := high
    dLo := low
else
    dHi := na(dHi) ? high : math.max(dHi, high)
    dLo := na(dLo) ? low  : math.min(dLo, low)

gapUp = not na(prevDayClose) and not na(prevDayATR) and (dayOpenPx - prevDayClose) > gapMlt * prevDayATR
gapDn = not na(prevDayClose) and not na(prevDayATR) and (prevDayClose - dayOpenPx) > gapMlt * prevDayATR
inGapWin = not na(minsSinceOpen) and minsSinceOpen < 60
gapAllowL = not useGap or not inGapWin or not gapDn
gapAllowS = not useGap or not inGapWin or not gapUp

atrOK     = (atrMinPts == 0 or atr14 >= atrMinPts) and (atrMaxPts == 0 or atr14 <= atrMaxPts)
isThu     = dayofweek == dayofweek.thursday
expiryOK  = not skipExpiry or not isThu
dowOK     = not (skipMon and dayofweek == dayofweek.monday) and not (skipFri and dayofweek == dayofweek.friday) and expiryOK

// ============ REGIME ============
regimeUp = (ema9 > ema21 or coilOn) and (not use200 or close > ema200)
regimeDn = (ema9 < ema21 or coilOn) and (not use200 or close < ema200)
aboveBoth = coilOn ? close > emaHi3 : (close > ema9 and close > ema21)
belowBoth = coilOn ? close < emaLo3 : (close < ema9 and close < ema21)

rng      = math.max(high - low, syminfo.mintick)
lowerWick = math.min(open, close) - low
upperWick = high - math.max(open, close)
bigLowerWick = lowerWick >= wickFrac * rng
bigUpperWick = upperWick >= wickFrac * rng

// ============ PULLBACK TRACKER ============
var bool  pbActL   = false
var float pbLowL   = na
var int   pbBarL   = na
var int   lastAbvL = na
var bool  pbWickL  = false
var bool  pbHoldL  = false
var bool  pbActS   = false
var float pbHiS    = na
var int   pbBarS   = na
var int   lastBlwS = na
var bool  pbWickS  = false
var bool  pbHoldS  = false

if not regimeUp
    pbActL := false
    pbLowL := na
    pbBarL := na
    pbWickL := false
    pbHoldL := false
else
    if aboveBoth
        lastAbvL := bar_index
    if low <= ema9 and not na(lastAbvL) and (bar_index - lastAbvL) <= pbLookback
        fresh = not pbActL or (not na(pbBarL) and lastAbvL >= pbBarL)
        pbLowL := fresh ? low : math.min(pbLowL, low)
        pbWickL := fresh ? bigLowerWick : (pbWickL or bigLowerWick)
        pbHoldL := close > ema21
        pbActL := true
        pbBarL := bar_index

if not regimeDn
    pbActS := false
    pbHiS  := na
    pbBarS := na
    pbWickS := false
    pbHoldS := false
else
    if belowBoth
        lastBlwS := bar_index
    if high >= ema9 and not na(lastBlwS) and (bar_index - lastBlwS) <= pbLookback
        fresh = not pbActS or (not na(pbBarS) and lastBlwS >= pbBarS)
        pbHiS := fresh ? high : math.max(pbHiS, high)
        pbWickS := fresh ? bigUpperWick : (pbWickS or bigUpperWick)
        pbHoldS := close < ema21
        pbActS := true
        pbBarS := bar_index

// ============ COOLDOWN ============
var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars

// ============ QUALITY ============
qVolL  = volX >= volMlt * volSma
qVolS  = qVolL
qSepOK = math.abs(ema9 - ema21) >= qSepMlt * atr14
qVwapL = not useVwapQ or (not na(vwap) and close > vwap)
qVwapS = not useVwapQ or (not na(vwap) and close < vwap)
qRsiL  = rsiLoN <= rsiResetL
qRsiS  = rsiHiN >= rsiResetS

scoreL = (pbWickL ? 1 : 0) + (qVolL ? 1 : 0) + (qSepOK ? 1 : 0) + (qVwapL ? 1 : 0) + (pbHoldL ? 1 : 0) + (qRsiL ? 1 : 0)
scoreS = (pbWickS ? 1 : 0) + (qVolS ? 1 : 0) + (qSepOK ? 1 : 0) + (qVwapS ? 1 : 0) + (pbHoldS ? 1 : 0) + (qRsiS ? 1 : 0)
coreOKL = not coreWV or (pbWickL and qVolL and qVwapL and pbHoldL)
coreOKS = not coreWV or (pbWickS and qVolS and qVwapS and pbHoldS)
qualOKL = (minQual == 0 or scoreL >= minQual) and coreOKL
qualOKS = (minQual == 0 or scoreS >= minQual) and coreOKS

// ============ TRIGGER SETUP ============
var bool  waitL   = false
var int   waitBarL = na
var float trigL   = na
var float slL     = na
var float tpL     = na
var float tp1LvL  = na
var bool  waitS   = false
var int   waitBarS = na
var float trigS   = na
var float slS     = na
var float tpS     = na
var float tp1LvS  = na

gateL = regimeUp and aboveBoth and inSession and atrOK and dowOK and gapAllowL and qualOKL and adxOK and rsi3 < rsiMaxLong
gateS = regimeDn and belowBoth and inSession and atrOK and dowOK and gapAllowS and qualOKS and adxOK and rsi3 > rsiMinShort

if strategy.opentrades == 0 and coolOK and not waitL and pbActL and gateL and not na(pbBarL) and (bar_index - pbBarL) <= reclaimWin
    _slTight = coilOn ? (emaLo3 - clusterBuf * atr14) : (pbLowL - atrBuffMlt * atr14)
    _sl      = stopFloorATR > 0 ? math.min(_slTight, high - stopFloorATR * atr14) : _slTight
    _riskSL  = high - _sl
    if _riskSL > 0 and (maxStopATR == 0 or _riskSL <= maxStopATR * atr14)
        waitL := true
        waitBarL := bar_index
        trigL := high
        slL := _sl
        tpL := high + _riskSL * rMultiple
        tp1LvL := tp1R > 0 ? high + _riskSL * tp1R : na
        pbActL := false

if strategy.opentrades == 0 and coolOK and not waitS and pbActS and gateS and not na(pbBarS) and (bar_index - pbBarS) <= reclaimWin
    _slTight = coilOn ? (emaHi3 + clusterBuf * atr14) : (pbHiS + atrBuffMlt * atr14)
    _sl      = stopFloorATR > 0 ? math.max(_slTight, low + stopFloorATR * atr14) : _slTight
    _riskSL  = _sl - low
    if _riskSL > 0 and (maxStopATR == 0 or _riskSL <= maxStopATR * atr14)
        waitS := true
        waitBarS := bar_index
        trigS := low
        slS := _sl
        tpS := low - _riskSL * rMultiple
        tp1LvS := tp1R > 0 ? low - _riskSL * tp1R : na
        pbActS := false

if waitL and (bar_index - waitBarL > reclaimWin or not regimeUp)
    waitL := false
if waitS and (bar_index - waitBarS > reclaimWin or not regimeDn)
    waitS := false

// ============ ENTRIES ============
if waitL and not inFlat
    strategy.entry("Long", strategy.long, stop=trigL)
else
    strategy.cancel("Long")
if waitS and not inFlat
    strategy.entry("Short", strategy.short, stop=trigS)
else
    strategy.cancel("Short")

longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
var float riskL0 = na
if longEntryBar
    waitL := false
    riskL0 := strategy.position_avg_price - slL
if shortEntryBar
    waitS := false
    riskL0 := slS - strategy.position_avg_price

// ============ EXITS ============
if strategy.position_size > 0
    waitL := false
    _sl = slL
    if beAfterR > 0 and not na(riskL0) and high >= strategy.position_avg_price + beAfterR * riskL0
        _sl := math.max(_sl, strategy.position_avg_price)
    if useTrail and not na(riskL0) and high >= strategy.position_avg_price + riskL0
        _sl := math.max(_sl, strategy.position_avg_price, high - 0.5 * atr14)
    slL := _sl
    if tp1R > 0 and not na(tp1LvL)
        strategy.exit("TP1 L", from_entry="Long", qty_percent=tp1Pct, stop=slL, limit=tp1LvL)
    strategy.exit("Exit L", from_entry="Long", stop=slL, limit=tpL)
    if inFlat
        strategy.close("Long", comment="EOD")
if strategy.position_size < 0
    waitS := false
    _sl = slS
    if beAfterR > 0 and not na(riskL0) and low <= strategy.position_avg_price - beAfterR * riskL0
        _sl := math.min(_sl, strategy.position_avg_price)
    if useTrail and not na(riskL0) and low <= strategy.position_avg_price - riskL0
        _sl := math.min(_sl, strategy.position_avg_price, low + 0.5 * atr14)
    slS := _sl
    if tp1R > 0 and not na(tp1LvS)
        strategy.exit("TP1 S", from_entry="Short", qty_percent=tp1Pct, stop=slS, limit=tp1LvS)
    strategy.exit("Exit S", from_entry="Short", stop=slS, limit=tpS)
    if inFlat
        strategy.close("Short", comment="EOD")

// ============ VISUALS ============
bgcolor(showBG and regimeUp and aboveBoth and adxOK ? color.new(color.green, 92) : na)
bgcolor(showBG and regimeDn and belowBoth and adxOK ? color.new(color.red,   92) : na)
bgcolor(showBG and useSession and not inSession ? color.new(color.gray, 92) : na)
plot(ema9,   "EMA9",   color=color.aqua,   linewidth=1)
plot(ema21,  "EMA21",  color=color.orange, linewidth=1)
plot(ema200, "EMA200", color=color.new(color.purple, 0), linewidth=2)
plot(useVwapQ ? vwap : na, "VWAP", color=color.new(color.white, 0), style=plot.style_circles, linewidth=1)
plotshape(showMarks and longEntryBar,  "Long",  shape.triangleup,   location.belowbar, color.new(color.teal, 0), size=size.small, text="LONG",  textcolor=color.white)
plotshape(showMarks and shortEntryBar, "Short", shape.triangledown, location.abovebar, color.new(color.red, 0),  size=size.small, text="SHORT", textcolor=color.white)
plot(strategy.position_size > 0 ? tpL : strategy.position_size < 0 ? tpS : na, "TP", color=color.new(color.green, 0), style=plot.style_linebr, linewidth=1)
plot(strategy.position_size > 0 ? slL : strategy.position_size < 0 ? slS : na, "SL", color=color.new(color.red, 0), style=plot.style_linebr, linewidth=1)
