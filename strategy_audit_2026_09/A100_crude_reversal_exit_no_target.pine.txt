//@version=5
// CRUDE v4.2 candidate: crude #1's entry/stop UNCHANGED (SHA flip + EMA9/22 alignment, 10-bar swing stop,
// min 1.5xATR floor, skip if >3.0xATR wide). NO fixed R:R target. Exit only via the stop, or a reversal
// exit: close the position when an opposite SHA-flip+trend signal fires (same gating as a fresh entry),
// fill at the next bar's open (Pine's default order timing -- no lookahead). Round 7 harness candidate,
// pre-registration: pre_registration_crude_round7_2026_09_27.md. Commission 0 (gross), matching the
// session-start replay convention already used for crude #1 on this chart.
strategy("CRUDE v4.2 reversal-exit no-target (test)", overlay=true,
     initial_capital=100000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.0,
     max_labels_count=500, max_lines_count=500)

shaLen1  = input.int(10, "SHA pre-smooth EMA", group="Trend")
shaLen2  = input.int(10, "SHA post-smooth EMA", group="Trend")
useEmaAl = input.bool(true, "Require EMA9 vs EMA22 alignment", group="Trend")
swLen    = input.int(10, "SL swing lookback (bars)", group="Risk")
swBuf    = input.float(0.1, "SL buffer beyond swing (x ATR)", step=0.05, group="Risk")
minSL    = input.float(1.5, "SL min distance (x ATR)", step=0.1, group="Risk")
maxSL    = input.float(3.0, "SL max distance (x ATR, skip wider)", step=0.1, group="Risk")
useSess  = input.bool(true, "Session filter", group="Time")
sess     = input.session("0915-2330", "Entry session", group="Time")

o1 = ta.ema(open, shaLen1)
c1 = ta.ema(close, shaLen1)
h1 = ta.ema(high, shaLen1)
l1 = ta.ema(low, shaLen1)
haC = (o1 + h1 + l1 + c1) / 4
var float haO = na
haO := na(haO[1]) ? (o1 + c1) / 2 : (haO[1] + haC[1]) / 2
shaUp = ta.ema(haC, shaLen2) > ta.ema(haO, shaLen2)
flipUp = shaUp and not shaUp[1]
flipDn = not shaUp and shaUp[1]

e9  = ta.ema(close, 9)
e22 = ta.ema(close, 22)
atr = ta.atr(14)
swLo = ta.lowest(low, swLen)
swHi = ta.highest(high, swLen)

trendL = not useEmaAl or e9 > e22
trendS = not useEmaAl or e9 < e22
inSess = not useSess or not na(time(timeframe.period, sess))

riskL = math.max(close - (swLo - swBuf * atr), minSL * atr)
riskS = math.max((swHi + swBuf * atr) - close, minSL * atr)
// sigL/sigS: "this bar is a valid signal in this direction" -- independent of current position,
// used both for fresh entries (gated by flat below) and for the reversal-exit check.
sigL = inSess and flipUp and trendL and riskL <= maxSL * atr
sigS = inSess and flipDn and trendS and riskS <= maxSL * atr and not sigL

flat = strategy.position_size == 0
okL = flat and sigL
okS = flat and sigS

var float slPx = na
var float entryPx = na
if okL
    slPx := close - riskL
    entryPx := close
    strategy.entry("L", strategy.long)
    strategy.exit("xL", "L", stop=slPx)
if okS
    slPx := close + riskS
    entryPx := close
    strategy.entry("S", strategy.short)
    strategy.exit("xS", "S", stop=slPx)

// Reversal exit -- closes at the NEXT bar's open (Pine's default order fill timing), same convention
// as the entries above and as crude #1's own stop/target fills. No same-bar lookahead.
if strategy.position_size > 0 and sigS
    strategy.close("L", comment="REV")
if strategy.position_size < 0 and sigL
    strategy.close("S", comment="REV")

longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
plotshape(longEntryBar,  "Long entry",  shape.triangleup,   location.belowbar, color.new(color.lime, 0),   size=size.small, text="BUY",  textcolor=color.white)
plotshape(shortEntryBar, "Short entry", shape.triangledown, location.abovebar, color.new(color.red, 0),    size=size.small, text="SELL", textcolor=color.white)
plot(strategy.position_size != 0 ? entryPx : na, "Entry", color=color.new(color.blue, 0), style=plot.style_linebr, linewidth=2)
plot(strategy.position_size != 0 ? slPx    : na, "Stop",  color=color.new(color.red, 0),  style=plot.style_linebr, linewidth=2)
plot(e9, "EMA9", color=color.aqua)
plot(e22, "EMA22", color=color.orange)

// ===== AUDIT BLOCK (identical format to crude #1's own -- same parser: n,w,sw,sl,dd) =====
var int aud_from = na
if bar_index == 400
    aud_from := (math.floor(time / 86400000.0) + 1) * 86400000
var float aud_n = 0.0
var float aud_w = 0.0
var float aud_sw = 0.0
var float aud_sl = 0.0
var float aud_eq = 0.0
var float aud_pk = 0.0
var float aud_mn = 0.0
var float aud_dd = 0.0
var float aud_d2 = 0.0
var float aud_dayP = 0.0
var int aud_nD = 0
var int aud_dayStart = 0
aud_newDay = ta.change(time("D")) != 0
if aud_newDay
    if not na(aud_from) and aud_dayStart >= aud_from
        aud_nD += 1
        aud_d2 += aud_dayP * aud_dayP
    aud_dayP := 0.0
    aud_dayStart := time
aud_nc = strategy.closedtrades - nz(strategy.closedtrades[1])
if aud_nc > 0
    for aud_k = 0 to aud_nc - 1
        aud_i = strategy.closedtrades - 1 - aud_k
        if not na(aud_from) and strategy.closedtrades.entry_time(aud_i) >= aud_from
            aud_sz = math.abs(strategy.closedtrades.size(aud_i))
            aud_p = strategy.closedtrades.profit(aud_i) / syminfo.pointvalue / (aud_sz > 0 ? aud_sz : 1)
            aud_n += 1
            if aud_p > 0
                aud_w += 1
                aud_sw += aud_p
            else
                aud_sl += aud_p
            aud_eq += aud_p
            aud_pk := math.max(aud_pk, aud_eq)
            aud_mn := math.min(aud_mn, aud_eq)
            aud_dd := math.max(aud_dd, aud_pk - aud_eq)
            aud_dayP += aud_p
if barstate.islastconfirmedhistory or barstate.islast
    var table aud_t = table.new(position.bottom_center, 1, 1, bgcolor=color.new(color.black, 10))
    aud_incl = not na(aud_from) and aud_dayStart >= aud_from
    table.cell(aud_t, 0, 0, "AUD|" + syminfo.ticker + "|" + timeframe.period + "|from=" + str.tostring(aud_from) + "|last=" + str.tostring(time) + "|nD=" + str.tostring(aud_nD + (aud_incl ? 1 : 0)) + "|pv=" + str.tostring(syminfo.pointvalue) + "|" + str.tostring(aud_n) + "," + str.tostring(aud_w) + "," + str.tostring(aud_sw, "#.#") + "," + str.tostring(aud_sl, "#.#") + "," + str.tostring(aud_dd, "#.#") + "," + str.tostring(aud_d2 + (aud_incl ? aud_dayP * aud_dayP : 0), "#.#") + "," + str.tostring(aud_pk, "#.#") + "," + str.tostring(aud_mn, "#.#"), text_color=color.yellow, text_size=size.tiny, text_halign=text.align_left)
