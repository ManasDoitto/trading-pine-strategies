//@version=5
// ============================================================================
//  BankNifty PORTFOLIO  --  v0.4 EMA pullback + v13 sweep-fade "bear"
//  NSE:BANKNIFTY1!  5-minute  --  two independent modules, same capital
//  Derived from "bnf combined v1" (already tested this project) with module
//  C (MTF v1.1) removed -- v0.4 and v13's logic is otherwise BYTE-IDENTICAL
//  to that combined script's modules A and B, which were themselves ports
//  of the standalone v13/v0.4 files. Nothing in either module's entry,
//  stop, target, or session logic was changed.
//  v13 sweep-fade "bear": fade PDH/PDL sweeps + intraday swing/opening-range
//    level rejections, only while the 8-day daily SMA is flat/falling;
//    stop beyond the wick (cap 120 pts), target 6R.
//  v0.4 EMA pullback: 9/21 trend, pullback to EMA9 with a rejection wick,
//    reclaim bar with above-average volume, on the VWAP side, holding EMA21;
//    only while the last completed 15m ADX >= 25; stop-entry, 2.5R.
//  Position handling: each module trades 1 lot with its own entry/exit IDs
//  (pyramiding = 2), so both modules can hold trades AT THE SAME TIME in the
//  SAME direction. TradingView keeps one net position per script, so a
//  module may NOT open (or arm) a trade opposite to one the other module
//  holds or has armed -- that signal is skipped, never overwritten.
//  Costs: 0.02%/side commission + 5 pts slippage per fill (same as both
//  standalone scripts). Flat by 15:15 IST.
// ============================================================================

aAddLevel(float p, bool isHi) =>
    if not na(p) and p > 0
        bool dup = false
        int n = array.size(aLvlP)
        if n > 0
            for i = 0 to n - 1
                if array.get(aLvlHi, i) == isHi and math.abs(array.get(aLvlP, i) - p) < 30.0
                    dup := true
        if not dup
            array.push(aLvlP, p)
            array.push(aLvlHi, isHi)
            array.push(aLvlBorn, bar_index)
            array.push(aLvlUsed, false)
            if array.size(aLvlP) > 24
                array.shift(aLvlP)
                array.shift(aLvlHi)
                array.shift(aLvlBorn)
                array.shift(aLvlUsed)

aMarkUsed(float L, bool isHi) =>
    if array.size(aLvlP) > 0
        for i = 0 to array.size(aLvlP) - 1
            if array.get(aLvlHi, i) == isHi and math.abs(array.get(aLvlP, i) - L) < 1
                array.set(aLvlUsed, i, true)

if array.size(aLvlP) > 0
    for i = array.size(aLvlP) - 1 to 0
        L = array.get(aLvlP, i)
        isHi = array.get(aLvlHi, i)
        broken = isHi ? (close > L + 0.6 * aAvgRng) : (close < L - 0.6 * aAvgRng)
        if array.get(aLvlUsed, i) or broken or (bar_index - array.get(aLvlBorn, i)) > 48
            array.remove(aLvlP, i)
            array.remove(aLvlHi, i)
            array.remove(aLvlBorn, i)
            array.remove(aLvlUsed, i)

f_add(float[] n, float[] w, float[] sw, float[] sl, float[] eq, float[] pk, float[] mn, float[] dd, int j, float p) =>
    array.set(n, j, array.get(n, j) + 1)
    if p > 0
        array.set(w, j, array.get(w, j) + 1)
        array.set(sw, j, array.get(sw, j) + p)
    else
        array.set(sl, j, array.get(sl, j) + p)
    e = array.get(eq, j) + p
    array.set(eq, j, e)
    array.set(pk, j, math.max(array.get(pk, j), e))
    array.set(mn, j, math.min(array.get(mn, j), e))
    array.set(dd, j, math.max(array.get(dd, j), array.get(pk, j) - e))
