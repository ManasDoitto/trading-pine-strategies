//@version=6
// ============================================================================
//  Silver v5.3 — Optimized v5.x combining sweep-winning params with v4.1's
//  low-filter approach. 25,000-iteration sweep winner.
//  MCX:SILVER1!, 5m
//  ---------------------------------------------------------------------------
//  From Python sweep (25,000 iterations on 4.7 months cached data):
//  Winner: ADX>=5, RR=3.0, PB=3.0, SWB=0.15, SHA=1, ATR_min=0, MinSL=1.0, BO=10
//  Results: 106 trades (22.6/mo), PF 1.395, net +22,684 pts, L/W 0.47
//  Best net: Same config with MinSL=3.0: 108 trades, PF 1.395, net +24,895
//  ---------------------------------------------------------------------------
//  Key insight: Very loose filters (ADX>=5, PB=3.0, SHA=1) are close to v4.1's
//  "no filter" approach but retain EMA200 trend alignment for quality.
// ============================================================================
strategy("SILVER v5.3 optimized (sweep winner)", overlay=true,
     initial_capital=100000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent,
     commission_value=0.02,
     max_labels_count=500, max_lines_count=500,
     calc_on_order_fills=false,
     calc_on_every_tick=false)

// ============ INPUTS — sweep-optimized for Silver ============
// Trend
shaLen1  = input.int(10, "SHA pre-smooth EMA", group="Trend")
shaLen2  = input.int(10, "SHA post-smooth EMA", group="Trend")
useEma200 = input.bool(true, "Require close vs EMA200 filter", group="Trend")
shaMinHold = input.int(1, "SHA must hold >= N bars before flip", group="Trend")

// Time
useSess  = input.bool(true, "Session filter", group="Time")
sessTime = input.session("0915-2330", "Entry session (IST)", group="Time")
forceFlatTime = input.session("2245-2330", "Force-flat window", group="Time")

// ADX Gate
htfAdxLen = input.int(14, "15m ADX length", group="ADX Gate")
htfAdxMin = input.float(5.0, "15m ADX min threshold (0=off)", step=1, group="ADX Gate")

// Pullback proximity (loose = 3.0 = allow entries far from EMA9)
usePullback  = input.bool(true, "Enable pullback proximity filter", group="Entry Quality")
pbAtrMult    = input.float(3.0, "Max distance from EMA9 (x ATR)", step=0.5, group="Entry Quality")

// Donchian breakout
useDonchian  = input.bool(true, "Enable Donchian breakout entry", group="Breakout")
boLookback   = input.int(10, "Donchian breakout lookback (bars)", minval=2, group="Breakout")

// Risk
swLen    = input.int(10, "SL swing lookback (bars)", group="Risk")
swBuf    = input.float(0.15, "SL buffer beyond swing (x ATR)", step=0.05, group="Risk")
minSL    = input.float(1.0, "SL min distance (x ATR)", step=0.5, group="Risk")
maxSL    = input.float(5.0, "SL max distance (x ATR, skip wider)", step=0.5, group="Risk")
rr       = input.float(3.0, "Target R:R (fixed)", step=0.5, group="Risk")
dayLossLimit = input.float(350.0, "Daily loss limit (pts, 0=off)", step=10, group="Risk Management")

// ============ SHA CALCULATION ============
o1 = ta.ema(open, shaLen1)
c1 = ta.ema(close, shaLen1)
h1 = ta.ema(high, shaLen1)
l1 = ta.ema(low, shaLen1)
haC = (o1 + h1 + l1 + c1) / 4.0
var float haO = na
haO := na(haO[1]) ? (o1 + c1) / 2.0 : (haO[1] + haC[1]) / 2.0
shaUp = ta.ema(haC, shaLen2) > ta.ema(haO, shaLen2)
flipUp = shaUp and not shaUp[1]
flipDn = not shaUp and shaUp[1]

// SHA stability filter
shaFlipDiff = ta.valuewhen(shaUp != shaUp[1], bar_index, 0)
shaPrevFlip = ta.valuewhen(shaUp != shaUp[1], bar_index, 1)
shaPrevGroupSize = na(shaPrevFlip) ? 1 : (shaFlipDiff - shaPrevFlip)
shaStable = not na(shaPrevFlip) and (shaPrevGroupSize >= shaMinHold)

// ============ INDICATORS ============
e9 = ta.ema(close, 9)
e22 = ta.ema(close, 22)
e200 = ta.ema(close, 200)
atr = ta.atr(14)
swLo = ta.lowest(low, swLen)
swHi = ta.highest(high, swLen)

// Trend alignment
trendL = (e9 > e22) and (not useEma200 or close > e200)
trendS = (e9 < e22) and (not useEma200 or close < e200)
inSess = not useSess or not na(time(timeframe.period, sessTime))
inForceFlat = not na(time(timeframe.period, forceFlatTime))
atrOK = atr >= 0  // always true (atrMinPts=0 in sweep winner)

// ADX gate
f_adxPrev() =>
    [_p, _m, _a] = ta.dmi(htfAdxLen, htfAdxLen)
    _a[1]
hAdx = request.security(syminfo.tickerid, "15", f_adxPrev(), lookahead=barmerge.lookahead_on)
adxOK = htfAdxMin == 0 or (not na(hAdx) and hAdx >= htfAdxMin)

// EMA9 pullback proximity
nearEma9L = not usePullback or close <= e9 + pbAtrMult * atr
nearEma9S = not usePullback or close >= e9 - pbAtrMult * atr

// Donchian breakout
dcHi = useDonchian ? ta.highest(high, boLookback)[1] : na
dcLo = useDonchian ? ta.lowest(low, boLookback)[1] : na
breakoutL = useDonchian and close > dcHi
breakoutS = useDonchian and close < dcLo

// Risk
riskL = math.max(close - (swLo - swBuf * atr), minSL * atr)
riskS = math.max((swHi + swBuf * atr) - close, minSL * atr)
cap = maxSL * atr

// ============ ENTRY CONDITIONS ============
flat = strategy.position_size == 0

// Flip entry (SHA flip)
okFlipL = flipUp and trendL and shaStable and atrOK and adxOK and nearEma9L and (riskL <= cap) and not inForceFlat
okFlipS = flipDn and trendS and shaStable and atrOK and adxOK and nearEma9S and (riskS <= cap) and not inForceFlat

// Donchian breakout entry
okBoL = breakoutL and trendL and atrOK and adxOK and nearEma9L and (riskL <= cap) and not inForceFlat
okBoS = breakoutS and trendS and atrOK and adxOK and nearEma9S and (riskS <= cap) and not okBoL and not inForceFlat

// Combined: flip OR breakout (either can trigger)
okL = inSess and flat and (okFlipL or okBoL)
okS = inSess and flat and (okFlipS or okBoS) and not okL

// ============ DAILY LOSS CIRCUIT BREAKER ============
newDay = ta.change(time("D")) != 0
var float dayRealized = 0.0
var bool dayLocked = false
if newDay
    dayRealized := 0.0
    dayLocked := false
nClosed = strategy.closedtrades - nz(strategy.closedtrades[1])
if nClosed > 0
    for k = 0 to nClosed - 1
        i = strategy.closedtrades - 1 - k
        dayRealized += strategy.closedtrades.profit(i) / syminfo.pointvalue
if dayLossLimit > 0 and dayRealized <= -dayLossLimit and not dayLocked
    dayLocked := true
    strategy.close_all(comment="Daily loss limit")

// ============ ENTRIES ============
var float slPx = na
var float tpPx = na
var float entryPx = na
var string entryKind = ""
var label entryLbl = na

if okL and not dayLocked
    slPx := close - riskL
    tpPx := close + rr * riskL
    entryPx := close
    strategy.entry("L", strategy.long)
    strategy.exit("xL", "L", stop=slPx, limit=tpPx)
    entryKind := "FLIP"
    if okBoL and not okFlipL
        entryKind := "BO"

if okS and not dayLocked
    slPx := close + riskS
    tpPx := close - rr * riskS
    entryPx := close
    strategy.entry("S", strategy.short)
    strategy.exit("xS", "S", stop=slPx, limit=tpPx)
    entryKind := "FLIP"
    if okBoS and not okFlipS
        entryKind := "BO"

// ============ FORCED FLAT (session end) ============
if strategy.position_size != 0 and inForceFlat
    strategy.close_all(comment="FORCE FLAT")

// ============ VISUALS ============
longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
plotshape(longEntryBar and okFlipL, "Flip Long", shape.triangleup, location.belowbar, color.new(color.lime, 0), size=size.small, text="F")
plotshape(longEntryBar and okBoL and not okFlipL, "BO Long", shape.triangleup, location.belowbar, color.new(color.yellow, 0), size=size.small, text="B")
plotshape(shortEntryBar and okFlipS, "Flip Short", shape.triangledown, location.abovebar, color.new(color.red, 0), size=size.small, text="F")
plotshape(shortEntryBar and okBoS and not okFlipS, "BO Short", shape.triangledown, location.abovebar, color.new(color.orange, 0), size=size.small, text="B")
plot(e9, "EMA9", color=color.new(color.aqua, 0))
plot(e22, "EMA22", color=color.new(color.orange, 0))
plot(e200, "EMA200", color=color.new(color.purple, 0), linewidth=1)
bgcolor(dayLocked ? color.new(color.red, 85) : na)
