//@version=6
// ============================================================================
//  SILVER v4.1 = working_strategies #1 (v4.0 wide-ATR + fixed 350 daily limit)
//                + Donchian breakout as a SECOND entry trigger.
//  MCX:SILVER1!, 5m.  FORWARD TEST build.
//  ---------------------------------------------------------------------------
//  Base is UNCHANGED from #1: SHA 10/10 flip + EMA9>EMA22 only. No EMA200, no
//  EMA9-pullback gate, no ADX gate, no vol-regime filter, no SHA-stability
//  filter. Those are the v5.x filters and every one of them cost points
//  (strategy_audit_2026_09/v40_breakout_results_2026_09_26.md).
//
//  The ONLY addition: entry = SHA flip OR Donchian breakout of the prior
//  boLookback-bar high/low. Channel is shifted [1] so there is no lookahead.
//
//  Harness results, 30 months (2024-03-25..2026-09-24), gross points, qty 1:
//    incumbent flip-only   747 tr  PF 1.503  +222,240 pts  maxDD 33,693  conc 61.4%
//    boLookback = 5 (B2)   825 tr  PF 1.519  +307,265 pts  maxDD 30,646  conc 72.0%
//    boLookback = 3 (B1)   836 tr  PF 1.563  +337,941 pts  maxDD 71,324  conc 55.4%
//  B2 is the default: more points AND lower drawdown than the incumbent.
//  Set boLookback = 3 for the B1 variant (more points, 2.1x the drawdown).
//
//  NOT yet validated out of sample. The lookback was chosen from six values on
//  the same history. That is what this forward test is for.
// ============================================================================
strategy("SILVER v4.1 flip+breakout (forward test)", overlay=true,
     initial_capital=100000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent,
     commission_value=0.02,
     max_labels_count=500, max_lines_count=500)

// ============ INPUTS (base frozen at working_strategies/Silver #1) ============
shaLen1  = input.int(10, "SHA pre-smooth EMA", group="Trend")
shaLen2  = input.int(10, "SHA post-smooth EMA", group="Trend")
useEmaAl = input.bool(true, "Require EMA9 vs EMA22 alignment", group="Trend")
swLen    = input.int(10, "SL swing lookback (bars)", group="Risk")
swBuf    = input.float(0.1, "SL buffer beyond swing (x ATR)", step=0.05, group="Risk")
minSL    = input.float(2.5, "SL min distance (x ATR)", step=0.1, group="Risk")
maxSL    = input.float(5.0, "SL max distance (x ATR, skip wider)", step=0.1, group="Risk")
rr       = input.float(3.0, "Target R:R (fixed, no trailing)", step=0.1, group="Risk")
dayLossLimit = input.float(350.0, "Daily loss limit (pts, 0=off)", step=10, group="Risk")
useSess  = input.bool(true, "Session filter", group="Time")
sess     = input.session("0915-2330", "Entry session", group="Time")

// ---- the one addition ----
useBreakout = input.bool(true, "Enable Donchian breakout entry", group="Breakout")
boLookback  = input.int(5, "Donchian lookback (5 = B2 default, 3 = B1)", minval=2, group="Breakout")

// ---- forward-test boundary: everything at/after this is out-of-sample ----
fwdStart = input.time(timestamp("27 Sep 2026 00:00 +0530"), "Forward-test start", group="Forward test")

// ============ SHA ============
o1 = ta.ema(open, shaLen1)
c1 = ta.ema(close, shaLen1)
h1 = ta.ema(high, shaLen1)
l1 = ta.ema(low, shaLen1)
haC = (o1 + h1 + l1 + c1) / 4
var float haO = na
haO := na(haO[1]) ? (o1 + c1) / 2 : (haO[1] + haC[1]) / 2
shaUp = ta.ema(haC, shaLen2) > ta.ema(haO, shaLen2)
flipUp = shaUp and not shaUp[1]
flipDn = not shaUp and shaUp[1]

// ============ INDICATORS ============
e9   = ta.ema(close, 9)
e22  = ta.ema(close, 22)
atr  = ta.atr(14)
swLo = ta.lowest(low, swLen)
swHi = ta.highest(high, swLen)

trendL = not useEmaAl or e9 > e22
trendS = not useEmaAl or e9 < e22
inSess = not useSess or not na(time(timeframe.period, sess))
flat = strategy.position_size == 0

// Donchian breakout, channel shifted one bar so the current bar cannot see itself
dcHi = ta.highest(high, boLookback)[1]
dcLo = ta.lowest(low, boLookback)[1]
breakoutL = useBreakout and close > dcHi
breakoutS = useBreakout and close < dcLo

riskL = math.max(close - (swLo - swBuf * atr), minSL * atr)
riskS = math.max((swHi + swBuf * atr) - close, minSL * atr)
cap = maxSL * atr

// ============ ENTRY: flip OR breakout ============
okL = flat and inSess and (flipUp or breakoutL) and trendL and riskL <= cap
okS = flat and inSess and (flipDn or breakoutS) and trendS and riskS <= cap and not okL

// ============ DAILY LOSS CIRCUIT BREAKER (unchanged from #1) ============
newDay = ta.change(time("D")) != 0
var float dayRealized = 0.0
var bool  dayLocked = false
if newDay
    dayRealized := 0.0
    dayLocked := false
nClosed = strategy.closedtrades - nz(strategy.closedtrades[1])
if nClosed > 0
    for k = 0 to nClosed - 1
        i = strategy.closedtrades - 1 - k
        dayRealized += strategy.closedtrades.profit(i) / syminfo.pointvalue
if dayLossLimit > 0 and dayRealized <= -dayLossLimit and not dayLocked
    dayLocked := true
    strategy.close_all(comment="Daily loss limit")

// ============ ENTRIES ============
var float slPx = na
var float tpPx = na
var float entryPx = na
var string entryKind = ""
if okL and not dayLocked
    slPx := close - riskL
    tpPx := close + rr * riskL
    entryPx := close
    entryKind := flipUp ? "FLIP" : "BO"
    strategy.entry("L", strategy.long)
    strategy.exit("xL", "L", stop=slPx, limit=tpPx)
    alert("🟢 SILVER LONG | Entry: " + str.tostring(close, "#.0") + " | SL: " + str.tostring(slPx, "#.0") + " | TP: " + str.tostring(tpPx, "#.0") + " | " + entryKind, alert.freq_once_per_bar)
if okS and not dayLocked
    slPx := close + riskS
    tpPx := close - rr * riskS
    entryPx := close
    entryKind := flipDn ? "FLIP" : "BO"
    strategy.entry("S", strategy.short)
    strategy.exit("xS", "S", stop=slPx, limit=tpPx)
    alert("🔴 SILVER SHORT | Entry: " + str.tostring(close, "#.0") + " | SL: " + str.tostring(slPx, "#.0") + " | TP: " + str.tostring(tpPx, "#.0") + " | " + entryKind, alert.freq_once_per_bar)

// ============ BACKTEST vs FORWARD-TEST SPLIT TABLE ============
var int   bN = 0
var int   bW = 0
var float bWin = 0.0
var float bLos = 0.0
var float bEq = 0.0
var float bPk = 0.0
var float bDD = 0.0
var int   fN = 0
var int   fW = 0
var float fWin = 0.0
var float fLos = 0.0
var float fEq = 0.0
var float fPk = 0.0
var float fDD = 0.0
var int   fFlip = 0
var int   fBO = 0

if nClosed > 0
    for k = 0 to nClosed - 1
        idx = strategy.closedtrades - 1 - k
        sz = math.abs(strategy.closedtrades.size(idx))
        p = strategy.closedtrades.profit(idx) / syminfo.pointvalue / (sz > 0 ? sz : 1)
        isFwd = strategy.closedtrades.entry_time(idx) >= fwdStart
        if isFwd
            fN += 1
            if p > 0
                fW += 1
                fWin += p
            else
                fLos += p
            fEq += p
            fPk := math.max(fPk, fEq)
            fDD := math.max(fDD, fPk - fEq)
        else
            bN += 1
            if p > 0
                bW += 1
                bWin += p
            else
                bLos += p
            bEq += p
            bPk := math.max(bPk, bEq)
            bDD := math.max(bDD, bPk - bEq)

if barstate.islastconfirmedhistory or barstate.islast
    var table t = table.new(position.bottom_right, 5, 4, border_width=1,
         bgcolor=color.new(color.black, 15))
    table.cell(t, 0, 0, "SILVER v4.1", text_color=color.white, bgcolor=color.new(color.navy, 0))
    table.cell(t, 1, 0, "trades", text_color=color.white, bgcolor=color.new(color.navy, 0))
    table.cell(t, 2, 0, "PF", text_color=color.white, bgcolor=color.new(color.navy, 0))
    table.cell(t, 3, 0, "net pts", text_color=color.white, bgcolor=color.new(color.navy, 0))
    table.cell(t, 4, 0, "maxDD", text_color=color.white, bgcolor=color.new(color.navy, 0))
    table.cell(t, 0, 1, "BACKTEST", text_color=color.gray)
    table.cell(t, 1, 1, str.tostring(bN), text_color=color.gray)
    table.cell(t, 2, 1, bLos < 0 ? str.tostring(bWin / -bLos, "#.###") : "-", text_color=color.gray)
    table.cell(t, 3, 1, str.tostring(bEq, "#"), text_color=color.gray)
    table.cell(t, 4, 1, str.tostring(bDD, "#"), text_color=color.gray)
    table.cell(t, 0, 2, "FORWARD", text_color=color.yellow)
    table.cell(t, 1, 2, str.tostring(fN), text_color=color.yellow)
    table.cell(t, 2, 2, fLos < 0 ? str.tostring(fWin / -fLos, "#.###") : "-", text_color=color.yellow)
    table.cell(t, 3, 2, str.tostring(fEq, "#"), text_color=color.yellow)
    table.cell(t, 4, 2, str.tostring(fDD, "#"), text_color=color.yellow)
    table.cell(t, 0, 3, "FWD win%", text_color=color.yellow)
    table.cell(t, 1, 3, fN > 0 ? str.tostring(100.0 * fW / fN, "#.#") : "-", text_color=color.yellow)
    table.cell(t, 2, 3, "lookback", text_color=color.yellow)
    table.cell(t, 3, 3, str.tostring(boLookback), text_color=color.yellow)
    table.cell(t, 4, 3, "pv " + str.tostring(syminfo.pointvalue), text_color=color.yellow)

// ============ VISUALS ============
longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
plotshape(longEntryBar,  "Long entry",  shape.triangleup,   location.belowbar, color.new(color.lime, 0), size=size.tiny)
plotshape(shortEntryBar, "Short entry", shape.triangledown, location.abovebar, color.new(color.red, 0),  size=size.tiny)
plot(strategy.position_size != 0 ? slPx : na, "Stop",   color=color.new(color.red, 0),   style=plot.style_linebr)
plot(strategy.position_size != 0 ? tpPx : na, "Target", color=color.new(color.green, 0), style=plot.style_linebr)
plot(e9, "EMA9", color=color.new(color.aqua, 0))
plot(e22, "EMA22", color=color.new(color.orange, 0))
bgcolor(dayLocked ? color.new(color.red, 88) : na)
bgcolor(time >= fwdStart ? color.new(color.blue, 93) : na)
