//@version=6
// ============================================================================
//  STOCKS-IN-PLAY  MA BASE BREAKOUT  v1.0  -- liquid NSE stock options, 5m
//  ---------------------------------------------------------------------------
//  Built from the 9EMA / 20SMA / 200SMA day-trading method (YouTube 2e4WayvjLJo),
//  mechanised and validated on Dhan 5m history (Jan 2018 - Sep 2026) for all 210
//  NSE stock-option underlyings (stockopt/ in this repo). Not tuned to any one
//  stock: every threshold is ATR- or percent-normalised.
//
//  What the video says                  -> what this script does
//  "trade stocks with a catalyst"       -> gap >= 0.75% AND first-30-min volume
//                                          >= 1.25x its 20-day average
//  "start from the daily chart"         -> prior close above the daily 20SMA for
//                                          longs (below for shorts)
//  "never trade a flat 20SMA"           -> 20SMA slope >= 0.15 ATR over 10 bars
//  "9EMA above the 20SMA in an uptrend" -> EMA9 > SMA20 (mirror for shorts)
//  "don't chase an extended move"       -> close within 2.5 ATR of the 20SMA
//  "base breakout" (setup #2)           -> 15-bar (75 min) base, range <= 2.5 ATR,
//                                          on the 9/20 zone; stop order past it
//  "momentum / volume behind the move"  -> signal-bar volume >= its 20-day
//                                          same-time-of-day average
//  "stocks showing strength" (A+ tier)  -> stock beats NIFTY since the open by
//                                          >= 1.5% in the trade direction
//
//  Exit: stop past the base, 5R target, flat at 15:15. No trailing -- trailing
//  under the 20SMA raised the win rate but cut the edge out of sample.
//  Rejected in testing: the video's pullback setup (PF 1.87 -> 1.24), a 15m
//  higher-timeframe gate, an upper slope cap, and NIFTY-direction alignment.
//
//  Locked-spec results, 1 unit, gross, 210 stocks (see README.md in this folder):
//    OOS 2025-01..2026-09  Core  916 tr  win 45.7%  PF 1.45  +0.19%/trade
//                          A+    385 tr  win 49.9%  PF 1.70  +0.27%/trade
//
//  OPTION BUYING: long = buy ATM/1-ITM CE, short = buy ATM PE. Entries 09:45-13:15.
//  Pine sees one chart; stockopt/scan_live.py applies the same rules to all 210.
// ============================================================================
strategy("Stocks-in-Play MA Base Breakout v1.0", overlay=true,
     initial_capital=100000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.0,
     process_orders_on_close=true,
     max_labels_count=500)

// ============================ INPUTS ============================
emaLen      = input.int(9,    "Fast EMA",                    group="Moving averages")
smaLen      = input.int(20,   "Mid SMA",                     group="Moving averages")
longLen     = input.int(200,  "Long SMA (plot only)",        group="Moving averages")
atrLen      = input.int(14,   "ATR length",                  group="Moving averages")

gapMin      = input.float(0.75, "Min |gap| %",                         step=0.05, group="Catalyst (stock in play)")
orvolMin    = input.float(1.25, "Min opening-volume ratio (x 20d avg)", step=0.05, group="Catalyst (stock in play)")
useCatalyst = input.bool(true,  "Require catalyst",                               group="Catalyst (stock in play)")
useDaily    = input.bool(true,  "Daily bias: prior close vs daily 20SMA",         group="Catalyst (stock in play)")

slopeLb     = input.int(10,     "20SMA slope lookback (bars)",       group="Trend")
slopeMin    = input.float(0.15, "Min 20SMA slope (x ATR)", step=0.05, group="Trend")
extMax      = input.float(2.5,  "Max distance close-20SMA (x ATR)", step=0.1, group="Trend")

baseBars    = input.int(15,     "Base length (bars)",               group="Base breakout")
baseTight   = input.float(2.5,  "Max base range (x ATR)", step=0.1, group="Base breakout")
baseZone    = input.float(0.5,  "Base low within x ATR of 9EMA", step=0.1, group="Base breakout")
rvolMin     = input.float(1.0,  "Min signal-bar volume (x same-slot 20d avg)", step=0.1, group="Base breakout")
armBars     = input.int(3,      "Buy/sell-stop valid for (bars)",   group="Base breakout")

aplusOnly   = input.bool(false, "Trade A+ only",                    group="A+ tier (relative strength)")
rsMin       = input.float(1.5,  "A+: stock minus NIFTY since open (%)", step=0.25, group="A+ tier (relative strength)")
indexSym    = input.symbol("NSE:NIFTY", "Index for relative strength",   group="A+ tier (relative strength)")

stopBuf     = input.float(0.15, "Stop beyond base (x ATR)",  step=0.05, group="Risk")
minRisk     = input.float(0.8,  "Min stop distance (x ATR)", step=0.1,  group="Risk")
rr          = input.float(5.0,  "Target (R)",                step=0.5,  group="Risk")
trailAfter  = input.float(0.0,  "Trail under 20SMA after (R), 0 = off", step=0.25, group="Risk")
trailBuf    = input.float(0.25, "Trail buffer (x ATR)",      step=0.05, group="Risk")

entryFrom   = input.int(30,  "No entries before (min after 09:15)", group="Session")
entryTo     = input.int(240, "No entries after (min after 09:15)",  group="Session")
eodMin      = input.int(360, "Flat at (min after 09:15)",           group="Session")
maxTrades   = input.int(2,   "Max trades per day",                  group="Session")
showTable   = input.bool(true, "Show stats table",                  group="Display")

// ============================ SERIES ============================
ema9   = ta.ema(close, emaLen)
sma20  = ta.sma(close, smaLen)
sma200 = ta.sma(close, longLen)
atr    = ta.atr(atrLen)

// minutes since 09:15 IST of this bar's open
mins = (hour(time, "Asia/Kolkata") * 60 + minute(time, "Asia/Kolkata")) - (9 * 60 + 15)
newDay = time("D") != time("D")[1]

// previous completed session only (non-repainting)
[dClosePrev, dSma20Prev] = request.security(syminfo.tickerid, "D",
     [close[1], ta.sma(close, 20)[1]], lookahead=barmerge.lookahead_on)

// ---- catalyst: gap and opening relative volume ----
var float dayOpen = na
var float ovToday = 0.0
var array<float> ovHist = array.new<float>()
if newDay
    if ovToday > 0
        array.push(ovHist, ovToday)
        if array.size(ovHist) > 20
            array.shift(ovHist)
    ovToday := 0.0
    dayOpen := open
if mins < 30
    ovToday += volume
gapPct = dClosePrev > 0 ? (dayOpen - dClosePrev) / dClosePrev * 100.0 : na
orvol  = array.size(ovHist) >= 5 ? ovToday / array.avg(ovHist) : na
catalystOK = not useCatalyst or (math.abs(gapPct) >= gapMin and orvol >= orvolMin)

// ---- same-time-of-day relative volume (20-session ring buffer per 5m slot) ----
var matrix<float> slotVol = matrix.new<float>(75, 20, na)
var int ringCol = 0
if newDay
    ringCol := (ringCol + 1) % 20
slot = math.max(0, math.min(74, int(mins / 5)))
float vSum = 0.0
int vCnt = 0
for k = 0 to 19
    x = matrix.get(slotVol, slot, k)
    if not na(x)
        vSum += x
        vCnt += 1
rvol = vCnt >= 5 and vSum > 0 ? volume / (vSum / vCnt) : na
matrix.set(slotVol, slot, ringCol, volume)   // written after use: window = previous 20 sessions

// ---- relative strength vs NIFTY since the open ----
nClose   = request.security(indexSym, timeframe.period, close)
nDayOpen = request.security(indexSym, "D", open, lookahead=barmerge.lookahead_on)
rs = (close / dayOpen - 1.0) * 100.0 - (nClose / nDayOpen - 1.0) * 100.0

// ---- trend stack ----
slope  = (sma20 - sma20[slopeLb]) / atr
ext    = (close - sma20) / atr
volOK  = rvol >= rvolMin
stackL = ema9 > sma20 and slope >= slopeMin and ext <= extMax and volOK and (not useDaily or dClosePrev > dSma20Prev)
stackS = ema9 < sma20 and slope <= -slopeMin and -ext <= extMax and volOK and (not useDaily or dClosePrev < dSma20Prev)
if aplusOnly
    stackL := stackL and rs >= rsMin
    stackS := stackS and -rs >= rsMin

// ---- base ----
bh    = ta.highest(high, baseBars)
bl    = ta.lowest(low, baseBars)
tight = (bh - bl) <= baseTight * atr
sigL  = stackL and tight and bl <= ema9 + baseZone * atr and bh >= sma20
sigS  = not sigL and stackS and tight and bh >= ema9 - baseZone * atr and bl <= sma20
aplusL = sigL and rs >= rsMin
aplusS = sigS and -rs >= rsMin

// ---- trades today ----
var int entriesAtDayStart = 0
entriesTotal = strategy.opentrades + strategy.closedtrades
if newDay
    entriesAtDayStart := entriesTotal - strategy.opentrades
tradesToday = entriesTotal - entriesAtDayStart

inWindow = mins >= entryFrom and mins <= entryTo
flat = strategy.position_size == 0

// ============================ ORDERS ============================
var int   armBar   = na
var float plannedStop = na
var float plannedRisk = na
var float trailStop   = na
var float extremeSinceEntry = na

canArm = flat and inWindow and catalystOK and tradesToday < maxTrades and atr > 0
if canArm and sigL
    px  = bh + syminfo.mintick
    stp = bl - stopBuf * atr
    stp := px - stp < minRisk * atr ? px - minRisk * atr : stp
    strategy.cancel("S")
    strategy.entry("L", strategy.long, stop=px, comment=aplusL ? "A+ L" : "L")
    strategy.exit("xL", from_entry="L", stop=stp, limit=px + rr * (px - stp))
    armBar := bar_index
    plannedStop := stp
    plannedRisk := px - stp
else if canArm and sigS
    px  = bl - syminfo.mintick
    stp = bh + stopBuf * atr
    stp := stp - px < minRisk * atr ? px + minRisk * atr : stp
    strategy.cancel("L")
    strategy.entry("S", strategy.short, stop=px, comment=aplusS ? "A+ S" : "S")
    strategy.exit("xS", from_entry="S", stop=stp, limit=px - rr * (stp - px))
    armBar := bar_index
    plannedStop := stp
    plannedRisk := stp - px

// pending order expiry: valid for armBars bars, never past the entry window
if flat and not na(armBar) and (bar_index - armBar >= armBars or mins >= entryTo)
    strategy.cancel("L")
    strategy.cancel("S")
    armBar := na

// ---- manage open position: re-anchor the target to the real fill ----
if strategy.position_size != 0
    entryPx = strategy.position_avg_price
    if strategy.position_size[1] == 0
        trailStop := plannedStop
        extremeSinceEntry := strategy.position_size > 0 ? high : low
        armBar := na
    risk = strategy.position_size > 0 ? entryPx - plannedStop : plannedStop - entryPx
    risk := risk > 0 ? risk : plannedRisk
    if strategy.position_size > 0
        extremeSinceEntry := math.max(extremeSinceEntry, high)
        if trailAfter > 0 and (extremeSinceEntry - entryPx) / risk >= trailAfter
            trailStop := math.max(trailStop, sma20 - trailBuf * atr)
        strategy.exit("xL", from_entry="L", stop=trailStop, limit=entryPx + rr * risk)
    else
        extremeSinceEntry := math.min(extremeSinceEntry, low)
        if trailAfter > 0 and (entryPx - extremeSinceEntry) / risk >= trailAfter
            trailStop := math.min(trailStop, sma20 + trailBuf * atr)
        strategy.exit("xS", from_entry="S", stop=trailStop, limit=entryPx - rr * risk)

if mins >= eodMin
    strategy.close_all(comment="EOD")
    strategy.cancel_all()

// ============================ DISPLAY ============================
plot(ema9,  "EMA 9",   color=color.orange, linewidth=2)
plot(sma20, "SMA 20",  color=color.blue,   linewidth=2)
plot(sma200, "SMA 200", color=color.purple, linewidth=2)
plotshape(canArm and sigL and not aplusL, "Long base",  shape.triangleup,   location.belowbar, color.new(color.green, 0), size=size.tiny)
plotshape(canArm and sigS and not aplusS, "Short base", shape.triangledown, location.abovebar, color.new(color.red, 0),   size=size.tiny)
plotshape(canArm and aplusL, "A+ long base",  shape.triangleup,   location.belowbar, color.new(color.lime, 0),    size=size.small, text="A+")
plotshape(canArm and aplusS, "A+ short base", shape.triangledown, location.abovebar, color.new(color.fuchsia, 0), size=size.small, text="A+")
bgcolor(catalystOK and useCatalyst ? color.new(color.yellow, 92) : na, title="Stock in play")

var table tb = table.new(position.top_right, 2, 9, border_width=1)
if showTable and barstate.islast
    wins = 0
    gp = 0.0
    gl = 0.0
    if strategy.closedtrades > 0
        for k = 0 to strategy.closedtrades - 1
            p = strategy.closedtrades.profit(k)
            if p > 0
                wins += 1
                gp += p
            else
                gl -= p
    n = strategy.closedtrades
    table.cell(tb, 0, 0, "MA Base Breakout v1.0", text_color=color.white, bgcolor=color.navy)
    table.cell(tb, 1, 0, syminfo.ticker, text_color=color.white, bgcolor=color.navy)
    table.cell(tb, 0, 1, "Trades")
    table.cell(tb, 1, 1, str.tostring(n))
    table.cell(tb, 0, 2, "Win %")
    table.cell(tb, 1, 2, n > 0 ? str.tostring(100.0 * wins / n, "#.0") : "-")
    table.cell(tb, 0, 3, "Profit factor")
    table.cell(tb, 1, 3, gl > 0 ? str.tostring(gp / gl, "#.00") : "-")
    table.cell(tb, 0, 4, "Net points")
    table.cell(tb, 1, 4, str.tostring(strategy.netprofit, "#.00"))
    table.cell(tb, 0, 5, "Today gap %")
    table.cell(tb, 1, 5, str.tostring(gapPct, "#.00"))
    table.cell(tb, 0, 6, "Opening vol x")
    table.cell(tb, 1, 6, str.tostring(orvol, "#.00"))
    table.cell(tb, 0, 7, "Rel. strength %")
    table.cell(tb, 1, 7, str.tostring(rs, "#.00"))
    table.cell(tb, 0, 8, "In play today")
    table.cell(tb, 1, 8, catalystOK ? "YES" : "no", text_color=catalystOK ? color.green : color.gray)
