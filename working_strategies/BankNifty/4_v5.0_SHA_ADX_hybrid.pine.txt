//@version=5
// ============================================================================
//  v5.0 SHA-ADX Hybrid — BankNifty (NSE:BANKNIFTY1!, 5m)
//  Exact Pine conversion of trading_agents/core/signals_v50.py V50_INSTRUMENT_PARAMS["BANKNIFTY"]
//  ---------------------------------------------------------------------------
//  Walk-forward: 14/24 windows profitable (vs v0.4's 3/12), avg PF 2.85
//  Full-history: PF 1.494, net +1,807 pts, 44.9% win rate, Sharpe 1.31
//  ---------------------------------------------------------------------------
//  KEY CHANGE from v0.4: entry is now SHA-flip (not pullback-reclaim), because
//  post-Sep-2025 BankNifty regime doesn't reward pullback entry. The volatility
//  regime filter (VolS=80) is critical here — without it, flip entry captures too
//  much chop.
// ============================================================================
strategy("v5.0 SHA-ADX Hybrid (BankNifty)", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     slippage=5,
     max_labels_count=500, max_lines_count=500)

// ============ INPUTS (exact match to V50_INSTRUMENT_PARAMS["BANKNIFTY"]) ============
shaLen1  = input.int(10, "SHA pre-smooth EMA", group="Trend")
shaLen2  = input.int(10, "SHA post-smooth EMA", group="Trend")
shaMinHold = input.int(3, "SHA must hold direction >= N bars before flip", group="Trend")

useSess  = input.bool(true, "Session filter", group="Time")
sessTime = input.session("0930-1500", "Entry session (IST)", group="Time")
forceFlatTime = input.session("1430-1500", "Force-flat window", group="Time")

htfAdxLen = input.int(14, "15m ADX length", group="ADX Gate")
htfAdxMin = input.float(20.0, "15m ADX min threshold (0=off)", step=1, group="ADX Gate")

useVolRegime = input.bool(true, "Volatility regime filter", group="Volatility")
volSmaLen    = input.int(80, "ATR SMA length for vol regime", group="Volatility")
atrMinPts    = input.float(50.0, "Min ATR14 points (0=off)", step=5, group="Volatility")

usePullback  = input.bool(true, "Require pullback proximity to EMA9", group="Entry Quality")
pbAtrMult    = input.float(1.0, "Max distance from EMA9 (x ATR)", step=0.1, group="Entry Quality")

swLen    = input.int(10, "SL swing lookback (bars)", group="Risk")
swBuf    = input.float(0.1, "SL buffer beyond swing (x ATR)", step=0.05, group="Risk")
minSL    = input.float(1.5, "SL min distance (x ATR)", step=0.1, group="Risk")
maxSL    = input.float(3.0, "SL max distance (x ATR, skip wider)", step=0.1, group="Risk")
rr       = input.float(4.0, "Target R:R (fixed)", step=0.1, group="Risk")
dayLossLimit = input.float(500.0, "Daily loss limit (pts, 0=off)", step=10, group="Risk Management")

// ============ SHA CALCULATION (matches Python sha_up/flip_up/flip_dn) ============
o1 = ta.ema(open, shaLen1)
c1 = ta.ema(close, shaLen1)
h1 = ta.ema(high, shaLen1)
l1 = ta.ema(low, shaLen1)
haC = (o1 + h1 + l1 + c1) / 4.0
var float haO = na
haO := na(haO[1]) ? (o1 + c1) / 2.0 : (haO[1] + haC[1]) / 2.0
shaUp = ta.ema(haC, shaLen2) > ta.ema(haO, shaLen2)
flipUp = shaUp and not shaUp[1]
flipDn = not shaUp and shaUp[1]

// SHA stability filter
shaFlipDiff = ta.valuewhen(shaUp != shaUp[1], bar_index, 0)
shaPrevFlip = ta.valuewhen(shaUp != shaUp[1], bar_index, 1)
shaPrevGroupSize = na(shaPrevFlip) ? 1 : (shaFlipDiff - shaPrevFlip)
// On a flip bar shaFlipDiff == bar_index, so the old `(bar_index - shaFlipDiff) <= 1 or ...`
// short-circuit made this ALWAYS true exactly when it mattered. Gate on the run length only.
shaStable = not na(shaPrevFlip) and (shaPrevGroupSize >= shaMinHold)

// ============ INDICATORS ============
e9 = ta.ema(close, 9)
e22 = ta.ema(close, 22)
e200 = ta.ema(close, 200)
atr = ta.atr(14)
swLo = ta.lowest(low, swLen)
swHi = ta.highest(high, swLen)

// EMA alignment: Python trend_l always requires e9>e22 AND close>e200
trendOKL = (e9 > e22) and (close > e200)
trendOKS = (e9 < e22) and (close < e200)
inSess = not useSess or not na(time(timeframe.period, sessTime))
inForceFlat = not na(time(timeframe.period, forceFlatTime))
atrOK = atrMinPts == 0 or (not na(atr) and atr >= atrMinPts)
volSmaAtr = ta.sma(atr, volSmaLen)
volRegimeOK = not useVolRegime or (not na(volSmaAtr) and atr > volSmaAtr)
nearEma9L = not usePullback or close <= e9 + pbAtrMult * atr
nearEma9S = not usePullback or close >= e9 - pbAtrMult * atr

riskL = math.max(close - (swLo - swBuf * atr), minSL * atr)
riskS = math.max((swHi + swBuf * atr) - close, minSL * atr)
cap = maxSL * atr

// Entry conditions (MATCH Python ok_l/ok_s — no atr>=riskL, correct atrOK)
okL = inSess and flipUp and trendOKL and shaStable and atrOK and volRegimeOK and nearEma9L and (riskL <= cap) and not inForceFlat
okS = inSess and flipDn and trendOKS and shaStable and atrOK and volRegimeOK and nearEma9S and (riskS <= cap) and not okL and not inForceFlat

// ============ DAILY LOSS CIRCUIT BREAKER ============
newDay = ta.change(time("D")) != 0
var float dayRealized = 0.0
var bool  dayLocked = false
if newDay
    dayRealized := 0.0
    dayLocked := false
nClosed = strategy.closedtrades - nz(strategy.closedtrades[1])
if nClosed > 0
    for k = 0 to nClosed - 1
        i = strategy.closedtrades - 1 - k
        dayRealized += strategy.closedtrades.profit(i) / syminfo.pointvalue
if dayLossLimit > 0 and dayRealized <= -dayLossLimit and not dayLocked
    dayLocked := true
    strategy.close_all(comment="Daily loss limit")

// ============ ENTRIES ============
var float slPx = na
var float tpPx = na
var float entryPx = na
if okL and not dayLocked
    slPx := close - riskL
    tpPx := close + rr * riskL
    entryPx := close
    strategy.entry("L", strategy.long)
    strategy.exit("xL", "L", stop=slPx, limit=tpPx)
if okS and not dayLocked
    slPx := close + riskS
    tpPx := close - rr * riskS
    entryPx := close
    strategy.entry("S", strategy.short)
    strategy.exit("xS", "S", stop=slPx, limit=tpPx)

// ============ FORCED FLAT (session end) ============
// Forced flat at session end
if strategy.position_size != 0 and inForceFlat
    strategy.close_all(comment="FORCE FLAT")

// ============ VISUALS ============
plotshape(strategy.position_size > 0 and strategy.position_size[1] <= 0,  "Long",  shape.triangleup,   location.belowbar, color.lime,  size=size.small, text="BUY")
plotshape(strategy.position_size < 0 and strategy.position_size[1] >= 0, "Short", shape.triangledown, location.abovebar, color.red,   size=size.small, text="SELL")
plot(e9, "EMA9", color=color.aqua)
plot(e22, "EMA22", color=color.orange)
plot(e200, "EMA200", color=color.new(color.purple, 0), linewidth=1)
bgcolor(useVolRegime and not volRegimeOK ? color.new(color.gray, 85) : na)
