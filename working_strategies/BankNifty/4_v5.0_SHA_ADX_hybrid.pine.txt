//@version=5
// ============================================================================
//  v5.0 SHA-ADX Hybrid — BankNifty (NSE:BANKNIFTY1!, 5m)
//  Walk-forward: 14/24 windows profitable (vs v0.4's 3/12), avg PF 2.85
//  Optimized defaults (param sweep winner): ADX=25, PB=0.5, RR=3.5, vol-regime ON
//  Key difference vs existing BankNifty v0.4: SHA-flip entry instead of pullback
//  Copy to TradingView → paste in Pine Editor → "Add to chart" → enable strategy
// ============================================================================
strategy("v5.0 SHA-ADX Hybrid (BankNifty)", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     slippage=5,
     max_labels_count=500, max_lines_count=500)

shaLen1  = input.int(10, "SHA pre-smooth EMA", group="Trend")
shaLen2  = input.int(10, "SHA post-smooth EMA", group="Trend")
useEmaAl = input.bool(true, "Require EMA9 vs EMA22 alignment", group="Trend")
use200   = input.bool(false, "Require close vs EMA200", group="Trend")
shaMinHold = input.int(3, "SHA must hold direction >= N bars before flip", group="Trend")
useSess  = input.bool(true, "Session filter", group="Time")
sessTime = input.session("0930-1500", "Entry session (IST)", group="Time")
forceFlatTime = input.session("1445-1500", "Force-flat window", group="Time")
htfAdxLen = input.int(14, "15m ADX length", group="ADX Gate")
htfAdxMin = input.float(25.0, "15m ADX min threshold (0=off)", step=1, group="ADX Gate")
useVolRegime = input.bool(true, "Volatility regime filter", group="Volatility")
volSmaLen    = input.int(80, "ATR SMA length for vol regime", group="Volatility")
usePullback  = input.bool(true, "Require pullback proximity to EMA9", group="Entry Quality")
pbAtrMult    = input.float(0.5, "Max distance from EMA9 (x ATR)", step=0.1, group="Entry Quality")
swLen    = input.int(10, "SL swing lookback (bars)", group="Risk")
swBuf    = input.float(0.1, "SL buffer beyond swing (x ATR)", step=0.05, group="Risk")
minSL    = input.float(1.5, "SL min distance (x ATR)", step=0.1, group="Risk")
maxSL    = input.float(3.0, "SL max distance (x ATR, skip wider)", step=0.1, group="Risk")
rr       = input.float(3.5, "Target R:R (fixed)", step=0.1, group="Risk")
dayLossLimit = input.float(0.0, "Daily loss limit (pts, 0=off)", step=10, group="Risk Management")
atrMinPts = input.float(0.0, "ATR floor (pts, 0=off)", step=0.5, group="Risk")

// ============ SHA CALCULATION ============
o1 = ta.ema(open, shaLen1)
c1 = ta.ema(close, shaLen1)
h1 = ta.ema(high, shaLen1)
l1 = ta.ema(low, shaLen1)
haC = (o1 + h1 + l1 + c1) / 4
var float haO = na
haO := na(haO[1]) ? (o1 + c1) / 2 : (haO[1] + haC[1]) / 2
shaUp = ta.ema(haC, shaLen2) > ta.ema(haO, shaLen2)
flipUp = shaUp and not shaUp[1]
flipDn = not shaUp and shaUp[1]
shaFlipBar = ta.valuewhen(shaUp != shaUp[1], bar_index, 0)
shaPrevFlipBar = ta.valuewhen(shaUp != shaUp[1], bar_index, 1)
shaStable = (bar_index - shaFlipBar) <= 1 or (not na(shaPrevFlipBar) and (shaFlipBar - shaPrevFlipBar) >= shaMinHold + 1)

// ============ INDICATORS ============
e9 = ta.ema(close, 9)
e22 = ta.ema(close, 22)
e200 = ta.ema(close, 200)
atr = ta.atr(14)
swLo = ta.lowest(low, swLen)
swHi = ta.highest(high, swLen)
trendL = (not useEmaAl or e9 > e22) and (not use200 or close > e200)
trendS = (not useEmaAl or e9 < e22) and (not use200 or close < e200)
inSess = not useSess or not na(time(timeframe.period, sessTime))
inForceFlat = not na(time(timeframe.period, forceFlatTime))

f_adxPrev() =>
    [_p, _m, _a] = ta.dmi(htfAdxLen, htfAdxLen)
    _a[1]
hAdx = request.security(syminfo.tickerid, "15", f_adxPrev(), lookahead=barmerge.lookahead_on)
adxOK = htfAdxMin == 0 or (not na(hAdx) and hAdx >= htfAdxMin)
volSmaAtr = ta.sma(atr, volSmaLen)
volRegimeOK = not useVolRegime or (not na(volSmaAtr) and atr > volSmaAtr)
nearEma9L = not usePullback or close <= e9 + pbAtrMult * atr
nearEma9S = not usePullback or close >= e9 - pbAtrMult * atr

riskL = math.max(close - (swLo - swBuf * atr), minSL * atr)
riskS = math.max((swHi + swBuf * atr) - close, minSL * atr)
cap = maxSL * atr
atrOK = atrMinPts == 0 or atr >= atrMinPts
flat = strategy.position_size == 0

okL = flat and inSess and flipUp and trendL and shaStable and volRegimeOK and adxOK and atrOK and riskL <= cap and not inForceFlat and nearEma9L
okS = flat and inSess and flipDn and trendS and shaStable and volRegimeOK and adxOK and atrOK and riskS <= cap and not okL and not inForceFlat and nearEma9S

// ============ FORCED FLAT (session end) ============
if flat == false and inForceFlat
    strategy.close_all(comment="EOD")

// ============ ENTRIES ============
var float slPx = na
var float tpPx = na
var float entryPx = na
if okL
    slPx := close - riskL
    tpPx := close + rr * riskL
    entryPx := close
    strategy.entry("L", strategy.long)
    strategy.exit("xL", "L", stop=slPx, limit=tpPx)
if okS
    slPx := close + riskS
    tpPx := close - rr * riskS
    entryPx := close
    strategy.entry("S", strategy.short)
    strategy.exit("xS", "S", stop=slPx, limit=tpPx)

// ============ VISUALS ============
plotshape(strategy.position_size > 0 and strategy.position_size[1] <= 0,  "Long",  shape.triangleup,   location.belowbar, color.lime, size=size.small, text="BUY")
plotshape(strategy.position_size < 0 and strategy.position_size[1] >= 0, "Short", shape.triangledown, location.abovebar, color.red,   size=size.small, text="SELL")
plot(strategy.position_size != 0 ? entryPx : na, "Entry", color=color.blue, style=plot.style_linebr, linewidth=1)
plot(strategy.position_size != 0 ? slPx    : na, "Stop",  color=color.red,  style=plot.style_linebr, linewidth=1)
plot(strategy.position_size != 0 ? tpPx    : na, "TP",    color=color.green, style=plot.style_linebr, linewidth=1)
plot(e9, "EMA9", color=color.aqua)
plot(e22, "EMA22", color=color.orange)
