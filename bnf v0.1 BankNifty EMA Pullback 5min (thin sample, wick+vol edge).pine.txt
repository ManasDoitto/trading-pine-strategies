//@version=5
// ============================================================================
//  BankNifty EMA Pullback (5m) v0.1   --  NSE:BANKNIFTY1!  (pure 5-minute)
//  Built to the "BankNifty Intraday Pullback" spec. Structural template =
//  crude v2.0/v1.0 pullback-continuation engine. Pure 5m: NO request.security
//  (no repaint), NO lookahead.
//
//  BankNifty adaptations vs crude: EMA21 (not 22) | VWAP confirm | rejection-
//  wick pullback-quality filter | volume>=mult x SMA20 | ATR points band
//  (50-400) | gap-direction lock (first 60m) | Thursday-expiry skip | EOD flat
//  by 15:15 | skip first 15m at open | entry window 09:30-15:00.
//
//  SIZING NOTE: on NSE:BANKNIFTY1! futures, qty=1 already = ONE lot (point value
//  encodes the lot). The spec's qty=15 = 15 LOTS -> commission ballooned to
//  ~Rs.3.4M and dwarfed all P&L. This file uses qty=1 + 0.02% (~Rs.700/RT, real).
//
//  ---- HONEST BACKTEST (NSE:BANKNIFTY1! 5m, ~10 months, qty=1) ----
//  Filter ladder (gross PF), showing WHICH filters carry the edge:
//    bare pullback ................... PF 0.66  219 trades  (loser)
//    + VWAP .......................... PF 0.70  191 trades
//    + volume>=1.2x .................. PF 0.77   43 trades   (either filter
//    + wick-quality (alone) .......... PF 0.67  126 trades    alone = loser)
//    + volume AND wick together ...... PF 1.02   ~18 trades  <- edge appears
//  So the edge is the INTERSECTION of "rejection wick" + "above-avg volume".
//  Neither works alone; together they lift PF over 1 but starve the sample.
//
//  Adopted config (volMlt 1.0, wick+VWAP+vol on, sep/gap off), R:R ladder:
//    2.0R -> PF 1.22 | 50% win | avg Rs.387 | DD 2.5% | 26 trades
//    2.5R -> PF 1.43 | 50% win | avg Rs.756 | DD 2.5% | 26 trades  (default)
//    3.0R -> PF 1.59 | 50% win | avg Rs.1026| DD 2.5% | Sharpe 0.53
//  WARNING - the R:R knob is largely INERT above ~2R: winners mostly exit at
//  the 15:15 EOD-flat, NOT at the target (verified in the trade list - winner
//  exits are MARKET/EOD, largest_win frozen across 3R/4R/5R). Higher R just
//  lets winners ride to the forced close. Do NOT read "3R" as a 3R hit-rate.
//
//  VERDICT: genuinely net-positive with a tiny drawdown, and it clears most of
//  the spec targets (win 50%, DD<2.5%, avg>Rs.500 at >=2.5R, R:R>=2). BUT it
//  rests on only 26 trades / 13 winners in 10 months (~2.6 trades/mo) - too thin
//  to trust as a standalone edge, and well below crude v1.0 (PF 1.89, 203 trades).
//  5m-only: 3m PF 0.87 and 15m PF 0.80 both lose. FORWARD-TEST before sizing.
// ============================================================================
strategy("BankNifty EMA Pullback (5m) v0.1", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     slippage=5,
     max_lines_count=500, max_labels_count=500)

// ============ INPUTS ============
emaFastLen  = input.int(9,   "EMA Fast",  group="EMA")
emaMedLen   = input.int(21,  "EMA Med",   group="EMA")
emaSlowLen  = input.int(200, "EMA Slow",  group="EMA")
use200      = input.bool(true, "Trend filter: EMA200",      group="EMA")
useVwap     = input.bool(true, "Confirm with VWAP",          group="EMA")
sepMlt      = input.float(0.0,  "Min EMA9-EMA21 sep (x ATR14, 0=off)", step=0.05, group="EMA")

useSession  = input.bool(true, "Session filter",             group="Session")
sessTime    = input.session("0930-1500", "Entry window (IST)", group="Session")
skipOpenMin = input.int(15, "Skip first N minutes",          group="Session")
flatTime    = input.session("1515-1530", "Force-flat window", group="Session")

atrMinPts   = input.float(50.0,  "Min ATR14 points (0=off)",  step=5, group="Volatility")
atrMaxPts   = input.float(400.0, "Max ATR14 points (0=off)",  step=10, group="Volatility")
atrBuffMlt   = input.float(0.15, "SL buffer beyond pullback (x ATR14)", step=0.05, group="Risk")
stopFloorATR = input.float(0.5,  "Min SL floor (x ATR14)",    step=0.05, group="Risk")
maxStopATR  = input.float(2.0,   "Max stop (x ATR14, 0=off)", step=0.25, group="Risk")
rMultiple   = input.float(2.5,   "R:R target",                step=0.1, group="Risk")

rsiLen      = input.int(3,   "RSI Length",                    group="Filters")
rsiMaxLong  = input.int(80,  "No long if RSI(3) above",       group="Filters")
rsiMinShort = input.int(20,  "No short if RSI(3) below",      group="Filters")
useVol      = input.bool(true, "Entry bar volume >= mult x SMA20", group="Filters")
volMlt      = input.float(1.0, "  volume mult",               step=0.1, group="Filters")
useWick     = input.bool(true, "Require rejection wick >= 50% in pullback", group="Filters")
useGap      = input.bool(false, "Gap-direction lock (first 60m)", group="Filters")
gapMlt      = input.float(1.5, "  gap > x prior-day ATR",     step=0.1, group="Filters")
skipExpiry  = input.bool(false, "Skip Thursday (weekly expiry)", group="Filters")
pbLookback  = input.int(10,  "Pullback within N bars of above-both", group="Engine")
reclaimWin  = input.int(8,   "Max bars: touch -> reclaim -> entry", group="Engine")
coolBars    = input.int(3,   "Cooldown bars after exit",      group="Engine")

allowCoil   = input.bool(true, "COIL: trade EMA-convergence break", group="Coil")
clusterMlt  = input.float(1.0, "  coil: 3-EMA span <= x ATR14", step=0.1, group="Coil")
clusterBuf  = input.float(0.25, "  coil SL buf beyond band (x ATR14)", step=0.05, group="Coil")

useTrail    = input.bool(false, "EXP: trail SL after +1R (BE then 0.5xATR)", group="Experimental")
beAfterR    = input.float(0.0,  "EXP: SL to breakeven after +R (0=off)", step=0.1, group="Experimental")
tp1R        = input.float(0.0,  "EXP: partial TP at +R (0=off)", step=0.1, group="Experimental")
tp1Pct      = input.int(50,     "EXP: partial TP % of position", group="Experimental")
skipMon     = input.bool(false, "EXP: skip Monday",           group="Experimental")
skipFri     = input.bool(false, "EXP: skip Friday",           group="Experimental")
showBG      = input.bool(true,  "Trend background",           group="Visuals")
showMarks   = input.bool(true,  "Entry/exit marks",           group="Visuals")
showZones   = input.bool(false, "Pullback zone boxes",        group="Visuals")

// ============ INDICATORS ============
ema9   = ta.ema(close, emaFastLen)
ema21  = ta.ema(close, emaMedLen)
ema200 = ta.ema(close, emaSlowLen)
rsi3   = ta.rsi(close, rsiLen)
atr14  = ta.atr(14)
vwap   = ta.vwap(hlc3)
volSma = ta.sma(volume, 20)

emaHi3 = math.max(ema9, math.max(ema21, ema200))
emaLo3 = math.min(ema9, math.min(ema21, ema200))
coilOn = allowCoil and (emaHi3 - emaLo3) <= clusterMlt * atr14

// ---- session / time gating ----
inSess   = not useSession or not na(time(timeframe.period, sessTime))
inFlat   = not na(time(timeframe.period, flatTime))
newDay   = ta.change(time("D")) != 0
var int  dayOpenTime = na
var float dayOpenPx  = na
if newDay
    dayOpenTime := time
    dayOpenPx   := open
minsSinceOpen = na(dayOpenTime) ? na : (time - dayOpenTime) / 60000.0
afterOpenSkip = na(minsSinceOpen) or minsSinceOpen >= skipOpenMin
inSession = inSess and afterOpenSkip and not inFlat

// ---- prior-day close & prior-day ATR (for gap filter, no security) ----
var float dHi = na
var float dLo = na
var float prevDayClose = na
var float prevDayATR   = na
if newDay
    if not na(dHi) and not na(dLo)
        _tr = dHi - dLo
        prevDayATR := na(prevDayATR) ? _tr : (prevDayATR * 13 + _tr) / 14.0
    prevDayClose := close[1]
    dHi := high
    dLo := low
else
    dHi := na(dHi) ? high : math.max(dHi, high)
    dLo := na(dLo) ? low  : math.min(dLo, low)

gapUp = not na(prevDayClose) and not na(prevDayATR) and (dayOpenPx - prevDayClose) > gapMlt * prevDayATR
gapDn = not na(prevDayClose) and not na(prevDayATR) and (prevDayClose - dayOpenPx) > gapMlt * prevDayATR
inGapWin = not na(minsSinceOpen) and minsSinceOpen < 60
gapAllowL = not useGap or not inGapWin or not gapDn
gapAllowS = not useGap or not inGapWin or not gapUp

// ---- misc filters ----
emaSepOK  = sepMlt == 0 or math.abs(ema9 - ema21) >= sepMlt * atr14 or coilOn
atrOK     = (atrMinPts == 0 or atr14 >= atrMinPts) and (atrMaxPts == 0 or atr14 <= atrMaxPts)
volOK     = not useVol or volume >= volMlt * volSma
isThu     = dayofweek == dayofweek.thursday
expiryOK  = not skipExpiry or not isThu
dowOK     = not (skipMon and dayofweek == dayofweek.monday) and not (skipFri and dayofweek == dayofweek.friday) and expiryOK
vwapOKL   = not useVwap or close > vwap
vwapOKS   = not useVwap or close < vwap

// ============ REGIME ============
regimeUp = (ema9 > ema21 or coilOn) and (not use200 or close > ema200)
regimeDn = (ema9 < ema21 or coilOn) and (not use200 or close < ema200)
aboveBoth = coilOn ? close > emaHi3 : (close > ema9 and close > ema21)
belowBoth = coilOn ? close < emaLo3 : (close < ema9 and close < ema21)

rng      = math.max(high - low, syminfo.mintick)
lowerWick = math.min(open, close) - low
upperWick = high - math.max(open, close)
bigLowerWick = lowerWick >= 0.5 * rng
bigUpperWick = upperWick >= 0.5 * rng

// ============ PULLBACK TRACKER ============
var bool  pbActL   = false
var float pbLowL   = na
var int   pbBarL   = na
var int   lastAbvL = na
var bool  pbWickL  = false
var bool  pbCloseOKL = false
var bool  pbActS   = false
var float pbHiS    = na
var int   pbBarS   = na
var int   lastBlwS = na
var bool  pbWickS  = false
var bool  pbCloseOKS = false

if not regimeUp
    pbActL := false
    pbLowL := na
    pbBarL := na
    pbWickL := false
    pbCloseOKL := false
else
    if aboveBoth
        lastAbvL := bar_index
    if low <= ema9 and not na(lastAbvL) and (bar_index - lastAbvL) <= pbLookback
        fresh = not pbActL or (not na(pbBarL) and lastAbvL >= pbBarL)
        pbLowL := fresh ? low : math.min(pbLowL, low)
        pbWickL := fresh ? bigLowerWick : (pbWickL or bigLowerWick)
        pbCloseOKL := close > ema21
        pbActL := true
        pbBarL := bar_index

if not regimeDn
    pbActS := false
    pbHiS  := na
    pbBarS := na
    pbWickS := false
    pbCloseOKS := false
else
    if belowBoth
        lastBlwS := bar_index
    if high >= ema9 and not na(lastBlwS) and (bar_index - lastBlwS) <= pbLookback
        fresh = not pbActS or (not na(pbBarS) and lastBlwS >= pbBarS)
        pbHiS := fresh ? high : math.max(pbHiS, high)
        pbWickS := fresh ? bigUpperWick : (pbWickS or bigUpperWick)
        pbCloseOKS := close < ema21
        pbActS := true
        pbBarS := bar_index

// ============ COOLDOWN ============
var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars

// ============ TRIGGER SETUP ============
var bool  waitL   = false
var int   waitBarL = na
var float trigL   = na
var float slL     = na
var float tpL     = na
var float tp1LvL  = na
var bool  waitS   = false
var int   waitBarS = na
var float trigS   = na
var float slS     = na
var float tpS     = na
var float tp1LvS  = na

wickOKL = not useWick or pbWickL
wickOKS = not useWick or pbWickS

gateL = regimeUp and aboveBoth and emaSepOK and inSession and atrOK and dowOK and volOK and vwapOKL and gapAllowL and wickOKL and pbCloseOKL and rsi3 < rsiMaxLong
gateS = regimeDn and belowBoth and emaSepOK and inSession and atrOK and dowOK and volOK and vwapOKS and gapAllowS and wickOKS and pbCloseOKS and rsi3 > rsiMinShort

if strategy.opentrades == 0 and coolOK and not waitL and pbActL and gateL and not na(pbBarL) and (bar_index - pbBarL) <= reclaimWin
    _slTight = coilOn ? (emaLo3 - clusterBuf * atr14) : (pbLowL - atrBuffMlt * atr14)
    _sl      = stopFloorATR > 0 ? math.min(_slTight, high - stopFloorATR * atr14) : _slTight
    _riskSL  = high - _sl
    if _riskSL > 0 and (maxStopATR == 0 or _riskSL <= maxStopATR * atr14)
        waitL := true
        waitBarL := bar_index
        trigL := high
        slL := _sl
        tpL := high + _riskSL * rMultiple
        tp1LvL := tp1R > 0 ? high + _riskSL * tp1R : na
        pbActL := false

if strategy.opentrades == 0 and coolOK and not waitS and pbActS and gateS and not na(pbBarS) and (bar_index - pbBarS) <= reclaimWin
    _slTight = coilOn ? (emaHi3 + clusterBuf * atr14) : (pbHiS + atrBuffMlt * atr14)
    _sl      = stopFloorATR > 0 ? math.max(_slTight, low + stopFloorATR * atr14) : _slTight
    _riskSL  = _sl - low
    if _riskSL > 0 and (maxStopATR == 0 or _riskSL <= maxStopATR * atr14)
        waitS := true
        waitBarS := bar_index
        trigS := low
        slS := _sl
        tpS := low - _riskSL * rMultiple
        tp1LvS := tp1R > 0 ? low - _riskSL * tp1R : na
        pbActS := false

if waitL and (bar_index - waitBarL > reclaimWin or not regimeUp)
    waitL := false
if waitS and (bar_index - waitBarS > reclaimWin or not regimeDn)
    waitS := false

// ============ ENTRIES ============
if waitL and not inFlat
    strategy.entry("Long", strategy.long, stop=trigL)
else
    strategy.cancel("Long")
if waitS and not inFlat
    strategy.entry("Short", strategy.short, stop=trigS)
else
    strategy.cancel("Short")

longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
var float riskL0 = na
if longEntryBar
    waitL := false
    riskL0 := strategy.position_avg_price - slL
if shortEntryBar
    waitS := false
    riskL0 := slS - strategy.position_avg_price

// ============ EXITS ============
if strategy.position_size > 0
    waitL := false
    _sl = slL
    if beAfterR > 0 and not na(riskL0) and high >= strategy.position_avg_price + beAfterR * riskL0
        _sl := math.max(_sl, strategy.position_avg_price)
    if useTrail and not na(riskL0) and high >= strategy.position_avg_price + riskL0
        _sl := math.max(_sl, strategy.position_avg_price, high - 0.5 * atr14)
    slL := _sl
    if tp1R > 0 and not na(tp1LvL)
        strategy.exit("TP1 L", from_entry="Long", qty_percent=tp1Pct, stop=slL, limit=tp1LvL)
    strategy.exit("Exit L", from_entry="Long", stop=slL, limit=tpL)
    if inFlat
        strategy.close("Long", comment="EOD")
if strategy.position_size < 0
    waitS := false
    _sl = slS
    if beAfterR > 0 and not na(riskL0) and low <= strategy.position_avg_price - beAfterR * riskL0
        _sl := math.min(_sl, strategy.position_avg_price)
    if useTrail and not na(riskL0) and low <= strategy.position_avg_price - riskL0
        _sl := math.min(_sl, strategy.position_avg_price, low + 0.5 * atr14)
    slS := _sl
    if tp1R > 0 and not na(tp1LvS)
        strategy.exit("TP1 S", from_entry="Short", qty_percent=tp1Pct, stop=slS, limit=tp1LvS)
    strategy.exit("Exit S", from_entry="Short", stop=slS, limit=tpS)
    if inFlat
        strategy.close("Short", comment="EOD")

// ============ VISUALS ============
bgcolor(showBG and regimeUp and aboveBoth ? color.new(color.green, 92) : na)
bgcolor(showBG and regimeDn and belowBoth ? color.new(color.red,   92) : na)
bgcolor(showBG and useSession and not inSession ? color.new(color.gray, 92) : na)
plot(ema9,   "EMA9",   color=color.aqua,   linewidth=1)
plot(ema21,  "EMA21",  color=color.orange, linewidth=1)
plot(ema200, "EMA200", color=color.new(color.purple, 0), linewidth=2)
plot(useVwap ? vwap : na, "VWAP", color=color.new(color.white, 0), style=plot.style_circles, linewidth=1)
plotshape(showMarks and longEntryBar,  "Long",  shape.triangleup,   location.belowbar, color.new(color.teal, 0), size=size.small, text="LONG",  textcolor=color.white)
plotshape(showMarks and shortEntryBar, "Short", shape.triangledown, location.abovebar, color.new(color.red, 0),  size=size.small, text="SHORT", textcolor=color.white)
plot(strategy.position_size > 0 ? tpL : strategy.position_size < 0 ? tpS : na, "TP", color=color.new(color.green, 0), style=plot.style_linebr, linewidth=1)
plot(strategy.position_size > 0 ? slL : strategy.position_size < 0 ? slS : na, "SL", color=color.new(color.red, 0), style=plot.style_linebr, linewidth=1)
