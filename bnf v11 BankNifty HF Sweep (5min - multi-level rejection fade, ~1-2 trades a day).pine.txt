//@version=5
// ============================================================================
//  BankNifty HF Sweep v11  --  NSE:BANKNIFTY1!  (5-minute)  --  POINTS / SCALP
//  ---------------------------------------------------------------------------
//  Higher-frequency cousin of v10.5. Same idea (fade a liquidity sweep back
//  inside, only with the multi-day trend) but instead of just PDH/PDL it
//  watches a rolling BASKET of intraday liquidity levels, so a setup prints
//  most days, ~1-2 / day on average (not forced daily).
//
//  LEVEL BASKET (toggle each)
//    useSwing : confirmed 5m swing highs / lows (ta.pivot, length swPivLen)
//    useOR    : opening-range high / low (orSess window)
//    usePD    : prior-day high / low
//    useRound : round numbers every roundStep pts near price
//  Levels are de-duplicated (minLvlGapPts), expire after lvlExpiryBars, drop
//  once used or once price closes decisively through them.
//
//  ENTRY  (one bar, at the nearest valid level to price)
//    short : bar wicks >= a high-side level, closes back below it by a real
//            body, wick beyond <= maxSweepMlt x avgRange, close < open,
//            close < low[1] (unless relaxTrig), AND the daily-trend gate allows
//            shorts (5m close <= N-day SMA of daily close + regimeSlack band).
//    long  : mirror on a low-side level.
//  STOP  : min(sweep wick +/- bufMlt x avgRange, entry +/- stopCapPts).  <=120.
//  TARGET: tgtMode "R" -> rFixed x risk ; "nextLevel" -> nearest opposing
//          basket level past minRR, else rFixed x risk.
//  One fade per level. Flat 15:15. On-chart SCALP SCOREBOARD table (points).
//
//  DAILY-DIRECTION FILTER (dayTrendFilt, on by default)
//    Only fade WITH the multi-day trend: shorts need the regimeLen-day SMA of
//    daily close to be falling over the last dayTrendLb days; longs need it
//    rising. Without it the model bleeds in uptrends; with it, 3 of 4 windows
//    are net-positive.
//
//  WALK-FORWARD  (5m replay, tuned default: maxSweepMlt 0.5, regimeSlack -1.0,
//  rFixed 6, dayTrendFilt on. "Net@20" = points minus 20/round-trip.)
//
//    Window (approx)   Market      Tr  Tr/day Win%  Gross  PF    Net@20  Net/mo
//    Apr-Oct 2025      strong up   43  0.33   48.8%  +957  1.57   +97    +16
//    Jun-Dec 2025      up          41  0.32   36.6%  +191  1.11   -629   -103
//    Feb-Jun 2026      down        52  0.40   26.9% +1863  1.51   +823   +132
//    Feb-Sep 2026      down        51  0.37   39.2% +2556  1.78  +1536   +235
//    --------------------------------------------------------------------------
//    HONEST VERDICT
//    - Frequency: ~0.3-0.7 trades / day (7-15 / month), fast ~15-bar holds.
//      Genuine 1-2/day needs looser filters, which every test made LESS
//      profitable - at BankNifty 5m a *profitable* level-rejection fade tops
//      out near ~0.7 trades/day.
//    - Net-positive in 3 of 4 windows; still loses ~-100 pts/mo in a grinding
//      uptrend. More trades x fixed 20-pt option friction eats the thin
//      non-downtrend edge.
//    - It does NOT beat v10.5 as a standalone (v10.5: every window +59..+365
//      net/mo, worst +59). v11's role: a higher-frequency COMPANION to run on
//      top of v10.5 only when the daily trend is down / flat / high-vol, where
//      it adds ~+130..+235 net pts/month.
//  v10.3/10.4/10.5 untouched in their own files.
// ============================================================================
strategy("BankNifty HF Sweep v11", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.0,
     slippage=0, max_lines_count=500, max_labels_count=500, calc_on_every_tick=false)

// ============ INPUTS ============ (// N)
useSwing    = input.bool(true,  "Levels: 5m swing highs/lows", group="Levels") // 0
swPivLen    = input.int(3, "  swing pivot length", minval=1, group="Levels") // 1
useOR       = input.bool(true,  "Levels: opening range H/L", group="Levels") // 2
orSess      = input.session("0930-1000", "  opening-range window", group="Levels") // 3
usePD       = input.bool(true,  "Levels: prior-day H/L", group="Levels") // 4
useRound    = input.bool(false, "Levels: round numbers", group="Levels") // 5
roundStep   = input.float(100.0, "  round step (pts)", step=25, group="Levels") // 6
lvlExpiryBars = input.int(48, "Level expiry (bars)", group="Levels") // 7
minLvlGapPts  = input.float(30.0, "Min gap between levels (pts)", step=5, group="Levels") // 8
maxLevels   = input.int(24, "Max active levels", group="Levels") // 9
maxSweepMlt = input.float(0.5, "Max wick beyond level (x avg range)", step=0.1, group="Sweep") // 10
dispMlt     = input.float(0.0, "Reversal bar range >= x avg range (0=off)", step=0.1, group="Sweep") // 11
relaxTrig   = input.bool(false, "Drop momentum-bar confirmation", group="Sweep") // 12
dirFilter   = input.string("dtrend", "Regime filter", options=["off","dtrend"], group="Direction") // 13
regimeLen   = input.int(8, "dtrend: SMA of daily close (days)", group="Direction") // 14
regimeSlack = input.float(-1.0, "dtrend: band around SMA (x avg range; -ve stricter)", step=0.25, group="Direction") // 15
stopCapPts  = input.float(120.0, "STOP hard cap (points, 0=off)", step=5, group="Risk") // 16
bufMlt      = input.float(0.50, "Stop buffer beyond wick (x avg range)", step=0.05, group="Risk") // 17
tgtMode     = input.string("R", "TARGET mode", options=["R","nextLevel"], group="Risk") // 18
rFixed      = input.float(6.0, "Reward:Risk (R / fallback)", step=0.1, group="Risk") // 19
minRR       = input.float(1.5, "Min R:R to accept a level target", step=0.1, group="Risk") // 20
sessEntry   = input.session("0935-1500", "Entry window (IST)", group="Time") // 21
flatTime    = input.session("1515-1530", "Force-flat window", group="Time") // 22
coolBars    = input.int(1, "Cooldown bars after exit", group="Time") // 23
maxPerDay   = input.int(4, "Max entries per day", group="Time") // 24
maxHoldBars = input.int(0, "Hard time-exit after N bars (0=off)", group="Time") // 25
skipExpiry  = input.bool(false, "Skip Thursday", group="Time") // 26
avgRngLen   = input.int(20, "Avg candle-range length", group="Core") // 27
costPtsRound= input.float(20.0, "Option cost pts / round-trip (report)", step=1, group="Core") // 28
showViz     = input.bool(true, "Visuals", group="Visuals") // 29
dayTrendFilt= input.bool(true, "Daily-direction filter (short only if daily trend down)", group="Direction") // 30
dayTrendLb  = input.int(3, "  daily-SMA slope lookback (days)", minval=1, group="Direction") // 31

// ============ CORE ============
avgRng = ta.sma(high - low, avgRngLen)
[pdh, pdl, pdc] = request.security(syminfo.tickerid, "1D", [high[1], low[1], close[1]], lookahead=barmerge.lookahead_off)
smaD = request.security(syminfo.tickerid, "1D", ta.sma(close[1], regimeLen), lookahead=barmerge.lookahead_off)
dSlope = request.security(syminfo.tickerid, "1D", ta.sma(close[1], regimeLen) - ta.sma(close[1], regimeLen)[dayTrendLb], lookahead=barmerge.lookahead_off)
dayShortOK = not dayTrendFilt or (not na(dSlope) and dSlope <= 0)
dayLongOK  = not dayTrendFilt or (not na(dSlope) and dSlope >= 0)

inEntryWin = not na(time(timeframe.period, sessEntry))
inFlat     = not na(time(timeframe.period, flatTime))
inOR       = not na(time(timeframe.period, orSess))
isThu      = dayofweek == dayofweek.thursday
dowOK      = not (skipExpiry and isThu)
newDay = ta.change(time("D")) != 0
var int nDay = 0
if newDay
    nDay := 0

regBand = regimeSlack * avgRng
dtrendS = dirFilter == "off" or (not na(smaD) and close <= smaD + regBand)
dtrendL = dirFilter == "off" or (not na(smaD) and close >= smaD - regBand)

// ---- opening range ----
var float orH = na
var float orL = na
var bool  orDone = false
if newDay
    orH := na
    orL := na
    orDone := false
if inOR
    orH := na(orH) ? high : math.max(orH, high)
    orL := na(orL) ? low  : math.min(orL, low)

// ============ LEVEL BASKET ============
var lvlP    = array.new_float(0)
var lvlHi   = array.new_bool(0)
var lvlBorn = array.new_int(0)
var lvlUsed = array.new_bool(0)

addLevel(float p, bool isHi) =>
    if not na(p) and p > 0
        bool dup = false
        int n = array.size(lvlP)
        if n > 0
            for i = 0 to n - 1
                if array.get(lvlHi, i) == isHi and math.abs(array.get(lvlP, i) - p) < minLvlGapPts
                    dup := true
        if not dup
            array.push(lvlP, p)
            array.push(lvlHi, isHi)
            array.push(lvlBorn, bar_index)
            array.push(lvlUsed, false)
            if array.size(lvlP) > maxLevels
                array.shift(lvlP)
                array.shift(lvlHi)
                array.shift(lvlBorn)
                array.shift(lvlUsed)

// ---- cleanup: drop used / expired / broken ----
if array.size(lvlP) > 0
    for i = array.size(lvlP) - 1 to 0
        L      = array.get(lvlP, i)
        isHi   = array.get(lvlHi, i)
        used   = array.get(lvlUsed, i)
        born   = array.get(lvlBorn, i)
        broken = isHi ? (close > L + bufMlt * avgRng) : (close < L - bufMlt * avgRng)
        if used or broken or (bar_index - born) > lvlExpiryBars
            array.remove(lvlP, i)
            array.remove(lvlHi, i)
            array.remove(lvlBorn, i)
            array.remove(lvlUsed, i)

// ---- add new levels ----
ph = ta.pivothigh(high, swPivLen, swPivLen)
pl = ta.pivotlow(low, swPivLen, swPivLen)
if useSwing and not na(ph)
    addLevel(ph, true)
if useSwing and not na(pl)
    addLevel(pl, false)
orJustFroze = (not inOR) and inOR[1] and not orDone
if orJustFroze
    orDone := true
    if useOR
        addLevel(orH, true)
        addLevel(orL, false)
if newDay and usePD
    addLevel(pdh, true)
    addLevel(pdl, false)
if newDay and useRound and roundStep > 0
    base = math.floor(close / roundStep) * roundStep
    addLevel(base, false)
    addLevel(base + roundStep, true)
    addLevel(base + 2 * roundStep, true)
    addLevel(base - roundStep, false)

// ============ SCAN FOR TRIGGER ============
momOK_S = relaxTrig or close < low[1]
momOK_L = relaxTrig or close > high[1]
dispOK  = dispMlt == 0 or (high - low) >= dispMlt * avgRng

var float trigLvl  = na
var int   trigSide = 0
var float trigWick = na
trigLvl  := na
trigSide := 0
trigWick := na
float bestDist = 1e12
if array.size(lvlP) > 0
    for i = 0 to array.size(lvlP) - 1
        L    = array.get(lvlP, i)
        isHi = array.get(lvlHi, i)
        used = array.get(lvlUsed, i)
        sShort = not used and isHi and high >= L and close < L and (high - L) <= maxSweepMlt * avgRng and close < open and momOK_S and dtrendS and dayShortOK and dispOK
        sLong  = not used and (not isHi) and low <= L and close > L and (L - low) <= maxSweepMlt * avgRng and close > open and momOK_L and dtrendL and dayLongOK and dispOK
        d = math.abs(L - close)
        if sShort and d < bestDist
            bestDist := d
            trigLvl  := L
            trigSide := -1
            trigWick := high
        if sLong and d < bestDist
            bestDist := d
            trigLvl  := L
            trigSide := 1
            trigWick := low

// ============ RISK ============
var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars
baseOK = strategy.position_size == 0 and coolOK and inEntryWin and not inFlat and dowOK and nDay < maxPerDay

float slRaw = na
if trigSide == -1
    slRaw := trigWick + bufMlt * avgRng
if trigSide == 1
    slRaw := trigWick - bufMlt * avgRng
float sl = na
if trigSide == -1
    sl := (stopCapPts > 0) ? math.min(slRaw, close + stopCapPts) : slRaw
if trigSide == 1
    sl := (stopCapPts > 0) ? math.max(slRaw, close - stopCapPts) : slRaw
rk = trigSide == -1 ? (sl - close) : trigSide == 1 ? (close - sl) : na

nearestOpp(bool wantHigh, float refBeyond, bool above) =>
    float best = na
    if array.size(lvlP) > 0
        for i = 0 to array.size(lvlP) - 1
            L = array.get(lvlP, i)
            isHi = array.get(lvlHi, i)
            okSide = isHi == wantHigh
            okPos = above ? (L >= refBeyond) : (L <= refBeyond)
            if okSide and okPos and not array.get(lvlUsed, i)
                best := na(best) ? L : (above ? math.min(best, L) : math.max(best, L))
    best

float tgt = na
if trigSide == -1 and not na(rk) and rk > 0
    if tgtMode == "nextLevel"
        lv = nearestOpp(false, close - minRR * rk, false)
        tgt := not na(lv) ? lv : close - rFixed * rk
    else
        tgt := close - rFixed * rk
if trigSide == 1 and not na(rk) and rk > 0
    if tgtMode == "nextLevel"
        lv = nearestOpp(true, close + minRR * rk, true)
        tgt := not na(lv) ? lv : close + rFixed * rk
    else
        tgt := close + rFixed * rk

okShort = trigSide == -1 and not na(rk) and rk > 0 and not na(tgt) and tgt < close
okLong  = trigSide == 1  and not na(rk) and rk > 0 and not na(tgt) and tgt > close

// ============ ENTRIES ============
var float slLv = na
var float tpLv = na
var float slSv = na
var float tpSv = na
var int   entryBarIdx = na
var float maxSLused = 0.0

markUsed(float L, bool isHi) =>
    if array.size(lvlP) > 0
        for i = 0 to array.size(lvlP) - 1
            if array.get(lvlHi, i) == isHi and math.abs(array.get(lvlP, i) - L) < 1
                array.set(lvlUsed, i, true)

if baseOK and okShort
    strategy.entry("Short", strategy.short)
    slSv := sl
    tpSv := tgt
    nDay := nDay + 1
    markUsed(trigLvl, true)
if baseOK and okLong
    strategy.entry("Long", strategy.long)
    slLv := sl
    tpLv := tgt
    nDay := nDay + 1
    markUsed(trigLvl, false)

longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
if longEntryBar
    entryBarIdx := bar_index
    maxSLused := math.max(maxSLused, strategy.position_avg_price - slLv)
if shortEntryBar
    entryBarIdx := bar_index
    maxSLused := math.max(maxSLused, slSv - strategy.position_avg_price)

// ============ EXITS ============
barsHeld = not na(entryBarIdx) ? bar_index - entryBarIdx : 0
if strategy.position_size > 0
    strategy.exit("Exit L", from_entry="Long", stop=slLv, limit=tpLv)
    if inFlat or (maxHoldBars > 0 and barsHeld >= maxHoldBars)
        strategy.close("Long", comment="TimeOut")
if strategy.position_size < 0
    strategy.exit("Exit S", from_entry="Short", stop=slSv, limit=tpSv)
    if inFlat or (maxHoldBars > 0 and barsHeld >= maxHoldBars)
        strategy.close("Short", comment="TimeOut")

// ============ SCALP SCOREBOARD (points) ============
var float sumPts   = 0.0
var float sumWinP  = 0.0
var float sumLossP = 0.0
var int   nWin     = 0
var int   nLoss    = 0
var int   nTr      = 0
var int   sumBars  = 0
var float eqPts    = 0.0
var float eqPeak   = 0.0
var float maxDDpts = 0.0
var float bestT    = 0.0
var float worstT   = 0.0
var int   firstT   = 0

if firstT == 0 and bar_index > 0
    firstT := time

newlyClosed = strategy.closedtrades - strategy.closedtrades[1]
if newlyClosed > 0
    for k = 0 to newlyClosed - 1
        idx = strategy.closedtrades - 1 - k
        dir = strategy.closedtrades.size(idx) > 0 ? 1.0 : -1.0
        pts = (strategy.closedtrades.exit_price(idx) - strategy.closedtrades.entry_price(idx)) * dir
        bars = strategy.closedtrades.exit_bar_index(idx) - strategy.closedtrades.entry_bar_index(idx)
        sumPts   += pts
        nTr      += 1
        sumBars  += bars
        if pts >= 0
            sumWinP += pts
            nWin    += 1
        else
            sumLossP += pts
            nLoss    += 1
        bestT  := math.max(bestT, pts)
        worstT := math.min(worstT, pts)
        eqPts  += pts
        eqPeak := math.max(eqPeak, eqPts)
        maxDDpts := math.max(maxDDpts, eqPeak - eqPts)

avgPts   = nTr   > 0 ? sumPts   / nTr   : 0.0
avgWinP  = nWin  > 0 ? sumWinP  / nWin  : 0.0
avgLossP = nLoss > 0 ? sumLossP / nLoss : 0.0
winPct   = nTr   > 0 ? 100.0 * nWin / nTr : 0.0
ptsPF    = sumLossP < 0 ? sumWinP / (-sumLossP) : na
avgBars  = nTr   > 0 ? sumBars  / nTr   : 0
netAfter = sumPts - costPtsRound * nTr
expR     = (avgLossP < 0) ? (avgPts / (-avgLossP)) : na
months   = firstT > 0 ? (time - firstT) / (2629800.0 * 1000.0) : 0.0
trPerMo  = months > 0.3 ? nTr / months : 0.0
trPerDay = months > 0.3 ? nTr / (months * 21.0) : 0.0
netPerMo = months > 0.3 ? netAfter / months : 0.0

if showViz and barstate.islastconfirmedhistory
    var table t = table.new(position.top_right, 2, 16, border_width=1, frame_color=color.gray, frame_width=1)
    table.cell(t, 0, 0,  "HF SWEEP v11", text_color=color.white, bgcolor=color.new(color.blue, 0), text_size=size.small)
    table.cell(t, 1, 0,  "BANKNIFTY pts", text_color=color.white, bgcolor=color.new(color.blue, 0), text_size=size.small)
    table.cell(t, 0, 1,  "Trades", text_size=size.small)
    table.cell(t, 1, 1,  str.tostring(nTr), text_size=size.small)
    table.cell(t, 0, 2,  "Trades / day", text_size=size.small)
    table.cell(t, 1, 2,  str.tostring(trPerDay, "#.##"), text_size=size.small)
    table.cell(t, 0, 3,  "Trades / month", text_size=size.small)
    table.cell(t, 1, 3,  str.tostring(trPerMo, "#.#"), text_size=size.small)
    table.cell(t, 0, 4,  "Win %", text_size=size.small)
    table.cell(t, 1, 4,  str.tostring(winPct, "#.#") + "%", text_size=size.small)
    table.cell(t, 0, 5,  "Total points", text_size=size.small)
    table.cell(t, 1, 5,  str.tostring(sumPts, "#"), text_color=sumPts >= 0 ? color.green : color.red, text_size=size.small)
    table.cell(t, 0, 6,  "Points / trade", text_size=size.small)
    table.cell(t, 1, 6,  str.tostring(avgPts, "#.#"), text_color=avgPts >= 0 ? color.green : color.red, text_size=size.small)
    table.cell(t, 0, 7,  "Avg WIN / LOSS", text_size=size.small)
    table.cell(t, 1, 7,  str.tostring(avgWinP, "#") + " / " + str.tostring(avgLossP, "#"), text_size=size.small)
    table.cell(t, 0, 8,  "Max SL pts used", text_size=size.small)
    table.cell(t, 1, 8,  str.tostring(maxSLused, "#"), text_color=(stopCapPts > 0 and maxSLused > stopCapPts + 1) ? color.red : color.gray, text_size=size.small)
    table.cell(t, 0, 9,  "Points profit factor", text_size=size.small)
    table.cell(t, 1, 9,  na(ptsPF) ? "-" : str.tostring(ptsPF, "#.##"), text_color=(not na(ptsPF) and ptsPF >= 1) ? color.green : color.red, text_size=size.small)
    table.cell(t, 0, 10, "Expectancy (R)", text_size=size.small)
    table.cell(t, 1, 10, na(expR) ? "-" : str.tostring(expR, "#.##"), text_size=size.small)
    table.cell(t, 0, 11, "Best / Worst", text_size=size.small)
    table.cell(t, 1, 11, str.tostring(bestT, "#") + " / " + str.tostring(worstT, "#"), text_size=size.small)
    table.cell(t, 0, 12, "Max DD (pts)", text_size=size.small)
    table.cell(t, 1, 12, str.tostring(maxDDpts, "#"), text_color=color.red, text_size=size.small)
    table.cell(t, 0, 13, "Avg bars held", text_size=size.small)
    table.cell(t, 1, 13, str.tostring(avgBars), text_size=size.small)
    table.cell(t, 0, 14, "Net pts after " + str.tostring(costPtsRound, "#") + "/rt", text_size=size.small)
    table.cell(t, 1, 14, str.tostring(netAfter, "#"), text_color=netAfter >= 0 ? color.green : color.red, text_size=size.small)
    table.cell(t, 0, 15, "Net pts / month", text_size=size.small)
    table.cell(t, 1, 15, str.tostring(netPerMo, "#"), text_color=netPerMo >= 0 ? color.green : color.red, text_size=size.small)

// ============ VISUALS ============
plotshape(showViz and shortEntryBar, "S", shape.triangledown, location.abovebar, color.new(color.red,0),  size=size.tiny, text="S")
plotshape(showViz and longEntryBar,  "L", shape.triangleup,   location.belowbar, color.new(color.teal,0), size=size.tiny, text="L")
plot(strategy.position_size > 0 ? tpLv : strategy.position_size < 0 ? tpSv : na, "TP", color=color.new(color.green,0), style=plot.style_linebr, linewidth=1)
plot(strategy.position_size > 0 ? slLv : strategy.position_size < 0 ? slSv : na, "SL", color=color.new(color.red,0),   style=plot.style_linebr, linewidth=1)
plot(showViz ? smaD : na, "RegSMA", color=color.new(color.blue, 55))
