//@version=5
// ============================================================================
//  BankNifty SMC Pure v8  --  NSE:BANKNIFTY1!  (3-minute)   CLEAN CHART
//
//  !!! TESTED - NO EDGE. Research record only. !!!
//  Built to the "clean slate, pure Smart Money Concepts" brief - no EMA / VWAP
//  / RSI / ATR-direction. Chart shows ONLY this strategy's OB/FVG zones + trades.
//
//  Logic:
//    ORDER FLOW  = market structure. Swing pivots -> a close beyond the last
//                  swing = BOS/CHoCH -> directional bias (+1 / -1).
//    DISPLACEMENT= the leg that breaks structure must span >= dispMlt x the
//                  20-bar average candle range (a real impulse, not drift).
//    FVG         = 3-candle imbalance at the break (optional gate).
//    ORDER BLOCK = last opposite-colour candle before the displacement.
//    ENTRY       = after a valid break, wait for price to retrace into the POI
//                  (OB / FVG / their overlap); enter on a same-direction close.
//    STOP beyond the OB + buffer.  TARGET fixed 2.4R or the opposite swing.
//    Only internal average = SMA(high-low); nothing else, nothing plotted but
//    the zones and the trades.
//
//  RESULT on the 3m tester window (B&H ~ +113k, up-trending):
//    OB / confirm / dispMlt 1.5        : 85 tr | 31% win | PF 0.71 | -9%
//    OB.FVG overlap / dispMlt 2.0      : 21 tr | 33% win | PF 0.52 | -4%
//    FVG / tap entry                   : 125 tr | 22% win | PF 0.42 | -37%
//  Win rate is 22-33% at every setting - identical to all 10 earlier concept
//  families in this project. Pure structure + OB + FVG gives NO edge on
//  BankNifty 3m: a "valid" BOS retrace-into-OB does not reverse/continue any
//  more reliably than a coin flip after cost.
//
//  Concept family #11. Final conclusion for the project: there is no
//  regime-robust, commission-surviving mechanical BankNifty 3m intraday edge
//  in the rule-space explored (EMA pullback, ORB, momentum, MTF confluence,
//  SMC structure, OB pullback, VWAP revert, morning fade, level fakeout,
//  FVG order-flow, and now pure SMC).
// ============================================================================
strategy("BankNifty SMC Pure v8", overlay=true,
     initial_capital=500000,
     default_qty_type=strategy.fixed, default_qty_value=1,
     margin_long=0, margin_short=0,
     commission_type=strategy.commission.percent, commission_value=0.02,
     slippage=5, max_lines_count=500, max_labels_count=500)

// ============ INPUTS ============  (// N)
swLen     = input.int(5, "Swing pivot length", group="Structure") // 0
dispMlt   = input.float(1.5, "Displacement leg >= x avg candle range", step=0.1, group="Structure") // 1
obScan    = input.int(5, "OB: scan back N bars from the break", group="Order block") // 2
useFVG    = input.bool(true, "Require a Fair Value Gap at the break", group="FVG") // 3
poiMode   = input.string("OB", "Enter from POI", options=["OB","FVG","overlap"], group="Entry") // 4
entryMode = input.string("confirm", "Entry trigger", options=["confirm","tap"], group="Entry") // 5
poiExpiry = input.int(25, "POI valid for N bars after the break", group="Entry") // 6
bufMlt    = input.float(0.30, "Stop buffer beyond OB (x avg range)", step=0.05, group="Risk") // 7
tgtMode   = input.string("R", "Target mode", options=["R","swing"], group="Risk") // 8
rFixed    = input.float(2.4, "Fixed R:R (or fallback)", step=0.1, group="Risk") // 9
minRR     = input.float(1.5, "Min R:R to accept a swing target", step=0.1, group="Risk") // 10
maxStopMlt = input.float(3.0, "Reject if stop > x avg range (0=off)", step=0.25, group="Risk") // 11
sessEntry = input.session("0930-1500", "Entry window (IST)", group="Time") // 12
flatTime  = input.session("1515-1530", "Force-flat window", group="Time") // 13
coolBars  = input.int(2, "Cooldown bars after exit", group="Time") // 14
maxPerDay = input.int(6, "Max entries per day", group="Time") // 15
skipExpiry = input.bool(false, "Skip Thursday", group="Time") // 16
beAfterR  = input.float(0.0, "EXP: SL to BE after +R (0=off)", step=0.1, group="Exits") // 17
useTrail  = input.bool(false, "EXP: trail after +1R", group="Exits") // 18
trailMlt  = input.float(1.0, "  trail distance (x avg range)", step=0.1, group="Exits") // 19
avgRngLen = input.int(20, "Avg candle-range length", group="Core") // 20
showViz   = input.bool(true, "Show OB / FVG zones + trades", group="Visuals") // 21

// ============ CORE ============
rng    = high - low
avgRng = ta.sma(rng, avgRngLen)

inEntryWin = not na(time(timeframe.period, sessEntry))
inFlat     = not na(time(timeframe.period, flatTime))
isThu      = dayofweek == dayofweek.thursday
dowOK      = not (skipExpiry and isThu)
newDay = ta.change(time("D")) != 0
var int nDay = 0
if newDay
    nDay := 0

// ============ MARKET STRUCTURE (order flow) ============
ph = ta.pivothigh(high, swLen, swLen)
pl = ta.pivotlow(low, swLen, swLen)
var float lastPH = na
var float lastPL = na
if not na(ph)
    lastPH := ph
if not na(pl)
    lastPL := pl
bosUp = not na(lastPH) and close > lastPH and close[1] <= lastPH
bosDn = not na(lastPL) and close < lastPL and close[1] >= lastPL
var int bias = 0
if bosUp
    bias := 1
if bosDn
    bias := -1

// ============ DISPLACEMENT + FVG + OB at the break ============
dispUpLeg = high - math.min(low, low[1], low[2], low[3])
dispDnLeg = math.max(high, high[1], high[2], high[3]) - low
dispUpOK  = dispUpLeg >= dispMlt * avgRng
dispDnOK  = dispDnLeg >= dispMlt * avgRng

fvgUpNow = (low > high[2]) or (low[1] > high[3])
fvgDnNow = (high < low[2]) or (high[1] < low[3])
fvgUpLo  = low > high[2] ? high[2] : high[3]
fvgUpHi  = low > high[2] ? low : low[1]
fvgDnLo  = high < low[2] ? high : high[1]
fvgDnHi  = high < low[2] ? low[2] : low[3]

var float poiUpLo = na
var float poiUpHi = na
var int   poiUpBar = na
var bool  poiUpUsed = false
var float poiDnLo = na
var float poiDnHi = na
var int   poiDnBar = na
var bool  poiDnUsed = false

if bosUp and dispUpOK and (not useFVG or fvgUpNow)
    float _obl = na
    float _obh = na
    for i = 1 to obScan
        if na(_obl) and close[i] < open[i]
            _obl := low[i]
            _obh := high[i]
    _fl = fvgUpNow ? fvgUpLo : na
    _fh = fvgUpNow ? fvgUpHi : na
    float _lo = na
    float _hi = na
    if poiMode == "OB"
        _lo := _obl
        _hi := _obh
    else if poiMode == "FVG"
        _lo := _fl
        _hi := _fh
    else
        _lo := (not na(_obl) and not na(_fl)) ? math.max(_obl, _fl) : na
        _hi := (not na(_obh) and not na(_fh)) ? math.min(_obh, _fh) : na
    if not na(_lo) and not na(_hi) and _hi > _lo
        poiUpLo := _lo
        poiUpHi := _hi
        poiUpBar := bar_index
        poiUpUsed := false

if bosDn and dispDnOK and (not useFVG or fvgDnNow)
    float _obl2 = na
    float _obh2 = na
    for i = 1 to obScan
        if na(_obl2) and close[i] > open[i]
            _obl2 := low[i]
            _obh2 := high[i]
    _fl2 = fvgDnNow ? fvgDnLo : na
    _fh2 = fvgDnNow ? fvgDnHi : na
    float _lo2 = na
    float _hi2 = na
    if poiMode == "OB"
        _lo2 := _obl2
        _hi2 := _obh2
    else if poiMode == "FVG"
        _lo2 := _fl2
        _hi2 := _fh2
    else
        _lo2 := (not na(_obl2) and not na(_fl2)) ? math.max(_obl2, _fl2) : na
        _hi2 := (not na(_obh2) and not na(_fh2)) ? math.min(_obh2, _fh2) : na
    if not na(_lo2) and not na(_hi2) and _hi2 > _lo2
        poiDnLo := _lo2
        poiDnHi := _hi2
        poiDnBar := bar_index
        poiDnUsed := false

if not na(poiUpBar) and (bias != 1 or close < poiUpLo or (bar_index - poiUpBar) > poiExpiry)
    poiUpBar := na
if not na(poiDnBar) and (bias != -1 or close > poiDnHi or (bar_index - poiDnBar) > poiExpiry)
    poiDnBar := na

// ============ ENTRY TRIGGERS ============
tapUp = not na(poiUpBar) and not poiUpUsed and low <= poiUpHi
tapDn = not na(poiDnBar) and not poiDnUsed and high >= poiDnLo
trigUp = tapUp and (entryMode == "tap" or (close > open and close > poiUpLo))
trigDn = tapDn and (entryMode == "tap" or (close < open and close < poiDnHi))

// ============ RISK ============
var int lastExitBar = na
if strategy.position_size == 0 and strategy.position_size[1] != 0
    lastExitBar := bar_index
coolOK = coolBars == 0 or na(lastExitBar) or (bar_index - lastExitBar) > coolBars
baseOK = strategy.position_size == 0 and coolOK and inEntryWin and not inFlat and dowOK and nDay < maxPerDay

float slL = na
float slS = na
if trigUp
    slL := poiUpLo - bufMlt * avgRng
if trigDn
    slS := poiDnHi + bufMlt * avgRng
rkL = not na(slL) ? close - slL : na
rkS = not na(slS) ? slS - close : na

float tgtL = na
float tgtS = na
if trigUp and not na(rkL) and rkL > 0
    tgtL := (tgtMode == "swing" and not na(lastPH) and lastPH > close + minRR * rkL) ? lastPH : close + rFixed * rkL
if trigDn and not na(rkS) and rkS > 0
    tgtS := (tgtMode == "swing" and not na(lastPL) and lastPL < close - minRR * rkS) ? lastPL : close - rFixed * rkS

okL = trigUp and bias == 1 and not na(rkL) and rkL > 0 and (maxStopMlt == 0 or rkL <= maxStopMlt * avgRng) and not na(tgtL) and tgtL > close
okS = trigDn and bias == -1 and not na(rkS) and rkS > 0 and (maxStopMlt == 0 or rkS <= maxStopMlt * avgRng) and not na(tgtS) and tgtS < close

// ============ ENTRIES ============
var float slLv = na
var float tpLv = na
var float slSv = na
var float tpSv = na
var float riskInit = na

if baseOK and okL
    strategy.entry("Long", strategy.long)
    slLv := slL
    tpLv := tgtL
    nDay := nDay + 1
    poiUpUsed := true
if baseOK and okS
    strategy.entry("Short", strategy.short)
    slSv := slS
    tpSv := tgtS
    nDay := nDay + 1
    poiDnUsed := true

longEntryBar  = strategy.position_size > 0 and strategy.position_size[1] <= 0
shortEntryBar = strategy.position_size < 0 and strategy.position_size[1] >= 0
if longEntryBar
    riskInit := strategy.position_avg_price - slLv
if shortEntryBar
    riskInit := slSv - strategy.position_avg_price

// ============ EXITS ============
if strategy.position_size > 0
    _s = slLv
    if beAfterR > 0 and not na(riskInit) and high >= strategy.position_avg_price + beAfterR * riskInit
        _s := math.max(_s, strategy.position_avg_price)
    if useTrail and not na(riskInit) and high >= strategy.position_avg_price + riskInit
        _s := math.max(_s, strategy.position_avg_price, high - trailMlt * avgRng)
    slLv := _s
    strategy.exit("Exit L", from_entry="Long", stop=slLv, limit=tpLv)
    if inFlat
        strategy.close("Long", comment="EOD")
if strategy.position_size < 0
    _s = slSv
    if beAfterR > 0 and not na(riskInit) and low <= strategy.position_avg_price - beAfterR * riskInit
        _s := math.min(_s, strategy.position_avg_price)
    if useTrail and not na(riskInit) and low <= strategy.position_avg_price - riskInit
        _s := math.min(_s, strategy.position_avg_price, low + trailMlt * avgRng)
    slSv := _s
    strategy.exit("Exit S", from_entry="Short", stop=slSv, limit=tpSv)
    if inFlat
        strategy.close("Short", comment="EOD")

// ============ VISUALS ============
plot(showViz and not na(poiUpBar) ? poiUpHi : na, "POI up top", color=color.new(color.teal,0), style=plot.style_linebr, linewidth=1)
plot(showViz and not na(poiUpBar) ? poiUpLo : na, "POI up bot", color=color.new(color.teal,50), style=plot.style_linebr, linewidth=1)
plot(showViz and not na(poiDnBar) ? poiDnHi : na, "POI dn top", color=color.new(color.red,50), style=plot.style_linebr, linewidth=1)
plot(showViz and not na(poiDnBar) ? poiDnLo : na, "POI dn bot", color=color.new(color.red,0), style=plot.style_linebr, linewidth=1)
bgcolor(showViz and bias == 1 ? color.new(color.green, 94) : bias == -1 ? color.new(color.red, 94) : na)
plotshape(showViz and longEntryBar,  "L", shape.triangleup,   location.belowbar, color.new(color.teal,0), size=size.small, text="L")
plotshape(showViz and shortEntryBar, "S", shape.triangledown, location.abovebar, color.new(color.red,0),  size=size.small, text="S")
plot(strategy.position_size > 0 ? tpLv : strategy.position_size < 0 ? tpSv : na, "TP", color=color.new(color.green,0), style=plot.style_linebr, linewidth=1)
plot(strategy.position_size > 0 ? slLv : strategy.position_size < 0 ? slSv : na, "SL", color=color.new(color.red,0),   style=plot.style_linebr, linewidth=1)
